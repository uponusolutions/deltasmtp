// DotReader implements a smtp dot reader.
type DotReader struct {
        r     *bufio.Reader
        state int

        limited bool
        n       int64 // Maximum bytes remaining
}

// NewDotReader creates a new smtp dot reader.
func NewDotReader(r *bufio.Reader, maxMessageBytes int) *DotReader {
        dr := &DotReader{
                r: r,
        }

        if maxMessageBytes > 0 {
                dr.limited = true
                dr.n = int64(maxMessageBytes)
        }

        return dr
}

func (r *DotReader) Read(b []byte) (n int, err error) {
        if r.limited {
                if r.n <= 0 {
                        return 0, deltasmtp.ErrDataTooLarge
                }
                if int64(len(b)) > r.n {
                        b = b[0:r.n]
                }
        }

        // Code below is taken from net/textproto with only one modification to
        // not rewrite CRLF -> LF.

        // Run data through a simple state machine to
        // elide leading dots and detect End-of-Data (<CR><LF>.<CR><LF>) line.
        const (
                stateBeginLine = iota // beginning of line; initial state; must be zero
                stateDot              // read . at beginning of line
                stateDotCR            // read .\r at beginning of line
                stateCR               // read \r (possibly at end of line)
                stateData             // reading data in middle of line
                stateEOF              // reached .\r\n end marker line
        )
        for n < len(b) && r.state != stateEOF {
                var c byte
                c, err = r.r.ReadByte()
                if err != nil {
                        if err == io.EOF {
                                err = io.ErrUnexpectedEOF
                        }
                        break
                }
                switch r.state {
                case stateBeginLine:
                        if c == '.' {
                                r.state = stateDot
                                continue
                        }
                        if c == '\r' {
                                r.state = stateCR
                                break
                        }
                        r.state = stateData
                case stateDot:
                        if c == '\r' {
                                r.state = stateDotCR
                                continue
                        }
                        r.state = stateData
                case stateDotCR:
                        if c == '\n' {
                                r.state = stateEOF
                                continue
                        }
                        r.state = stateData
                case stateCR:
                        if c == '\n' {
                                r.state = stateBeginLine
                                break
                        }
                        r.state = stateData
                case stateData:
                        if c == '\r' {
                                r.state = stateCR
                        }
                default:
                }
                b[n] = c
                n++
        }
        if err == nil && r.state == stateEOF {
                err = io.EOF
        }

        if r.limited {
                r.n -= int64(n)
        }
        return n, err
}

// Package client implements the client of the Simple Mail Transfer Protocol as defined in RFC 5321.
// It provides basic functionality you need to use deltasmtp.
//
// It also implements the following extensions:
//
//   - 8BITMIME (RFC 1652)
//   - ENHANCEDSTATUSCODES (RFC 2034)
//   - AUTH (RFC 2554)
//   - DELIVERBY (RFC 2852)
//   - CHUNKING (RFC 3030)
//   - BINARYMIME (RFC 3030)
//   - STARTTLS (RFC 3207)
//   - DSN (RFC 3461, RFC 6533)
//   - SMTPUTF8 (RFC 6531)
//   - MT-PRIORITY (RFC 6710)
//   - RRVS (RFC 7293)
//   - REQUIRETLS (RFC 8689)
//

// Client implements a SMTP Client with .
type Client struct {
        cfg Config

        session  session
        deadline deadline.Renewer // command deadline of session.conn
        receiver *smtpreader.Receiver
        sender   *smtpwriter.Sender

        cmdBuf bytes.Buffer // see cmdBuffer

        // The writers of the message content, reused for every message. The
        // ContentCloser of a message is done with its writer once it is closed.
        dotWriter          smtpwriter.Dot
        bdatWriter         smtpwriter.Bdat
        bdatBufferedWriter smtpwriter.BdatBuffered
        bdatReply          func() error // see bdatReplyFunc
}

// session is the state of the current or, after disconnect, the last connection.
//
// Only conn is reset on disconnect, the rest is kept to be able to use
// ServerAddress(), ServerName() and Extension() after disconnect. The whole
// session is reset when dialing, so it never describes a server of a previous
// connection.
type session struct {
        // keep a reference to the connection so it can be used to create a TLS
        // connection later, nil if disconnected
        conn       net.Conn
        address    string            // Format address:port, set on dial.
        name       string            // server greet name, set when the server greets the client
        ext        map[string]string // supported extensions of the server after ehlo
        pipelining pipeliningState   // reset when ext is set
}

type pipeliningState struct {
        // How many responses are pending?
        // Is counted up when a request is sent without retrieving the response.
        pending int
        // Marks if the group has concluded through a command which must be the last command in a group.
        // If it is set, the pending responses needs to be consumed before sending commands again.
        concluded bool
}

type pipeliningType int

const (
        pipeliningTypeForbidden pipeliningType = iota
        pipeliningTypeAnywhere
        pipeliningTypeLast
)

// New returns a new smtp client.
func New(opts ...Option) *Client {
        cfg := DefaultConfig()

        for _, o := range opts {
                o(&cfg)
        }

        return NewFromConfig(cfg)
}

// NewFromConfig returns a new smtp client from existing config.
func NewFromConfig(cfg Config) *Client {
        return &Client{
                cfg: cfg,
        }
}

// MailOptions contains parameters for the MAIL command.
type MailOptions struct {
        // Size of the body. Can be 0 if not specified by client.
        Size int64

        // TLS is required for the message transmission.
        //
        // The message should be rejected if it can't be transmitted
        // with TLS.
        RequireTLS bool

        // SMTPUTF8 sends the SMTPUTF8 parameter (RFC 6531), fails if the server
        // doesn't support it.
        SMTPUTF8 bool

        // Value of RET= argument, FULL or HDRS.
        Return deltasmtp.DSNReturn

        // Envelope identifier set by the client.
        EnvelopeID string

        // Accepted Domain from Exchange Online, e.g. from OutgoingConnector
        XOORG *string

        // The authorization identity asserted by the message sender in decoded
        // form with angle brackets stripped.
        //
        // nil value indicates missing AUTH, non-nil empty string indicates
        // AUTH=<>.
        //
        // Defined in RFC 4954.
        Auth *string

        // Value of BY= argument or nil if unset.
        DeliverBy *deltasmtp.DeliverByOptions

        // Value of MT-PRIORITY= or nil if unset.
        MTPriority *int
}

// VrfyOptions contains parameters for the VRFY command.
type VrfyOptions struct {
        // SMTPUTF8 sends the SMTPUTF8 parameter (RFC 6531), fails if the server
        // doesn't support it.
        SMTPUTF8 bool
}

// Dial returns a connection to an SMTP server at addr. The addr must
// include a port, as in "mail.example.com:smtp".
// An existing connection is closed first.
// If an error occurs, the connection is closed.
func (c *Client) Dial(ctx context.Context, addr string) error {
        return c.dial(ctx, addr, false, nil)
}

// DialTLS returns a connection to an SMTP server at addr via TLS.
// The addr must include a port, as in "mail.example.com:smtps".
//
// A nil tlsConfig is equivalent to a zero tls.Config.
// An existing connection is closed first.
// If an error occurs, the connection is closed.
func (c *Client) DialTLS(ctx context.Context, config *tls.Config, addr string) error {
        return c.dial(ctx, addr, true, config)
}

func (c *Client) dial(ctx context.Context, addr string, implicitTLS bool, config *tls.Config) error {
        _ = c.Close()
        c.session = session{}

        dialer := &net.Dialer{Timeout: c.cfg.dialTimeout, Control: c.cfg.dialControl}
        conn, err := dialer.DialContext(ctx, "tcp", addr)
        if err != nil {
                return err
        }
        c.session.address = addr

        if implicitTLS {
                host, _, _ := net.SplitHostPort(addr)
                if conn, err = c.handshake(ctx, conn, config, host); err != nil {
                        return err
                }
        }

        return c.initConn(conn, true)
}

// handshake starts TLS on conn within the TLS handshake timeout, conn is closed on
// error. A nil config is equivalent to a zero tls.Config, serverName is used if the
// config has none.
func (c *Client) handshake(ctx context.Context, conn net.Conn, config *tls.Config, serverName string) (net.Conn, error) {
        if config == nil {
                config = &tls.Config{}
        }
        if config.ServerName == "" && serverName != "" {
                // Make a copy to avoid polluting argument
                config = config.Clone()
                config.ServerName = serverName
        }

        tlsConn := tls.Client(conn, config)
        deadline.Set(tlsConn, c.cfg.tlsHandshakeTimeout)
        if err := tlsConn.HandshakeContext(ctx); err != nil {
                _ = conn.Close()
                return nil, err
        }
        return tlsConn, nil
}

// initConn sets the underlying network connection for the client,
// expect greeting from the server if enabled and
// calls hello to finish basic setup
func (c *Client) initConn(conn net.Conn, expectGreet bool) error {
        c.setConn(conn)

        if expectGreet {
                if err := c.greet(); err != nil {
                        return err
                }
        }

        return c.Hello()
}

// setConn sets the underlying network connection.
func (c *Client) setConn(conn net.Conn) {
        c.session.conn = conn
        c.deadline.Reset()

        if c.receiver != nil {
                c.receiver.Reset(conn)
        } else {
                c.receiver = smtpreader.NewReceiver(conn, c.cfg.readerSize, c.cfg.maxLineLength, c.cfg.maxResponseLines)
        }

        if c.sender != nil {
                c.sender.Reset(conn)
        } else {
                c.sender = smtpwriter.NewSender(conn, c.cfg.writerSize)
        }
}

// Close closes the connection.
func (c *Client) Close() (err error) {
        if c.session.conn == nil {
                return nil
        }
        c.receiver.Reset(nil)
        c.sender.Reset(nil)
        err = c.session.conn.Close()
        c.session.conn = nil
        return err
}

// receiveResponse reads a reply.
//
// Any error other than an *deltasmtp.Status reply (e.g. a malformed or too long reply,
// a timeout) may leave the rest of the reply unread. The next command would read
// it as its own reply, so the connection is closed. So it is after a 421 reply, the
// server closes the connection (RFC 5321 section 3.8).
func (c *Client) receiveResponse() (*deltasmtp.Status, error) {
        status, err := c.receiver.ReadResponse()
        if err != nil || status.Code == 421 {
                _ = c.Close()
        }
        return status, err
}

// receiveResponseValid reads a reply and returns it as error if the code isn't expected.
// The connection is closed like in receiveResponse.
func (c *Client) receiveResponseValid(expectCode int) error {
        if !smtpreader.IsCodeUnexpected(421, expectCode) { // the fast path wouldn't tell a 421
                status, err := c.receiveResponse()
                if err == nil && smtpreader.IsCodeUnexpected(status.Code, expectCode) {
                        return status
                }
                return err
        }
        err := c.receiver.ReadResponseValid(expectCode)
        if st, ok := errors.AsType[*deltasmtp.Status](err); err != nil && (!ok || st.Code == 421) {
                _ = c.Close()
        }
        return err
}

// Greet reads the greeting of the server
// if an error occurred the connection is closed
func (c *Client) greet() error {
        // Initial greeting timeout. RFC 5321 recommends 5 minutes.
        c.deadline.Set(c.session.conn, c.cfg.commandTimeout)

        status, err := c.receiveResponse()
        if err != nil {
                return err
        }

        // status unexpected or no message received
        if status.Code != 220 || len(status.Lines) == 0 {
                _ = c.Close()
                return status
        }

        if idx := strings.IndexRune(status.Lines[0], ' '); idx >= 0 {
                c.session.name = status.Lines[0][:idx]
        } else {
                c.session.name = status.Lines[0]
        }

        return nil
}

// Hello runs a hello exchange
// if an error occurred the connection is closed
func (c *Client) Hello() error {
        // verify if local name is valid
        if strings.ContainsAny(c.cfg.localName, "\n\r") {
                _ = c.Close()
                return errors.New("smtp: the local name must not contain CR or LF")
        }

        err := c.ehlo()

        if status, ok := errors.AsType[*deltasmtp.Status](err); ok && (status.Code == 500 || status.Code == 502) {
                // The server doesn't support EHLO, fallback to HELO
                err = c.helo()
        }

        if err != nil {
                _ = c.Close()
        }

        return err
}

func (c *Client) setExt(ext map[string]string) {
        c.session.ext = ext
        c.session.pipelining.concluded = false
        c.session.pipelining.pending = 0
}

func (c *Client) consumeResponse() error {
        if !c.PipeliningActive() {
                return ErrPipeliningNotEnabled
        }

        if c.session.pipelining.pending == 0 {
                return ErrPipeliningNothingPending
        }

        // first consume concludes group
        c.session.pipelining.concluded = true
        c.session.pipelining.pending--

        if c.session.pipelining.pending == 0 {
                c.session.pipelining.concluded = false
        }

        return nil
}

// cmd is a convenience function that sends a command and returns the response
// does not support or handle anything related to pipelining
func (c *Client) cmd(expectCode int, message string) (*deltasmtp.Status, error) {
        if c.session.conn == nil {
                return nil, ErrNotConnected
        }

        c.deadline.Set(c.session.conn, c.cfg.commandTimeout)

        _, err := c.sender.WriteString(message)
        if err != nil {
                return nil, err
        }

        _, err = c.sender.Write(deltasmtp.Crnl)
        if err != nil {
                return nil, err
        }

        err = c.sender.Flush()
        if err != nil {
                return nil, err
        }

        status, err := c.receiveResponse()
        if err != nil {
                return nil, err
        }

        if smtpreader.IsCodeUnexpected(status.Code, expectCode) {
                return nil, status
        }

        return status, nil
}

// cmdValid is a convenience function that sends a command and returns an error if the expected code doesn't match
// can handle pipelining
func (c *Client) cmdValid(pType pipeliningType, expectCode int, message string) error {
        if err := c.cmdBegin(len(message)); err != nil {
                return err
        }
        if _, err := c.sender.WriteString(message); err != nil {
                return err
        }
        return c.cmdEnd(pType, expectCode)
}

// cmdBuffer returns the emptied buffer to build a command in, which is sent by
// cmdValidBuffer. The buffer is reused, building a command doesn't allocate.
func (c *Client) cmdBuffer() *bytes.Buffer {
        c.cmdBuf.Reset()
        return &c.cmdBuf
}

// cmdValidBuffer is cmdValid for the command built in cmdBuffer.
func (c *Client) cmdValidBuffer(pType pipeliningType, expectCode int) error {
        if err := c.cmdBegin(c.cmdBuf.Len()); err != nil {
                return err
        }
        if _, err := c.sender.Write(c.cmdBuf.Bytes()); err != nil {
                return err
        }
        return c.cmdEnd(pType, expectCode)
}

// cmdBegin prepares sending a command of length n, see cmdValid.
func (c *Client) cmdBegin(n int) error {
        if c.session.conn == nil {
                return ErrNotConnected
        }

        c.deadline.Set(c.session.conn, c.cfg.commandTimeout)

        if c.PipeliningActive() {
                // group is concluded, please consume all responses
                if c.session.pipelining.concluded {
                        return ErrPipeliningGroupConcluded
                }
                // RFC 2920
                // Client SMTP implementations MAY elect to operate in a nonblocking
                // fashion, processing server responses immediately upon receipt, even
                // if there is still data pending transmission from the client's
                // previous TCP send operation. If nonblocking operation is not
                // supported, however, client SMTP implementations MUST also check the
                // TCP window size and make sure that each group of commands fits
                // entirely within the window. The window size is usually, but not
                // always, 4K octets.  Failure to perform this check can lead to
                // deadlock conditions.

                // Need to flush to prevent congestion and conclude the group forcefully.
                // If the buffer can not hold the first request, ignore congestion.
                // Typically the 4k buffio is enough to make it fast, it don't expect any buffers from tcp.
                // Outcomment to see TestClient_SendMailDirectManyRcptsPipelining failing.
                if n+2 > c.sender.Available() && c.session.pipelining.pending > 0 {
                        err := c.sender.Flush()
                        if err != nil {
                                return err
                        }
                        c.session.pipelining.concluded = true
                        return ErrPipeliningCongestion
                }
        }

        return nil
}

// cmdEnd finishes sending a command, see cmdValid.
func (c *Client) cmdEnd(pType pipeliningType, expectCode int) error {
        _, err := c.sender.Write(deltasmtp.Crnl)
        if err != nil {
                return err
        }

        // Only commands allowed anywhere in a pipelining group stay buffered,
        // all others need their response now or conclude the group.
        if !c.PipeliningActive() || pType != pipeliningTypeAnywhere {
                err = c.sender.Flush()
                if err != nil {
                        return err
                }
        }

        if c.PipeliningActive() && pType > pipeliningTypeForbidden {
                c.session.pipelining.pending++
                if pType == pipeliningTypeLast {
                        c.session.pipelining.concluded = true
                }
                return nil
        }

        return c.receiveResponseValid(expectCode)
}

// Conclude concludes the current pipelining group.
// Do not call if pipelining is not active.
func (c *Client) Conclude() error {
        if c.session.conn == nil {
                return ErrNotConnected
        }

        c.deadline.Set(c.session.conn, c.cfg.commandTimeout)

        if !c.PipeliningActive() {
                return ErrPipeliningNotEnabled
        }
        c.session.pipelining.concluded = true
        return c.sender.Flush()
}

func (c *Client) readResponseValid(expectCode int) error {
        if err := c.awaitResponse(); err != nil {
                return err
        }
        return c.receiveResponseValid(expectCode)
}

// awaitResponse consumes a pending response and flushes the commands before reading it.
func (c *Client) awaitResponse() error {
        if c.session.conn == nil {
                return ErrNotConnected
        }
        if err := c.consumeResponse(); err != nil {
                return err
        }
        c.deadline.Set(c.session.conn, c.cfg.commandTimeout)
        // Probably already flushed if the group has ended.
        return c.sender.Flush()
}

// ClearResponses consumes x pending responses from pipelining, 0 means all, -1 all
// except 1. A 421 reply is returned without reading further, the connection is closed
// (e.g. Postfix after too many errors).
func (c *Client) ClearResponses(x int) error {
        if !c.PipeliningActive() {
                return ErrPipeliningNotEnabled
        }

        if x <= 0 {
                x += c.session.pipelining.pending
        }

        for range x {
                if err := c.awaitResponse(); err != nil {
                        return err
                }
                status, err := c.receiveResponse()
                if err != nil {
                        return err
                }
                if status.Code == 421 {
                        return status
                }
        }
        return nil
}

// helo sends the HELO greeting to the server. It should be used only when the
// server does not support ehlo.
func (c *Client) helo() error {
        c.setExt(nil)
        return c.cmdValid(pipeliningTypeForbidden, 250, "HELO "+c.cfg.localName)
}

// ehlo sends the EHLO (extended hello) greeting to the server. It
// should be the preferred greeting for servers that support it.
func (c *Client) ehlo() error {
        status, err := c.cmd(250, "EHLO "+c.cfg.localName)
        if err != nil {
                return err
        }
        ext := make(map[string]string, max(len(status.Lines)-1, 0))

        if len(status.Lines) > 1 {
                for _, line := range status.Lines[1:] {
                        key, value, _ := strings.Cut(line, " ")
                        ext[key] = value
                }
        }
        c.setExt(ext)
        return err
}

// StartTLS sends the STARTTLS command and encrypts all further communication.
// Only servers that advertise the STARTTLS extension support this function.
//
// A nil config is equivalent to a zero tls.Config.
//
// If the server rejects the command it returns the error and the connection is not closed.
// If the server accepts the command but the tls handshake fails, the connection is closed.
func (c *Client) StartTLS(config *tls.Config, serverName string) error {
        if c.PipeliningActive() && c.session.pipelining.pending > 0 {
                return ErrPipeliningNoPendingRequired
        }

        if err := c.cmdValid(pipeliningTypeForbidden, 220, "STARTTLS"); err != nil {
                return err
        }

        conn, err := c.handshake(context.Background(), c.session.conn, config, serverName)
        if err != nil {
                _ = c.Close()
                return err
        }

        return c.initConn(conn, false)
}

// TLSConnectionState returns the client's TLS connection state.
// The return values are their zero values if STARTTLS did
// not succeed.
func (c *Client) TLSConnectionState() (tls.ConnectionState, bool) {
        tc, ok := c.session.conn.(*tls.Conn)
        if !ok {
                return tls.ConnectionState{}, false
        }
        return tc.ConnectionState(), true
}

// PipeliningActive return if pipelining is active.
func (c *Client) PipeliningActive() bool {
        _, ok := c.session.ext["PIPELINING"]
        return ok && c.cfg.pipelining && c.session.conn != nil
}

// ChunkingActive reports whether Content and ContentResponse use BDAT instead of DATA.
// With pipelining active, Content doesn't send BDAT, ContentResponse does.
func (c *Client) ChunkingActive() bool {
        _, ok := c.session.ext["CHUNKING"]
        return ok && c.cfg.chunking
}

// Verify checks the validity of an email address on the server.
// If Verify returns nil, the server answered 250, 251 or 252 (RFC 5321 section 3.5.3).
// 252 means the server can't verify the address but accepts mail for it,
// many servers answer every VRFY with it for security reasons.
// A non-nil return does not necessarily indicate an invalid address.
//
// If server returns an error, it will be of type *deltasmtp.Status.
func (c *Client) Verify(addr string, opts *VrfyOptions) error {
        if err := validateLine(addr); err != nil {
                return err
        }

        sb := c.cmdBuffer()
        sb.WriteString("VRFY ")
        sb.WriteString(addr)

        if opts != nil && opts.SMTPUTF8 {
                if _, ok := c.session.ext["SMTPUTF8"]; !ok {
                        return refusedError("smtp: server does not support SMTPUTF8")
                }
                sb.WriteString(" SMTPUTF8")
        }

        return c.cmdValidBuffer(pipeliningTypeLast, 25)
}

// VerifyResponse returns the result of previous send verify command if pipelining is enabled
func (c *Client) VerifyResponse() error {
        return c.readResponseValid(25)
}

// Auth authenticates a client using the provided authentication mechanism.
// Only servers that advertise the AUTH extension support this function.
//
// If server returns an error, it will be of type *deltasmtp.
func (c *Client) Auth(saslClient sasl.Client) error {
        if c.PipeliningActive() && c.session.pipelining.pending > 0 {
                return ErrPipeliningNoPendingRequired
        }

        if saslClient == nil {
                return refusedError("smtp: SASL client is missing")
        }

        encoding := base64.StdEncoding
        mech, resp, err := saslClient.Start()
        if err != nil {
                return err
        }
        var resp64 []byte
        if len(resp) > 0 {
                resp64 = make([]byte, encoding.EncodedLen(len(resp)))
                encoding.Encode(resp64, resp)
        } else if resp != nil {
                resp64 = []byte{'='}
        }

        var status *deltasmtp.Status

        // 512 - len("AUTH") - 2 * len(" ") - len(\r\n)
        if len(mech)+len(resp64) > 504 || len(resp64) == 0 {
                // The initial response (if any) does not fit in the 512-octet command line (RFC 5321
                // section 4.5.3.1.4), so send it as the reply to the first challenge instead
                // (RFC 4954 section 4). https://github.com/emersion/go-smtp/issues/301
                status, err = c.cmd(0, "AUTH "+mech)
                if err == nil && status.Code == 334 && len(resp64) > 0 {
                        // The server must ask for the initial response with an empty challenge;
                        // anything else would make us drop the initial response, so abort.
                        if strings.Join(status.Lines, "") != "" {
                                return c.authAbort(errors.New("smtp: server sent a challenge instead of asking for the initial response"))
                        }
                        status, err = c.cmd(0, string(resp64))
                }
        } else {
                status, err = c.cmd(0, "AUTH "+mech+" "+string(resp64))
        }

        if err != nil {
                return err
        }

        for {
                switch status.Code {
                case 334:
                        // failed to decode answer => abort
                        msg, err := encoding.DecodeString(strings.Join(status.Lines, "\n"))
                        if err != nil {
                                return c.authAbort(err)
                        }
                        // sasl returned error or nil instead
                        resp, err = saslClient.Next(msg)
                        if err != nil {
                                return c.authAbort(err)
                        }
                        resp64 = make([]byte, encoding.EncodedLen(len(resp)))
                        encoding.Encode(resp64, resp)
                        status, err = c.cmd(0, string(resp64))
                        if err != nil {
                                return err
                        }
                case 235:
                        return nil
                default:
                        // status like 501 mean authentication aborted,
                        // but essentially everything except 335 and 235 can be interpreted as such.
                        return status
                }
        }
}

func (c *Client) authAbort(err error) error {
        // If abort fails, the state is strange
        if errAbort := c.cmdValid(pipeliningTypeForbidden, 501, "*"); errAbort != nil {
                err = errors.Join(err, errAbort)
        }
        return err
}

// Mail issues a MAIL command to the server using the provided email address.
// If the server supports the 8BITMIME extension, Mail adds the BODY=8BITMIME
// parameter.
// This initiates a mail transaction and is followed by one or more Rcpt calls.
//
// If opts is not nil, MAIL arguments provided in the structure will be added
// to the command. Handling of unsupported options depends on the extension.
//
// If server returns an error, it will be of type *deltasmtp.
//
//nolint:revive // cognitive-complexity: one branch per MAIL parameter, splitting hides the order.
func (c *Client) Mail(from string, opts *MailOptions) error {
        if err := validateLine(from); err != nil {
                return err
        }

        sb := c.cmdBuffer()
        sb.WriteString("MAIL FROM:<")
        sb.WriteString(from)
        sb.WriteString(">")
        if _, ok := c.session.ext["8BITMIME"]; ok {
                sb.WriteString(" BODY=8BITMIME")
        }
        if _, ok := c.session.ext["SIZE"]; ok && opts != nil && opts.Size != 0 {
                var num [20]byte
                sb.WriteString(" SIZE=")
                sb.Write(strconv.AppendInt(num[:0], opts.Size, 10))
        }
        if opts != nil && opts.RequireTLS {
                if _, ok := c.session.ext["REQUIRETLS"]; !ok {
                        return refusedError("smtp: server does not support REQUIRETLS")
                }
                sb.WriteString(" REQUIRETLS")
        }
        if opts != nil && opts.SMTPUTF8 {
                if _, ok := c.session.ext["SMTPUTF8"]; !ok {
                        return refusedError("smtp: server does not support SMTPUTF8")
                }
                sb.WriteString(" SMTPUTF8")
        }
        if _, ok := c.session.ext["DSN"]; ok && opts != nil {
                switch opts.Return {
                case deltasmtp.DSNReturnFull, deltasmtp.DSNReturnHeaders:
                        sb.WriteString(" RET=")
                        sb.WriteString(string(opts.Return))
                case "":
                        // This space is intentionally left blank
                default:
                        return refusedError("smtp: Unknown RET parameter value")
                }
                if opts.EnvelopeID != "" {
                        if !parse.IsPrintableASCII(opts.EnvelopeID) {
                                return refusedError("smtp: Malformed ENVID parameter value")
                        }
                        sb.WriteString(" ENVID=")
                        sb.WriteString(encodeXtext(opts.EnvelopeID))
                }
        }
        if opts != nil && opts.Auth != nil {
                if _, ok := c.session.ext["AUTH"]; ok {
                        sb.WriteString(" AUTH=")
                        sb.WriteString(encodeXtext(*opts.Auth))
                }
                // We can safely discard parameter if server does not support AUTH.
        }

        if opts != nil && opts.XOORG != nil {
                if _, ok := c.session.ext["XOORG"]; ok {
                        sb.WriteString(" XOORG=")
                        sb.WriteString(encodeXtext(*opts.XOORG))
                }
                // We can safely discard parameter if server does not support AUTH.
        }
        if _, ok := c.session.ext["DELIVERBY"]; ok && opts != nil && opts.DeliverBy != nil {
                if opts.DeliverBy.Mode == deltasmtp.DeliverByReturn &&
                        (opts.DeliverBy.Seconds < 1 || opts.DeliverBy.Seconds > 999999999) {
                        return refusedError("smtp: DELIVERBY mode must be between 1 and 999999999 with return mode")
                }
                if opts.DeliverBy.Mode == deltasmtp.DeliverByNotify &&
                        (opts.DeliverBy.Seconds < -999999999 || opts.DeliverBy.Seconds > 999999999) {
                        return refusedError("smtp: DELIVERBY mode must be between -999999999 and 999999999 with notify mode")
                }
                arg := fmt.Sprintf(" BY=%d;%s", opts.DeliverBy.Seconds, opts.DeliverBy.Mode)
                if opts.DeliverBy.Trace {
                        arg += "T"
                }
                sb.WriteString(arg)
        }
        if _, ok := c.session.ext["MT-PRIORITY"]; ok && opts != nil && opts.MTPriority != nil {
                if *opts.MTPriority < -9 || *opts.MTPriority > 9 {
                        return refusedError("smtp: MT-PRIORITY must be between -9 and 9")
                }
                sb.WriteString(" MT-PRIORITY=")
                sb.WriteString(strconv.Itoa(*opts.MTPriority))
        }
        return c.cmdValidBuffer(pipeliningTypeAnywhere, 250)
}

// MailResponse returns the result of previous send mail command if pipelining is enabled
func (c *Client) MailResponse() error {
        return c.readResponseValid(250)
}

// Rcpt issues a RCPT command to the server using the provided email address.
// A call to Rcpt must be preceded by a call to Mail and may be followed by
// a Data call or another Rcpt call.
//
// If opts is not nil, RCPT arguments provided in the structure will be added
// to the command. Handling of unsupported options depends on the extension.
//
// If server returns an error, it will be of type *deltasmtp.
func (c *Client) Rcpt(to string, opts *deltasmtp.RcptOptions) error {
        if err := validateLine(to); err != nil {
                return err
        }
        sb := c.cmdBuffer()
        sb.WriteString("RCPT TO:<")
        sb.WriteString(to)
        sb.WriteString(">")
        if _, ok := c.session.ext["DSN"]; ok && opts != nil {
                if err := rcptDSN(sb, opts, c.session.ext); err != nil {
                        return err
                }
        }
        if _, ok := c.session.ext["RRVS"]; ok && opts != nil && !opts.RequireRecipientValidSince.IsZero() {
                sb.WriteString(" RRVS=")
                sb.WriteString(opts.RequireRecipientValidSince.Format(time.RFC3339))
        }
        return c.cmdValidBuffer(pipeliningTypeAnywhere, 25)
}

// RcptResponse returns the result of previous send rcpt command if pipelining is enabled
func (c *Client) RcptResponse() error {
        return c.readResponseValid(25)
}

func rcptDSN(sb *bytes.Buffer, opts *deltasmtp.RcptOptions, ext map[string]string) error {
        if len(opts.Notify) != 0 {
                sb.WriteString(" NOTIFY=")
                if err := parse.CheckNotifySet(opts.Notify); err != nil {
                        return refusedError("smtp: Malformed NOTIFY parameter value")
                }
                for i, v := range opts.Notify {
                        if i != 0 {
                                sb.WriteString(",")
                        }
                        sb.WriteString(string(v))
                }
        }
        if opts.OriginalRecipient != "" {
                var enc string
                switch opts.OriginalRecipientType {
                case deltasmtp.DSNAddressTypeRFC822:
                        if !parse.IsPrintableASCII(opts.OriginalRecipient) {
                                return refusedError("smtp: Illegal address")
                        }
                        enc = encodeXtext(opts.OriginalRecipient)
                case deltasmtp.DSNAddressTypeUTF8:
                        if _, ok := ext["SMTPUTF8"]; ok {
                                enc = encodeUTF8AddrUnitext(opts.OriginalRecipient)
                        } else {
                                enc = encodeUTF8AddrXtext(opts.OriginalRecipient)
                        }
                default:
                        return refusedError("smtp: Unknown address type")
                }
                sb.WriteString(" ORCPT=")
                sb.WriteString(string(opts.OriginalRecipientType))
                sb.WriteString(";")
                sb.WriteString(enc)
        }
        return nil
}

// Content issues a DATA or BDAT (prefer BDAT if available) command to
// the server and returns a writer that
// can be used to write the mail headers and body. The caller should
// close the writer before calling any more methods on c. A call to
// Data must be preceded by one or more calls to Rcpt.
//
// If server returns an error, it will be of type *deltasmtp.
func (c *Client) Content(size int) (*ContentCloser, error) {
        if c.ChunkingActive() {
                if c.PipeliningActive() {
                        return nil, nil
                }
                return c.Bdat(size)
        }
        return c.Data()
}

// ContentResponse returns the result of previous send data command if pipelining is enabled
func (c *Client) ContentResponse(size int) (*ContentCloser, error) {
        if c.ChunkingActive() {
                return c.Bdat(size)
        }
        return c.DataResponse()
}

// Data issues a DATA command to the server and returns a writer that
// can be used to write the mail headers and body. The caller should
// close the writer before calling any more methods on c. A call to
// Data must be preceded by one or more calls to Rcpt.
//
// If server returns an error, it will be of type *deltasmtp.
func (c *Client) Data() (*ContentCloser, error) {
        if err := c.cmdValid(pipeliningTypeLast, 354, "DATA"); err != nil {
                return nil, err
        }
        if c.PipeliningActive() {
                return nil, nil
        }
        c.dotWriter.Reset(c.sender.Writer)
        return c.newContentCloser(&c.dotWriter), nil
}

// DataResponse returns the result of previous send data command if pipelining is enabled
func (c *Client) DataResponse() (*ContentCloser, error) {
        if err := c.readResponseValid(354); err != nil {
                return nil, err
        }
        c.dotWriter.Reset(c.sender.Writer)
        return c.newContentCloser(&c.dotWriter), nil
}

// Bdat issues a BDAT command to the server and returns a writer that
// can be used to write the mail headers and body. The caller should
// close the writer before calling any more methods on c. A call to
// Data must be preceded by one or more calls to Rcpt.
//
// If server returns an error, it will be of type *deltasmtp.
func (c *Client) Bdat(size int) (*ContentCloser, error) {
        if c.PipeliningActive() && c.session.pipelining.pending > 0 {
                return nil, ErrPipeliningNoPendingRequired
        }

        if !c.cfg.chunking {
                return nil, refusedError("smtp: chunking is disabled on the client")
        }
        if _, ok := c.session.ext["CHUNKING"]; !ok {
                return nil, refusedError("smtp: server doesn't support chunking")
        }

        // the size is unknown, buffer to send chunks of chunkingBufferSize
        if size == 0 && c.cfg.chunkingBufferSize > 0 {
                buffer := getChunkingBuffer(c.cfg.chunkingBufferSize)

                c.bdatBufferedWriter.Reset(c.sender.Writer, c.bdatReplyFunc(), size, (*buffer)[:c.cfg.chunkingBufferSize])
                cc := c.newContentCloser(&c.bdatBufferedWriter)
                cc.chunkingBuffer = buffer

                return cc, nil
        }

        c.bdatWriter.Reset(c.sender.Writer, c.bdatReplyFunc(), size)
        return c.newContentCloser(&c.bdatWriter), nil
}

// bdatReplyFunc returns the func the BDAT writers read the reply to a chunk
// with. It is bound once, not for every message.
func (c *Client) bdatReplyFunc() func() error {
        if c.bdatReply == nil {
                c.bdatReply = c.receiveBdatReply
        }
        return c.bdatReply
}

// receiveBdatReply reads the reply to a BDAT chunk.
func (c *Client) receiveBdatReply() error {
        return c.receiveResponseValid(250)
}

// Extension reports whether an extension is support by the server.
// The extension name is case-insensitive. If the extension is supported,
// Extension also returns a string that contains any parameters the
// server specifies for the extension.
func (c *Client) Extension(ext string) (bool, string) {
        ext = strings.ToUpper(ext)
        param, ok := c.session.ext[ext]
        return ok, param
}

// SupportsAuth checks whether an authentication mechanism is supported.
func (c *Client) SupportsAuth(mech string) bool {
        mechs, ok := c.session.ext["AUTH"]
        if !ok {
                return false
        }
        for m := range strings.SplitSeq(mechs, " ") {
                if strings.EqualFold(m, mech) {
                        return true
                }
        }
        return false
}

// MaxMessageSize returns the maximum message size accepted by the server.
// 0 means unlimited.
//
// If the server doesn't convey this information, ok = false is returned.
func (c *Client) MaxMessageSize() (size int, ok bool) {
        v := c.session.ext["SIZE"]
        if v == "" {
                return 0, false
        }
        size, err := strconv.Atoi(v)
        if err != nil || size < 0 {
                return 0, false
        }
        return size, true
}

// Quit sends the QUIT command and closes the connection to the server.
// If Quit fails the connection will still be closed.
// If pipelining is active and the command is send successful,
// the connection is only closed after QuitResponse is called.
func (c *Client) Quit() error {
        if c.session.conn == nil {
                return nil
        }
        if err := c.cmdValid(pipeliningTypeLast, 221, "QUIT"); err != nil {
                _ = c.Close()
                return err
        }

        if c.PipeliningActive() {
                return nil
        }

        return c.Close()
}

        if err := c.readResponseValid(221); err != nil {
                _ = c.Close()
                return err
        }
        return c.Close()
}

// DefaultConfig returns the default configuration of a client.
func DefaultConfig() Config {
        return Config{
                localName: "localhost",
                // As recommended by RFC 5321. For DATA command reply (3xx one) RFC
                // recommends a slightly shorter timeout but we do not bother
                // differentiating these.
                commandTimeout: 5 * time.Minute,
                // 10 minutes + 2 minute buffer in case the server is doing transparent
                // forwarding and also follows recommended timeouts.
                dataTimeout: 12 * time.Minute,
                // 30 seconds, very generous
                tlsHandshakeTimeout: 30 * time.Second,

                // 30 seconds, very generous
                dialTimeout: 30 * time.Second,

                // Doubled maximum line length per RFC 5321 (Section 4.5.3.1.6)
                maxLineLength: 2000,

                // Max lines of a single (multiline) server response.
                // Real world EHLO responses have less than 50 lines, so this
                // is generous while bounding a response to about 1 MB (with the default max
                // line length of 2000).
                maxResponseLines: 512,

                // The client only reads replies, they are small.
                readerSize: 4096,
                // The message is written through this buffer. DATA writes it line by
                // line, with 32 KiB a 13 MB message needs about 400 instead of 3300
                // writes, about 3 times faster. Larger buffers gain little.
                writerSize: 32 << 10,

                // Use chunking (BDAT) if the server supports it.
                chunking: true,

                // Messages of unknown size are buffered and sent in chunks of 2 MiB.
                chunkingBufferSize: 2 << 20,

                // Pipeline commands (RFC 2920) if the server supports it, saves a
                // round trip per command.
                pipelining: true,
        }
}

// Security describes how the connection is etablished.
type Security int32

const (
        // SecurityPreferStartTLS tries to use StartTls but fallbacks to plain.
        //
        // Security note: this is opportunistic TLS and an active attacker can
        // force the fallback to plain. Any authentication performed afterwards
        // sends the credentials unencrypted. Use SecurityStartTLS or SecurityTLS
        // if credentials must never be sent in plain text.
        SecurityPreferStartTLS Security = 0
        // SecurityPlain is always just a plain connection.
        SecurityPlain Security = 1
        // SecurityTLS does a implicit tls connection.
        SecurityTLS Security = 2
        // SecurityStartTLS always does starttls.
        SecurityStartTLS Security = 3
)

// Config contains all configuration needed to configure a smtp client.
type Config struct {
        localName string // the name to use in HELO/EHLO/LHLO

        // Time to wait for tls handshake to succeed.
        tlsHandshakeTimeout time.Duration

        // Time to wait for dial to succeed.
        dialTimeout time.Duration

        // Called with the resolved address of every connection before it is made, see net.Dialer.Control.
        dialControl func(network, address string, c syscall.RawConn) error

        // Time to wait for command responses (this includes 3xx reply to DATA).
        commandTimeout time.Duration

        // Time for a whole DATA or BDAT transfer, including the reply to the end of the message.
        dataTimeout time.Duration

        // Max line length, defaults to 2000
        maxLineLength int

        // Max lines of a single (multiline) response, defaults to 512.
        // A zero or negative value disables the limit.
        maxResponseLines int

        // Reader size
        readerSize int

        // Writer size
        writerSize int

        // Use chunking (BDAT) if the server supports it, defaults to true.
        chunking bool

        // Size of the buffer used for chunking if the message size is unknown,
        // defaults to 2 MiB. Every full buffer is sent as one chunk.
        // A zero or negative value disables buffering.
        chunkingBufferSize int

        // Use pipelining (RFC 2920) if the server supports it, defaults to true.
        pipelining bool
}

        return func(c *Config) {
                c.maxLineLength = maxLineLength
        }

// WithMaxResponseLines sets the maximum number of lines of a single (multiline)

                c.maxResponseLines = maxResponseLines
        }

// WithChunking sets if chunking (BDAT, RFC 3030) is used when the server
// supports it. It is enabled by default. If disabled, DATA is used.
//
// A message of known size is always sent as a single chunk.
func WithChunking(enabled bool) Option {
        return func(c *Config) {
                c.chunking = enabled
        }
}

// WithChunkingBufferSize sets the size of the buffer used for chunking if
// the message size is unknown. Every full buffer is sent as one chunk, so it
// is the chunk size of such messages. Defaults to 2 MiB.
//
// The buffers are shared by all clients (sync.Pool) and only held while a
// message is written, so a new connection does not allocate its own.
//
// A zero or negative value disables buffering, then every write is sent as
// its own chunk. Only do this if your writes are large enough.
func WithChunkingBufferSize(chunkingBufferSize int) Option {
        return func(c *Config) {
                c.chunkingBufferSize = chunkingBufferSize
        }

// its reply before the next one is sent.
func WithPipelining(enabled bool) Option {
        return func(c *Config) {
                c.pipelining = enabled
        }
}

// WithReaderSize sets the size of the read buffer, it holds the replies of
// the server. Defaults to 4 KiB, zero uses 4 KiB.
func WithReaderSize(readerSize int) Option {
        return func(c *Config) {
                c.readerSize = readerSize
        }
}

// WithWriterSize sets the size of the write buffer, commands and the message
// are written through it. Defaults to 32 KiB, zero uses 4 KiB.
//
// It is allocated for every connection. A larger buffer means fewer writes
// for a message sent with DATA.
func WithWriterSize(writerSize int) Option {
        return func(c *Config) {
                c.writerSize = writerSize
        }
}

import (
        "errors"
        "io"
        "sync"

        "github.com/uponusolutions/deltasmtp"
        "github.com/uponusolutions/deltasmtp/internal/deadline"
)

// ContentCloser writes the message content.
// The whole transfer, until the reply to the end of the message, has to finish within the data timeout.
type ContentCloser struct {
        writer io.WriteCloser
        c      *Client
        closed bool
        // chunkingBuffer is the buffer of the writer, returned to chunkingBuffers
        // once the writer is closed. Nil if the writer has none.
        chunkingBuffer *[]byte
}

// chunkingBuffers holds the buffers for chunking messages of unknown size,
// shared by all clients. A buffer is only needed while a message is written,
// so a new connection does not have to allocate (and zero) its own.
var chunkingBuffers sync.Pool

// getChunkingBuffer returns a buffer of at least size bytes from the pool,
// or a new one. A pooled buffer that is too small is left to the GC.
func getChunkingBuffer(size int) *[]byte {
        if b, ok := chunkingBuffers.Get().(*[]byte); ok && cap(*b) >= size {
                return b
        }
        b := make([]byte, size)
        return &b
}

// Writer returns inner io.Writer which possible implement more methods (e.g. ReadFrom)
func (d *ContentCloser) Writer() io.Writer {
        return d.writer
}

// Write writes do underlying writer.
func (d *ContentCloser) Write(p []byte) (n int, err error) {
        if d.c.session.conn == nil {
                return 0, ErrNotConnected
        }
        n, err = d.writer.Write(p)
        // The message is only partly sent, the connection can't be used anymore.
        // A rejected BDAT chunk (*deltasmtp.Status) leaves it in sync.
        if _, ok := errors.AsType[*deltasmtp.Status](err); err != nil && !ok {
                _ = d.c.Close()
        }
        return n, err
}

// CloseWithResponse closes the data closer and returns code, msg.
func (d *ContentCloser) CloseWithResponse() (*deltasmtp.Status, error) {
        if d.closed {
                return nil, errors.New("smtp: data writer closed twice")
        }
        d.closed = true

        // e.g. closed after a malformed reply to a BDAT chunk
        if d.c.session.conn == nil {
                return nil, ErrNotConnected
        }

        err := d.writer.Close()

        // The writer is done with its buffer, even if closing failed.
        if d.chunkingBuffer != nil {
                chunkingBuffers.Put(d.chunkingBuffer)
                d.chunkingBuffer = nil
        }

        if err != nil {
                if _, ok := errors.AsType[*deltasmtp.Status](err); !ok {
                        _ = d.c.Close()
                }
                return nil, err
        }

        status, err := d.c.receiveResponse()

        if err == nil && status.Code != 250 {
                err = status
                status = nil
        }

        return status, err
}

// Close closes the data closer.
func (d *ContentCloser) Close() error {
        _, err := d.CloseWithResponse()
        return err
}

// newContentCloser starts a transfer of the message content,
// the deadline of the data timeout covers the whole transfer.
func (c *Client) newContentCloser(writer io.WriteCloser) *ContentCloser {
        deadline.Set(c.session.conn, c.cfg.dataTimeout)
        c.deadline.Reset()
        return &ContentCloser{c: c, writer: writer}
}

func encodeXtext(raw string) string {
        var out strings.Builder
        out.Grow(len(raw))

        for _, ch := range raw {
                switch {
                case ch >= '!' && ch <= '~' && ch != '+' && ch != '=':
                        // printable non-space US-ASCII except '+' and '='
                        out.WriteRune(ch)
                default:
                        out.WriteRune('+')
                        out.WriteString(strings.ToUpper(strconv.FormatInt(int64(ch), 16)))
                }
        }
        return out.String()
}

// encodeUTF8AddrUnitext encodes raw string to the utf-8-addr-unitext form in RFC 6533.
func encodeUTF8AddrUnitext(raw string) string {
        var out strings.Builder
        out.Grow(len(raw))

        for _, ch := range raw {
                switch {
                case ch >= '!' && ch <= '~' && ch != '+' && ch != '=':
                        // printable non-space US-ASCII except '+' and '='
                        out.WriteRune(ch)
                case ch <= '\x7F':
                        // other ASCII: CTLs, space and specials
                        out.WriteRune('\\')
                        out.WriteRune('x')
                        out.WriteRune('{')
                        out.WriteString(strings.ToUpper(strconv.FormatInt(int64(ch), 16)))
                        out.WriteRune('}')
                default:
                        // UTF-8 non-ASCII
                        out.WriteRune(ch)
                }
        }
        return out.String()
}

// encodeUTF8AddrXtext encodes raw string to the utf-8-addr-xtext form in RFC 6533.
func encodeUTF8AddrXtext(raw string) string {
        var out strings.Builder

                }

}

// limit returns a numeric limit advertised by the LIMITS extension (RFC 9422).
//
// RFC 9422 section 3.7: if the LIMITS parameters are syntactically invalid,
// they are ignored entirely; if only the value of a limit is invalid, only that
// limit is ignored. Numeric limits are 1 to 999999 without leading zeros.
func limit(ext map[string]string, name string) (int, bool) {
        param, ok := ext["LIMITS"]
        if !ok {
                return 0, false
        }

        var value string
        found := false
        // limits-param = limit-name-value 0*[SP limit-name-value]
        for nv := range strings.SplitSeq(param, " ") {
                n, v, hasValue := strings.Cut(nv, "=")
                if !validLimitName(n) || (hasValue && !validLimitValue(v)) {
                        return 0, false
                }
                if !found && strings.EqualFold(n, name) {
                        value, found = v, true
                }
        }

        if !found || len(value) < 1 || len(value) > 6 || value[0] < '1' || value[0] > '9' {
                return 0, false
        }
        n, err := strconv.Atoi(value)
        if err != nil {
                return 0, false
        }
        return n, true
}

// validLimitName checks limit-name = 1*(ALPHA / DIGIT / "-" / "_").
func validLimitName(s string) bool {
        if s == "" {
                return false
        }
        for _, r := range s {
                alpha := r >= 'a' && r <= 'z' || r >= 'A' && r <= 'Z'
                digit := r >= '0' && r <= '9'
                if !alpha && !digit && r != '-' && r != '_' {
                        return false
                }
        }
        return true
}

// validLimitValue checks limit-value = 1*(%x21-3A / %x3C-7E), any VCHAR except ";".
func validLimitValue(s string) bool {
        if s == "" {
                return false
        }
        for i := range len(s) {
                if s[i] < 0x21 || s[i] > 0x7e || s[i] == ';' {
                        return false
                }
        }
        return true
}

// Package deadline sets connection deadlines from timeouts.
// A zero or negative timeout always means no deadline.
package deadline

import (
        "net"
        "time"
)

// Set sets the read and write deadline of conn to now plus timeout.
func Set(conn net.Conn, timeout time.Duration) {
        _ = conn.SetDeadline(at(timeout))
}

// SetWrite sets the write deadline of conn to now plus timeout.
func SetWrite(conn net.Conn, timeout time.Duration) {
        _ = conn.SetWriteDeadline(at(timeout))
}

// Renewer renews the deadline of a connection with fewer calls, setting a
// deadline on a net.Conn is comparatively expensive. A renewal sets the deadline
// to now plus 125% of the timeout and skips renewals for the first 25%, so the
// deadline is always at least the timeout from now. Reset is required when the
// deadline is set otherwise or the timeout changes. The zero value is ready to use.
type Renewer struct {
        next time.Time // renewals before are skipped
}

// Reset makes the next renewal set the deadline.
func (r *Renewer) Reset() { r.next = time.Time{} }

// Set renews the read and write deadline of conn.
func (r *Renewer) Set(conn net.Conn, timeout time.Duration) {
        if timeout <= 0 {
                _ = conn.SetDeadline(time.Time{})
                return
        }
        now := time.Now()
        if now.Before(r.next) {
                return
        }
        r.next = now.Add(timeout / 4)
        _ = conn.SetDeadline(now.Add(timeout + timeout/4))
}

func at(timeout time.Duration) time.Time {
        if timeout <= 0 {
                return time.Time{}
        }
        return time.Now().Add(timeout)
}

// CutPrefixFold is a version of strings.CutPrefix which is case-insensitive.
func CutPrefixFold(s, prefix string) (string, bool) {
        if len(s) < len(prefix) || !strings.EqualFold(s[:len(prefix)], prefix) {
                return "", false
        }
        return s[len(prefix):], true
}

// Cmd parses a line and returns the command, argument or an error.
// Command is converted to upper case.
func Cmd(line string) (cmd string, arg string, err error) {
        line = strings.TrimRight(line, "\r\n")

        l := len(line)
        switch {
        case l == 0:
                return "", "", nil
        case l < 4:
                return "", "", fmt.Errorf("command too short: %q", line)
        case l == 4:
                return strings.ToUpper(line), "", nil
        case l == 5:
                // Too long to be only command, too short to have args
                return "", "", fmt.Errorf("mangled command: %q", line)
        // Commands are always 4 characters long, except STARTTLS which is 8 characters long.
        // STARTTLS has no parameters (RFC 3207).
        case l == 8 && strings.EqualFold(line, "STARTTLS"):
                return "STARTTLS", "", nil
        }

        // If we made it here, command is long enough to have args
        if line[4] != ' ' {
                // There wasn't a space after the command?
                return "", "", fmt.Errorf("mangled command: %q", line)
        }

        return strings.ToUpper(line[0:4]), strings.TrimSpace(line[5:]), nil
}

// Arg is a single ESMTP argument (e.g. SIZE=1024) with an uppercased key.
type Arg struct {
        Key   string
        Value string
}

// Args takes the arguments proceeding a command and parses them
// into a list of key/value pairs after uppercasing each key. Sample arg
// string:
//
//        "BODY=8BITMIME SIZE=1024 SMTPUTF8"
//
// The pairs are appended to args. A caller passing a slice of an array on its
// stack, e.g. buf[:0] of a [8]Arg, parses the usual arguments without allocating.
func Args(args []Arg, s string) ([]Arg, error) {
        if len(s) == 0 {
                return args, nil
        }

        // Only a space is an allowed separator, strings.Fields would accept all Unicode whitespace.
        for arg := range strings.SplitSeq(s, " ") {
                if arg == "" {
                        return nil, fmt.Errorf("empty arg in arg string: %q", s)
                }
                key, value, _ := strings.Cut(arg, "=")
                if strings.IndexByte(value, '=') >= 0 {
                        return nil, fmt.Errorf("failed to parse arg string: %q", arg)
                }
                args = append(args, Arg{Key: strings.ToUpper(key), Value: value})
        }
        return args, nil
}

// HelloArgument parses helo argument
func HelloArgument(arg string) (string, error) {
        domain := arg
        if idx := strings.IndexRune(arg, ' '); idx >= 0 {
                domain = arg[:idx]
        }
        if domain == "" {
                return "", errors.New("invalid domain")
        }
        return domain, nil
}

// Parser parses command arguments defined in RFC 5321 section 4.1.2.
type Parser struct {
        S string
}

func (p *Parser) peekByte() (byte, bool) {
        if len(p.S) == 0 {
                return 0, false
        }
        return p.S[0], true
}

func (p *Parser) readByte() (byte, bool) {
        ch, ok := p.peekByte()
        if ok {
                p.S = p.S[1:]
        }
        return ch, ok
}

func (p *Parser) acceptByte(ch byte) bool {
        got, ok := p.peekByte()
        if !ok || got != ch {
                return false
        }
        p.readByte()
        return true
}

func (p *Parser) expectByte(ch byte) error {
        if !p.acceptByte(ch) {
                if len(p.S) == 0 {
                        return fmt.Errorf("expected '%v', got EOF", string(ch))
                }
                return fmt.Errorf("expected '%v', got '%v'", string(ch), string(p.S[0]))
        }
        return nil
}

// ReversePath parses a Reverse-path (RFC 5321 section 4.1.2):
//
//        Reverse-path = Path / "<>"
func (p *Parser) ReversePath() (string, error) {
        if after, ok := strings.CutPrefix(p.S, "<>"); ok {
                if after != "" && after[0] != ' ' {
                        return "", fmt.Errorf("unexpected %q after null reverse-path", after[0])
                }
                // remove the separator after the reverse-path
                p.S, _ = strings.CutPrefix(after, " ")
                return "", nil
        }
        return p.Path()
}

// Path parses a Path (RFC 5321 section 4.1.2):
//
//        Path  = "<" [ A-d-l ":" ] Mailbox ">"
//        A-d-l = At-domain *( "," At-domain )
//
// The source route (A-d-l) is parsed and discarded as required by RFC 5321
// section 4.1.1.3 and appendix C. For compatibility with broken clients the
// angle brackets may be omitted.
//
// The returned mailbox is in canonical form, see Mailbox.
func (p *Parser) Path() (string, error) {
        hasBracket := p.acceptByte('<')
        if hasBracket && p.acceptByte('@') {
                if err := p.adl(); err != nil {
                        return "", fmt.Errorf("in a-d-l: %w", err)
                }
        }
        mbox, err := p.Mailbox()
        if err != nil {
                return "", fmt.Errorf("in mailbox: %w", err)
        }
        if hasBracket {
                if err := p.expectByte('>'); err != nil {
                        return "", err
                }
        }

        if p.S != "" && p.S[0] != ' ' {
                return "", fmt.Errorf("unexpected %q after path", p.S[0])
        }

        // remove the separator after the path
        p.S, _ = strings.CutPrefix(p.S, " ")

        return mbox, nil
}

// adl parses the remainder of an A-d-l (the leading '@' has been consumed)
// including the terminating ':'.
func (p *Parser) adl() error {
        for {
                if _, err := p.domain(); err != nil {
                        return err
                }
                if p.acceptByte(':') {
                        return nil
                }
                if err := p.expectByte(','); err != nil {
                        return err
                }
                if err := p.expectByte('@'); err != nil {
                        return err
                }
        }
}

// Mailbox parses a Mailbox (RFC 5321 section 4.1.2, extended by RFC 6531
// section 3.3 for SMTPUTF8):
//
//        Mailbox    = Local-part "@" ( Domain / address-literal )
//        Local-part = Dot-string / Quoted-string
//
// The returned mailbox is in canonical form: a quoted local-part whose
// content is a valid Dot-string is returned unquoted (both forms are
// equivalent), any other quoted local-part is returned as a Quoted-string
// with only '"' and '\' escaped. A local-part starting with '"' is therefore
// always quoted, and an unquoted local-part never contains '@', '"', '<', '>'
// or whitespace.
func (p *Parser) Mailbox() (string, error) {
        mbox := p.S
        localPart, err := p.localPart()
        if err != nil {
                return "", fmt.Errorf("in local-part: %w", err)
        }

        if err := p.expectByte('@'); err != nil {
                return "", err
        }

        var domain string
        if ch, _ := p.peekByte(); ch == '[' {
                domain, err = p.addressLiteral()
                if err != nil {
                        return "", fmt.Errorf("in address-literal: %w", err)
                }
        } else {
                domain, err = p.domain()
                if err != nil {
                        return "", fmt.Errorf("in domain: %w", err)
                }
        }

        // Usually the parsed text is in canonical form already, it is returned as
        // it is instead of being put together again.
        parsed := mbox[:len(mbox)-len(p.S)]
        if len(parsed) == len(localPart)+1+len(domain) && parsed[len(localPart)] == '@' &&
                strings.HasPrefix(parsed, localPart) && strings.HasSuffix(parsed, domain) {
                return parsed, nil
        }

        return localPart + "@" + domain, nil
}

// localPart parses a Local-part and returns it in canonical form.
func (p *Parser) localPart() (string, error) {
        if !p.acceptByte('"') {
                s := p.dotString()
                if !isDotString(s) {
                        return "", errors.New("malformed dot-string")
                }
                return s, nil
        }

        // Quoted-string   = DQUOTE *QcontentSMTP DQUOTE
        // QcontentSMTP    = qtextSMTP / quoted-pairSMTP
        // quoted-pairSMTP = %d92 %d32-126
        // qtextSMTP       = %d32-33 / %d35-91 / %d93-126 / UTF8-non-ascii
        var sb strings.Builder
        for {
                ch, ok := p.readByte()
                if !ok {
                        return "", errors.New("unterminated quoted-string")
                }
                switch {
                case ch == '"':
                        content := sb.String()
                        if !utf8.ValidString(content) {
                                return "", errors.New("invalid UTF-8 in quoted-string")
                        }
                        if isDotString(content) {
                                return content, nil
                        }
                        return quoteLocalPart(content), nil
                case ch == '\\':
                        ch, ok = p.readByte()
                        if !ok || ch < 32 || ch > 126 {
                                return "", errors.New("malformed quoted-pair")
                        }
                case ch < 32 || ch == 127:
                        return "", fmt.Errorf("invalid character %q in quoted-string", ch)
                default:
                }
                sb.WriteByte(ch)
        }
}

// dotString consumes the longest run of atext and '.' characters.
func (p *Parser) dotString() string {
        i := 0
        for i < len(p.S) && (p.S[i] == '.' || isAtext(p.S[i])) {
                i++
        }
        s := p.S[:i]
        p.S = p.S[i:]
        return s
}

// isDotString reports whether s is a Dot-string:
//
//        Dot-string = Atom *("."  Atom)
//        Atom       = 1*atext
func isDotString(s string) bool {
        if s == "" || !utf8.ValidString(s) {
                return false
        }
        for atom := range strings.SplitSeq(s, ".") {
                if atom == "" {
                        return false
                }
                for i := range len(atom) {
                        if !isAtext(atom[i]) {
                                return false
                        }
                }
        }
        return true
}

// isAtext reports whether ch is atext (RFC 5322 section 3.2.3), extended by
// UTF8-non-ascii octets (RFC 6531 section 3.3).
func isAtext(ch byte) bool {
        switch {
        case 'a' <= ch && ch <= 'z', 'A' <= ch && ch <= 'Z', '0' <= ch && ch <= '9', ch >= 0x80:
                return true
        }
        return strings.IndexByte("!#$%&'*+-/=?^_`{|}~", ch) >= 0
}

// quoteLocalPart returns content as a Quoted-string.
func quoteLocalPart(content string) string {
        var sb strings.Builder
        sb.Grow(len(content) + 2)
        sb.WriteByte('"')
        for i := range len(content) {
                if content[i] == '"' || content[i] == '\\' {
                        sb.WriteByte('\\')
                }
                sb.WriteByte(content[i])
        }
        sb.WriteByte('"')
        return sb.String()
}

// domain parses a Domain (RFC 5321 section 4.1.2, extended by RFC 6531
// section 3.3 for U-labels):
//
//        Domain     = sub-domain *("." sub-domain)
//        sub-domain = Let-dig [Ldh-str] / U-label
func (p *Parser) domain() (string, error) {
        i := 0
        for i < len(p.S) && (p.S[i] == '.' || p.S[i] == '-' || isLetDig(p.S[i])) {
                i++
        }
        d := p.S[:i]
        if d == "" {
                return "", errors.New("domain is empty")
        }
        if !utf8.ValidString(d) {
                return "", errors.New("invalid UTF-8 in domain")
        }
        for label := range strings.SplitSeq(d, ".") {
                if label == "" || label[0] == '-' || label[len(label)-1] == '-' {
                        return "", fmt.Errorf("malformed domain %q", d)
                }
        }
        p.S = p.S[i:]
        return d, nil
}

// isLetDig reports whether ch is Let-dig, extended by UTF8-non-ascii octets.
func isLetDig(ch byte) bool {
        return 'a' <= ch && ch <= 'z' || 'A' <= ch && ch <= 'Z' || '0' <= ch && ch <= '9' || ch >= 0x80
}

// addressLiteral parses an address-literal (RFC 5321 section 4.1.3):
//
//        address-literal         = "[" ( IPv4-address-literal /
//                                  IPv6-address-literal /
//                                  General-address-literal ) "]"
//        IPv6-address-literal    = "IPv6:" IPv6-addr
//        General-address-literal = Standardized-tag ":" 1*dcontent
//        Standardized-tag        = Ldh-str
//        dcontent                = %d33-90 / %d94-126
func (p *Parser) addressLiteral() (string, error) {
        end := strings.IndexByte(p.S, ']')
        if p.S == "" || p.S[0] != '[' || end < 0 {
                return "", errors.New("malformed address-literal")
        }
        lit := p.S[1:end]
        for i := range len(lit) {
                if lit[i] < 33 || lit[i] > 126 || lit[i] == '[' || lit[i] == '\\' {
                        return "", fmt.Errorf("invalid character %q in address-literal", lit[i])
                }
        }

        tag, value, hasTag := strings.Cut(lit, ":")
        switch {
        case !hasTag:
                if addr, err := netip.ParseAddr(lit); err != nil || !addr.Is4() {
                        return "", fmt.Errorf("malformed IPv4 address-literal %q", lit)
                }
        case strings.EqualFold(tag, "IPv6"):
                addr, err := netip.ParseAddr(value)
                if err != nil || !addr.Is6() || addr.Zone() != "" {
                        return "", fmt.Errorf("malformed IPv6 address-literal %q", lit)
                }
        default:
                if !isLdhStr(tag) || value == "" {
                        return "", fmt.Errorf("malformed general address-literal %q", lit)
                }
        }

        s := p.S[:end+1]
        p.S = p.S[end+1:]
        return s, nil
}

// isLdhStr reports whether s is Let-dig [Ldh-str] (ASCII only).
func isLdhStr(s string) bool {
        if s == "" || s[0] == '-' || s[len(s)-1] == '-' {
                return false
        }
        for i := range len(s) {
                ch := s[i]
                if ch >= 0x80 || ch != '-' && !isLetDig(ch) {
                        return false
                }
        }
        return true
}

// IsASCII reports whether s only contains ASCII characters. Addresses
// containing non-ASCII characters require SMTPUTF8 (RFC 6531).
func IsASCII(s string) bool {
        for i := range len(s) {
                if s[i] >= utf8.RuneSelf {
                        return false
                }
        }
        return true
}

// IsPrintableASCII checks if string contains only printable ascii.
func IsPrintableASCII(val string) bool {
        for _, ch := range val {
                if ch < ' ' || '~' < ch {
                        return false
                }
        }
        return true
}

// CheckNotifySet checks if a DSNNotify array isn't malformed.
func CheckNotifySet(values []deltasmtp.DSNNotify) error {
        if len(values) == 0 {
                return errors.New("malformed NOTIFY parameter value")
        }

        seen := map[deltasmtp.DSNNotify]struct{}{}
        for _, val := range values {
                switch val {
                case deltasmtp.DSNNotifyNever, deltasmtp.DSNNotifyDelayed, deltasmtp.DSNNotifyFailure, deltasmtp.DSNNotifySuccess:
                        if _, ok := seen[val]; ok {
                                return errors.New("malformed NOTIFY parameter value")
                        }
                default:
                        return errors.New("malformed NOTIFY parameter value")
                }
                seen[val] = struct{}{}
        }
        if _, ok := seen[deltasmtp.DSNNotifyNever]; ok && len(seen) > 1 {
                return errors.New("malformed NOTIFY parameter value")
        }

        return nil
}

// Bdat reads a message sent in BDAT chunks (RFC 3030). The zero value is
// unusable, it is set up by NewBdat or Reset.
type Bdat struct {
        size            int64 // bytes left in the current chunk
        last            bool
        bytesReceived   int64
        maxMessageBytes int64
        input           io.Reader
        nextCommand     func() (string, string, error)
}

// BdatArg parses the arguments of a bdat command.
// Return smtp errors if something isn't parseable.
func BdatArg(arg string) (int64, bool, error) {
        // An array instead of strings.Fields avoids an allocation per chunk.
        var args [2]string
        n := 0
        for f := range strings.FieldsSeq(arg) {
                if n == len(args) {
                        return 0, true, deltasmtp.ErrBDATTooManyArgs
                }
                args[n] = f
                n++
        }
        if n == 0 {
                return 0, true, deltasmtp.ErrBDATMissingSize
        }

        last := false
        if n == 2 {
                if !strings.EqualFold(args[1], "LAST") {
                        return 0, true, deltasmtp.ErrBDATUnknownArg
                }
                last = true
        }

        // ParseUint instead of Atoi so we will not accept negative values.
        size, err := strconv.ParseUint(args[0], 10, 32)
        if err != nil || (size == 0 && !last) {
                return 0, true, deltasmtp.ErrBDATMalformedSize
        }

        return int64(size), last, nil
}

// NewBdat creates a new bdat reader.
func NewBdat(size int64, last bool, maxMessageBytes int64, input io.Reader, nextCommand func() (string, string, error)) io.Reader {
        d := &Bdat{}
        d.Reset(size, last, maxMessageBytes, input, nextCommand)
        return d
}

// Reset sets d up to read a new message like a Bdat of NewBdat. It lets a
// connection reuse its reader for every message.
func (d *Bdat) Reset(size int64, last bool, maxMessageBytes int64, input io.Reader, nextCommand func() (string, string, error)) {
        *d = Bdat{
                maxMessageBytes: maxMessageBytes,
                size:            size,
                last:            last,
                input:           input,
                nextCommand:     nextCommand,
        }
}

func (d *Bdat) Read(b []byte) (int, error) {
        if d.size == 0 {
                if d.last {
                        return 0, io.EOF
                }

                cmd, arg, err := d.nextCommand()
                if err != nil {
                        if errors.Is(err, io.EOF) {
                                return 0, deltasmtp.ErrConnection
                        }
                        return 0, err
                }

                switch cmd {
                case "BDAT":
                        d.size, d.last, err = BdatArg(arg)
                        if err != nil {
                                return 0, err
                        }

                        if d.last && d.size == 0 {
                                return 0, io.EOF
                        }
                case "RSET":
                        return 0, deltasmtp.Reset
                case "QUIT":
                        return 0, deltasmtp.Quit
                default:
                        return 0, deltasmtp.ErrBDATExpected
                }
        }

        if d.maxMessageBytes > 0 && d.bytesReceived+d.size > d.maxMessageBytes {
                return 0, deltasmtp.ErrDataTooLarge
        }

        // Never read beyond the chunk, without an io.LimitReader per chunk.
        if int64(len(b)) > d.size {
                b = b[:d.size]
        }

        n, err := d.input.Read(b)
        d.bytesReceived += int64(n)
        d.size -= int64(n)

        // the reader should never throw EOF
        if errors.Is(err, io.EOF) {
                err = deltasmtp.ErrConnection
        }

        return n, err
}

// Dot reads a dot-encoded message (RFC 5321 section 4.5.2) up to the end of
// data line. The zero value is unusable, it is set up by NewDot or Reset.
type Dot struct {
        r       *bufio.Reader
        state   int
        limited bool
        n       int64 // Maximum bytes remaining.
}

// NewDot creates a new dot reader.
func NewDot(reader *bufio.Reader, maxMessageBytes int64) io.Reader {
        dr := &Dot{}
        dr.Reset(reader, maxMessageBytes)
        return dr
}

// Reset sets r up to read a new message like a Dot of NewDot. It lets a
// connection reuse its reader for every message.
func (r *Dot) Reset(reader *bufio.Reader, maxMessageBytes int64) {
        *r = Dot{r: reader}

        if maxMessageBytes > 0 {
                r.limited = true
                r.n = maxMessageBytes
        }
}

const (
        stateBegin = iota // Initial state, beginning of first line.
        stateLine         // Somewhere inside a line.
        stateCR           // Wrote \r.
        stateEOF          // Reached .\r\n end marker line.
)

// Read reads in some more bytes.
// Run data through a simple state machine to
// elide leading dots and detect End-of-Data
// (<CR><LF>.<CR><LF>) line.
//
//nolint:revive // cognitive-complexity: a single state machine loop keeps the hot path fast.
func (r *Dot) Read(b []byte) (int, error) {
        if r.state == stateEOF {
                return 0, io.EOF
        }

        if r.limited {
                if r.n <= 0 {
                        return 0, deltasmtp.ErrDataTooLarge
                }

                if int64(len(b)) > r.n {
                        b = b[0:r.n]
                }
        }

        var n int       // Data written to b.
        var skipped int // How many.

        c, err := r.peek(len(b))

        // To reach this state it is necessary that 5 bytes are still in the buffer ready to be consumed.
        // So there shouldn't be a case where err happens and there aren't 5 bytes in c at this point.
        switch r.state {
        case stateCR:
                // write \n
                b[0] = '\n'
                n++
                skipped += 2
                if len(c) >= 5 && c[3] == '\r' && c[4] == '\n' {
                        r.state = stateEOF
                        skipped += 2 // skip .\n\r
                        return r.finalize(n, skipped, err)
                }
                r.state = stateLine
                b = b[1:]
                c = c[3:]
        case stateBegin:
                // nothing to do
                if len(c) == 0 {
                        return r.finalize(n, skipped, err)
                }

                if c[0] == '.' {
                        // need to check .\r\n
                        if len(c) < 3 {
                                return r.finalize(n, skipped, err)
                        }

                        // already finish, empty data
                        if c[1] == '\r' && c[2] == '\n' {
                                skipped += 3
                                r.state = stateEOF
                                return r.finalize(n, skipped, err)
                        }

                        skipped++
                        c = c[1:]
                }
                r.state = stateLine
        default:
        }

        for {
                i := bytes.Index(c, deltasmtp.Crlfdot)

                // No full \r\n. found.
                if i == -1 {
                        l := len(c)

                        if l > 1 && c[l-2] == '\r' && c[l-1] == '\n' && err == nil {
                                // Ends with \r\n, write everything before.
                                n += copy(b, c[:l-2])
                        } else if l > 0 && c[l-1] == '\r' && err == nil {
                                // Ends with \r, write everything before.
                                n += copy(b, c[:l-1])
                        } else {
                                n += copy(b, c)
                        }

                        break
                }

                // i is \r, \n.\r\n needs to be accessible
                if len(c)-1 < i+4 {
                        if err != nil {
                                n += copy(b, c[:i+2])
                                if len(b) >= len(c[:i+2]) {
                                        skipped = len(c) - (i + 2)
                                } else if len(b) == len(c[:i+2])-1 {
                                        r.state = stateCR // Next time we want to write '\n'.
                                        skipped--         // Prevent \r from being discarded
                                }
                        } else if i > 0 {
                                // Not enough bytes to check for \r\n.\r\n,
                                // write everything before
                                n += copy(b, c[:i])
                        }

                        break
                }

                p := copy(b, c[:i+2])
                n += p

                // b was to small
                if p < i+2 {
                        // we only wrote \r
                        if i+2-p == 1 {
                                r.state = stateCR // Next time we want to write '\n'.
                                skipped--         // Prevent \r from being discarded
                        }
                        break
                }

                // The end \r\n.\n\r
                if c[i+3] == '\r' && c[i+4] == '\n' {
                        r.state = stateEOF
                        skipped += 3 // skip .\r\n
                        break
                }

                skipped++ // . isn't written
                b = b[i+2:]
                c = c[i+3:]
        }

        return r.finalize(n, skipped, err)
}

func (r *Dot) finalize(n int, skipped int, err error) (int, error) {
        // n + skipped is always smaller then what was peeked,
        // so it is guaranteed to work
        _, _ = r.r.Discard(n + skipped)

        if r.limited {
                r.n -= int64(n)
        }

        // as long as something was written, we do not propagate the err if any
        if n > 0 {
                return n, nil
        }

        if errors.Is(err, io.EOF) && r.state != stateEOF {
                err = io.ErrUnexpectedEOF
        } else if err == nil && r.state == stateEOF {
                err = io.EOF
        }

        return n, err
}

func (r *Dot) peek(blen int) ([]byte, error) {
        minimumPeek := 5
        if r.state == stateBegin {
                minimumPeek = 3
        }

        // IMPORTANT: We cannot wait on read,
        // because no EOL returns. So we call peek with minimumPeek to fill the buffer probably with more
        // to get as much data as possible in the second peek.
        if r.r.Buffered() < minimumPeek {
                _, _ = r.r.Peek(minimumPeek)
        }
        // min 5, max buffer size, default len(b)
        return r.r.Peek(max(min(blen, r.r.Buffered()), minimumPeek))
}

import (

// ErrTooManyResponseLines is returned if a response exceeds the maximum response lines.
var ErrTooManyResponseLines = textproto.ProtocolError("response has too many lines")

// Receiver is used as a wrapper around a connection to read from it
type Receiver struct {
        *bufio.Reader
        maxLineLength      int
        maxResponseLines   int
        lineLengthExceeded bool
}

// NewReceiver creates a new connection wrapper.
func NewReceiver(
        conn io.Reader,
        readerSize int,
        maxLineLength int,
        maxResponseLines int,
) *Receiver {
        // No cmp.Or, it pushes NewReceiver over the inlining budget.
        if readerSize == 0 {
                readerSize = 4096 // default
        }

        return &Receiver{
                Reader:             bufio.NewReaderSize(conn, readerSize),
                maxLineLength:      maxLineLength,
                maxResponseLines:   maxResponseLines,
                lineLengthExceeded: false,
        }
}

// Reset discards all state and switches to reading from conn, e.g. after a reconnect.
// It also clears an exceeded line length, which otherwise blocks all further reads.
func (t *Receiver) Reset(conn io.Reader) {
        t.Reader.Reset(conn)
        t.lineLengthExceeded = false
}

// ReadFullLine reads a single line from r,
// eliding the final \n or \r\n from the returned string.
func (t *Receiver) ReadFullLine() (string, error) {
        line, err := t.ReadFullLineBytes()
        return string(line), err
}

// ReadResponse reads a multi-line response of the form:
//
//        code-message line 1
//        code-message line 2
//        ...
//        code message line n
//
// where code is a three-digit status code. The first line starts with the
// code and a hyphen. The response is terminated by a line that starts
// with the same code followed by a space. Each line in message is
// separated by a newline (\n).
//
// See page 36 of RFC 959 (https://www.ietf.org/rfc/rfc959.txt) for
// details of another form of response accepted:
//
//        code-message line 1
//        message line 2
//        ...
//        code message line n
func (t *Receiver) ReadResponse() (*deltasmtp.Status, error) {
        status, continued, err := t.readFirstCodeLine()
        if err != nil {
                return nil, err
        }

        if err = t.readResponseExtra(status, continued, true); err != nil {
                return nil, err
        }

        return status, nil
}

// ReadResponseValid returns an error if the code does not match expectation.
// An expected single line reply, the common case, is checked without allocating.
func (t *Receiver) ReadResponseValid(expectCode int) error {
        line, err := t.ReadFullLineBytes()
        if err != nil {
                return err
        }
        if code, ok := parseCode(line); ok && line[3] == ' ' && !IsCodeUnexpected(code, expectCode) {
                return nil
        }

        status, continued, err := parseFirstCodeLine(string(line))
        if err != nil {
                return err
        }

        unexpected := IsCodeUnexpected(status.Code, expectCode)

        err = t.readResponseExtra(status, continued, unexpected)
        if err != nil {
                return err
        }

        if unexpected {
                return status
        }

        return nil
}

func (t *Receiver) readResponseExtra(status *deltasmtp.Status, continued bool, appendMessage bool) error {
        if !continued {
                return nil
        }

        var message string
        var err error

        if appendMessage {
                // A reply usually arrives at once, the lines already buffered tell how
                // many to expect instead of growing the slice line by line.
                buffered, _ := t.Peek(t.Buffered())
                n := bytes.Count(buffered, []byte("\n"))
                if t.maxResponseLines > 0 {
                        n = min(n, t.maxResponseLines)
                }
                status.Lines = slices.Grow(status.Lines, n)
        }

        codeString := strconv.Itoa(status.Code)
        encCodePart := status.EnhancedCode.ToPart()
        for lines := 1; continued; lines++ {
                if t.maxResponseLines > 0 && lines >= t.maxResponseLines {
                        return ErrTooManyResponseLines
                }
                continued, message, err = t.readExtraCodeLine(codeString, encCodePart)
                if err != nil {
                        return err
                }
                if appendMessage {
                        status.Lines = append(status.Lines, message)
                }
        }
        return nil
}

func (t *Receiver) readFirstCodeLine() (*deltasmtp.Status, bool, error) {
        line, err := t.ReadFullLine()
        if err != nil {
                return nil, false, err
        }
        return parseFirstCodeLine(line)
}

func parseFirstCodeLine(line string) (*deltasmtp.Status, bool, error) {
        if len(line) < 4 || line[3] != ' ' && line[3] != '-' {
                return nil, false, textproto.ProtocolError(fmt.Sprintf("short response: %q", line))
        }
        continued := line[3] == '-'
        code, ok := parseCode(line)
        if !ok {
                return nil, false, textproto.ProtocolError(fmt.Sprintf("invalid response code: %q", line))
        }
        message := line[4:]

        // all 2xx, 4xx, and 5xx response lines
        if code >= 300 && code < 400 {
                return deltasmtp.NewStatusS(code, deltasmtp.NoEnhancedCode, message), continued, nil
        }

        enhPart, rest, ok := strings.Cut(message, " ")
        if !ok {
                return deltasmtp.NewStatusS(code, deltasmtp.NoEnhancedCode, message), continued, nil
        }
        enhCode, ok := parseEnhancedCode(enhPart, code)
        if !ok {
                return deltasmtp.NewStatusS(code, deltasmtp.NoEnhancedCode, message), continued, nil
        }

        return deltasmtp.NewStatusS(code, enhCode, rest), continued, nil
}

func (t *Receiver) readExtraCodeLine(codeString string, enhCodePart []byte) (continued bool, message string, err error) {
        line, err := t.ReadFullLine()
        if err != nil {
                return false, "", err
        }
        return parseExtraCodeLine(line, codeString, enhCodePart)
}

// parseExtraCodeLine does strict verification like described in RFC 5321
func parseExtraCodeLine(line string, codeString string, enhCodePart []byte) (bool, string, error) {
        if len(line) < 4+len(enhCodePart) ||
                (line[3] != ' ' && line[3] != '-') ||
                line[0:3] != codeString ||
                line[4:(4+len(enhCodePart))] != string(enhCodePart) {
                return false, "", textproto.ProtocolError(fmt.Sprintf("invalid response: %q", line))
        }
        return line[3] == '-', line[4+len(enhCodePart):], nil
}

// parseEnhancedCode parses an enhanced status code (RFC 3463, RFC 2034):
//
//        status-code = class "." subject "." detail
//        class       = "2" / "4" / "5"   ; must match the first digit of the reply code
//        subject     = 1*3digit          ; without leading zeros
//        detail      = 1*3digit          ; without leading zeros
//
// Anything else isn't an enhanced code, but the start of the reply text, ok is
// false then. That is common (e.g. the greeting), so no error is built for it.
func parseEnhancedCode(s string, replyCode int) (code deltasmtp.EnhancedCode, ok bool) {
        if strings.Count(s, ".") != 2 {
                return deltasmtp.NoEnhancedCode, false
        }

        for i := range code {
                var part string
                part, s, _ = strings.Cut(s, ".")
                maxDigits := 3
                if i == 0 {
                        maxDigits = 1
                }
                if len(part) == 0 || len(part) > maxDigits || len(part) > 1 && part[0] == '0' ||
                        strings.IndexFunc(part, func(r rune) bool { return r < '0' || r > '9' }) != -1 {
                        return deltasmtp.NoEnhancedCode, false
                }
                code[i], _ = strconv.Atoi(part)
        }

        if code[0] != replyCode/100 || code[0] != 2 && code[0] != 4 && code[0] != 5 {
                return deltasmtp.NoEnhancedCode, false
        }
        return code, true
}

// parseCode returns the reply code at the start of line, three digits from 100
// to 999. ok is false if line doesn't start with one or is shorter than four bytes.
func parseCode[T string | []byte](line T) (code int, ok bool) {
        if len(line) < 4 {
                return 0, false
        }
        for i := range 3 {
                c := line[i]
                if c < '0' || c > '9' || i == 0 && c == '0' {
                        return 0, false
                }
                code = code*10 + int(c-'0')
        }
        return code, true
}

// IsCodeUnexpected validates if the code is unexpected.
// If the prefix of the status does not match the digits in expectCode,
// ReadResponse returns with err set to &Error{code, message}.
// For example, if expectCode is 31, an error will be returned if
// the status is not in the range [310,319].
func IsCodeUnexpected(code int, expectCode int) bool {
        return 1 <= expectCode && expectCode < 10 && code/100 != expectCode ||
                10 <= expectCode && expectCode < 100 && code/10 != expectCode ||
                100 <= expectCode && expectCode < 1000 && code != expectCode
}

// ReadFullLineBytes is ReadFullLine without the conversion to a string. The
// returned bytes are only valid until the next read.
func (t *Receiver) ReadFullLineBytes() ([]byte, error) {
        // If the line limit was exceeded once, the connection shouldn't be used anymore.
        if t.lineLengthExceeded {
                return nil, deltasmtp.ErrTooLongLine
        }

        var line []byte
        for {
                l, more, err := t.ReadLine()
                if err != nil {
                        return nil, err
                }

                if t.maxLineLength > 0 && len(l)+len(line) > t.maxLineLength {
                        t.lineLengthExceeded = true
                        return nil, deltasmtp.ErrTooLongLine
                }

                // Avoid the copy if the first call produced a full line.
                if line == nil && !more {
                        return l, nil
                }
                line = append(line, l...)
                if !more {
                        break
                }
        }
        return line, nil
}

package smtpwriter

import (
        "bufio"
        "errors"
        "io"
        "strconv"
)

var (
        ending   = []byte("BDAT 0 LAST\r\n")
        prefix   = []byte("BDAT ")
        lastcrlf = []byte(" LAST\r\n")
        crlf     = []byte("\r\n")
)

// NewBdat returns a writer that can be used to write bdat commands to w.
// If size is known (> 0) the message is sent as a single chunk, otherwise
// every write is sent as its own chunk.
// The caller should close the BdatWriter before the next call to a method on w.
func NewBdat(writer *bufio.Writer, read func() error, size int) io.WriteCloser {
        d := &Bdat{}
        d.Reset(writer, read, size)
        return d
}

// Reset sets d up to write a new message like a Bdat of NewBdat. It lets a
// connection reuse its writer for every message.
func (d *Bdat) Reset(writer *bufio.Writer, read func() error, size int) {
        *d = Bdat{
                w:             writer,
                read:          read,
                remainingSize: size,
                knownSize:     size > 0,
        }
}

// Bdat writes a message as BDAT chunks, see NewBdat.
type Bdat struct {
        w    *bufio.Writer
        read func() error
        // size still expected to be written
        remainingSize int
        // if true then size is known, so remainingSize is respected
        knownSize bool
        // if larger then 0 bdat command is send and some bytes are left to write
        // only used if knownSize is true
        remainingChunkSize int
        // if Write was called at least once this flag is true.
        started bool
}

// Write writes bytes as bdat chunk, or as part of the single chunk if the size is known.
func (d *Bdat) Write(b []byte) (n int, err error) {
        d.started = true

        var p int

        // something left to write
        if d.remainingChunkSize > 0 {
                p, err = d.writeBdat(b[:min(len(b), d.remainingChunkSize)], d.remainingChunkSize == d.remainingSize)
                n += p
                b = b[p:]
                if err != nil {
                        return n, err
                }

                if len(b) == 0 {
                        return n, nil
                }
        }

        if d.knownSize && d.remainingSize < len(b) {
                return n, errors.New("got more bytes than expected, check length")
        }

        p, err = d.writeBdat(b, d.knownSize)
        n += p

        return n, err
}

// write writes b until everything is written
func (d *Bdat) write(b []byte) (n int, err error) {
        var p int
        for n < len(b) {
                p, err = d.w.Write(b[n:])
                n += p
                if err != nil {
                        return n, err
                }
        }

        if d.remainingChunkSize > 0 {
                d.remainingChunkSize -= n
        }
        if d.knownSize {
                d.remainingSize -= n
        }
        return n, nil
}

// writeBdat writes b as bdat command and checks return
// if size is known a single chunk with the remaining size is started
func (d *Bdat) writeBdat(b []byte, last bool) (n int, err error) {
        if d.remainingChunkSize == 0 {
                size := len(b)
                if d.knownSize {
                        size = d.remainingSize
                        d.remainingChunkSize = size
                }
                if err = d.bdat(size, last); err != nil {
                        return n, err
                }
        } else if len(b) > d.remainingChunkSize {
                // just finish the current chunk
                b = b[:d.remainingChunkSize]
        }

        n, err = d.write(b)
        if err != nil {
                return n, err
        }

        // Are there still bytes missing to finish bdat chunk?
        if d.remainingChunkSize > 0 {
                return n, nil
        }

        if err = d.w.Flush(); err != nil {
                return n, err
        }

        // last read is done outside of bdat reader
        if last {
                return n, nil
        }

        return n, d.read()
}

// write BDAT <SIZE> \r\n
func (d *Bdat) bdat(size int, last bool) (err error) {
        if _, err = d.w.Write(prefix); err != nil {
                return err
        }

        if _, err = d.w.Write(strconv.AppendInt(d.w.AvailableBuffer(), int64(size), 10)); err != nil {
                return err
        }

        if last {
                if _, err = d.w.Write(lastcrlf); err != nil {
                        return err
                }
                return nil
        }

        if _, err = d.w.Write(crlf); err != nil {
                return err
        }

        return nil
}

// Close finishes the message: a short message is padded with zero bytes to the
// announced size, and BDAT 0 LAST is sent if no chunk was marked as last.
func (d *Bdat) Close() error {
        // The close came to early, more bytes expected
        if d.knownSize && d.remainingSize > 0 {
                if _, err := d.Write(make([]byte, d.remainingSize)); err != nil {
                        return err
                }
        }

        // if size is not known a bdat 0 last is necessary
        if !d.knownSize || !d.started {
                if _, err := d.w.Write(ending); err != nil {
                        return err
                }
        }

        return d.w.Flush()
}

package smtpwriter

import (
        "bufio"
        "io"

// NewBdatWriterBuffered returns a writer that can be used to write bdat commands to w.
// The caller should close the BdatWriter before the next call to a method on w.
// Without a buffer (len(buffer) == 0) the unbuffered bdat writer is returned,
// buffering into an empty buffer would never make progress.
func NewBdatWriterBuffered(writer *bufio.Writer, read func() error, size int, buffer []byte) io.WriteCloser {
        if len(buffer) == 0 {
                return NewBdat(writer, read, size)
        }
        d := &BdatBuffered{}
        d.Reset(writer, read, size, buffer)
        return d
}

// BdatBuffered writes a message as BDAT chunks of the size of its buffer, see
// NewBdatWriterBuffered.
type BdatBuffered struct {
        buffer   []byte
        position int
        writer   Bdat
}

// Reset sets d up to write a new message like NewBdatWriterBuffered, the buffer
// must not be empty. It lets a connection reuse its writer for every message.
func (d *BdatBuffered) Reset(writer *bufio.Writer, read func() error, size int, buffer []byte) {
        *d = BdatBuffered{buffer: buffer}
        d.writer.Reset(writer, read, size)
}

// ReadFrom implements io.ReaderFrom, it reads r into the buffer and sends every
// full buffer as bdat chunk. Like Write, a full buffer is only sent once more
// data follows, so Close can send the last one as LAST.
func (d *BdatBuffered) ReadFrom(r io.Reader) (n int64, err error) {
        for {
                if err = d.flushFull(); err != nil {
                        return n, err
                }
                p, errRead := r.Read(d.buffer[d.position:])
                n += int64(p)
                d.position += p
                switch {
                case errRead == io.EOF: // not returned, see io.Copy
                        return n, nil
                case errRead != nil:
                        return n, errRead
                }
        }
}

// Write buffers b, every full buffer is sent as bdat chunk.
// Writes larger than the buffer are passed through if the buffer is empty.
func (d *BdatBuffered) Write(b []byte) (n int, err error) {
        var p int

        for n < len(b) {
                p, err = d.write(b[n:])
                n += p
                if err != nil {
                        return n, err
                }
        }

        return n, err
}

func (d *BdatBuffered) write(b []byte) (int, error) {
        if err := d.flushFull(); err != nil {
                return 0, err
        }
        // The buffer would only be filled and sent, send b directly.
        if d.position == 0 && len(b) > len(d.buffer) {
                return d.writer.Write(b)
        }
        n := copy(d.buffer[d.position:], b)
        d.position += n
        return n, nil
}

// flushFull sends the buffer as bdat chunk if it is full.
func (d *BdatBuffered) flushFull() error {
        if d.position < len(d.buffer) {
                return nil
        }
        d.position = 0
        _, err := d.writer.Write(d.buffer)
        return err
}

// Close sends the buffered rest of the message as the last chunk.
func (d *BdatBuffered) Close() error {
        if d.position > 0 {
                // force bdat last, if size is unknown
                if !d.writer.knownSize {
                        d.writer.knownSize = true
                        d.writer.remainingSize = d.position
                }
                if _, err := d.writer.Write(d.buffer[:d.position]); err != nil {
                        return err
                }
                d.position = 0
        }

        return d.writer.Close()
}

// NewDot returns a writer that can be used to write a dot-encoding to w.
// It takes care of inserting leading dots when necessary,
// translating line-ending \n into \r\n, and adding the final .\r\n line
// when the DotWriter is closed. The caller should close the
// DotWriter before the next call to a method on w.
//
// See the documentation for Reader's DotReader method for details about dot-encoding.
func NewDot(writer *bufio.Writer) io.WriteCloser {
        d := &Dot{}
        d.Reset(writer)
        return d
}

// Reset sets d up to write a new message like a Dot of NewDot. It lets a
// connection reuse its writer for every message.
func (d *Dot) Reset(writer *bufio.Writer) {
        *d = Dot{W: writer}
}

// Dot writes a dot-encoded message, see NewDot.
type Dot struct {
        W     *bufio.Writer
        state int
}

const (
        wstateBegin     = iota // starting state
        wstateBeginLine        // beginning of line
        wstateCR               // wrote \r (possibly at end of line)
        wstateData             // writing data in middle of line
)

func (d *Dot) Write(b []byte) (n int, err error) {
        var (
                i    int
                p    []byte
                pLen int
                bw   = d.W
        )
        for len(b) > 0 {
                i = bytes.IndexByte(b, '\n')
                if i >= 0 {
                        p, b = b[:i+1], b[i+1:]
                } else {
                        p, b = b, nil
                }
                pLen = len(p)
                // A leading dot must be stuffed at the beginning of every line,
                // including the very first one (wstateBegin).
                if (d.state == wstateBeginLine || d.state == wstateBegin) && p[0] == '.' {
                        err = bw.WriteByte('.')
                        if err != nil {
                                return n, err
                        }
                }

                if b == nil {
                        // no end of line found in p
                        if p[pLen-1] == '\r' {
                                // p ends with \r
                                d.state = wstateCR
                        } else {
                                // just write it down
                                d.state = wstateData
                        }

                        if _, err = bw.Write(p); err != nil {
                                return n, err
                        }
                } else if d.state == wstateCR && pLen == 1 {
                        // if b isn't nil and pLen is 1, then it must be a \n
                        // as \r was send before, just write crnl
                        d.state = wstateBeginLine
                        if err = bw.WriteByte('\n'); err != nil {
                                return n, err
                        }
                } else {
                        // line is ending
                        d.state = wstateBeginLine
                        if pLen >= 2 && p[pLen-2] == '\r' {
                                // fastpath if line ending is correct \r\n
                                if _, err = bw.Write(p); err != nil {
                                        return n, err
                                }
                        } else {
                                // data + crnl
                                if _, err = bw.Write(p[:pLen-1]); err != nil {
                                        return n, err
                                }
                                if _, err = bw.Write(deltasmtp.Crnl); err != nil {
                                        return n, err
                                }
                        }
                }

                n += pLen
        }
        return n, err
}

// Close ends the message with the end of data line and flushes it.
func (d *Dot) Close() error {
        bw := d.W
        // End an unfinished line before the end of data line.
        if d.state != wstateBeginLine {
                // normally \r gets ignored if no \n follows, but at closing we just take it as a line break
                // same behavior as original textproto
                if d.state != wstateCR {
                        if err := bw.WriteByte('\r'); err != nil {
                                return err
                        }
                }
                if err := bw.WriteByte('\n'); err != nil {
                        return err
                }
        }
        if _, err := bw.Write(deltasmtp.Dotcrnl); err != nil {
                return err
        }
        return bw.Flush()
}

                        // 4.5.3.1.8), used if it doesn't advertise RCPTMAX (RFC 9422).
                        maxRecipients: 100,
                        // Exim's default for smtp_accept_max_per_connection, used if the
                        // server doesn't advertise MAILMAX (RFC 9422).
                        maxTransactions: 1000,
                },
                client: client.DefaultConfig(),
        }

        cfg := DefaultConfig()

                o(&cfg)

// Security describes how the connection is established.
type Security int32

const (
        // SecurityPreferStartTLS tries to use StartTLS but fallbacks to plain if
        // StartTLS is not available, rejected by the server or the TLS handshake
        // failed. A handshake failure includes the client rejecting the server
        // certificate (e.g. expired or not matching the host name): the mail is
        // then sent over a plain connection instead of returning an error.
        //
        // Security note: this is opportunistic TLS and provides no protection
        // against an active attacker. An attacker on the network path can strip
        // STARTTLS from the EHLO response or break the TLS handshake to force the
        // fallback to a plain connection.
        //
        // If a SASL client is configured, authentication is still performed after
        // such a fallback, i.e. credentials are sent over the unencrypted
        // connection (for mechanisms like PLAIN or LOGIN the password is only
        // base64 encoded). This is intended: the mode prefers delivering over
        // failing. If credentials must never be sent in plain text, use
        // SecurityStartTLS or SecurityTLS instead.
        SecurityPreferStartTLS Security = 0
        // SecurityPlain is always just a plain connection.
        SecurityPlain Security = 1
        // SecurityTLS does a implicit tls connection.
        SecurityTLS Security = 2
        // SecurityStartTLS always does starttls and fails if the server doesn't
        // support it or the TLS handshake fails. This is the default.
        SecurityStartTLS Security = 3
)

// content is a message opened for a transaction: data or str, or r if it isn't nil.
type content struct {
        r    io.Reader
        data []byte
        str  string
}

// open opens msg for a transaction.
func open[M Message](msg M) (content, error) {
        switch m := any(msg).(type) {
        case string:
                return content{str: m}, nil
        case []byte:
                return content{data: m}, nil
        case func() io.Reader:
                return content{r: m()}, nil
        case func() (io.ReadCloser, error):
                r, err := m()
                if err != nil {
                        return content{}, fmt.Errorf("mailer: open message: %w", err)
                }
                return content{r: r}, nil
        default:
                panic("unreachable")
        }
}

// size returns the size of the message, 0 if unknown.
func (m content) size() int {
        if m.r == nil {
                return len(m.data) + len(m.str)
        }
        if l, ok := m.r.(Len); ok {
                return l.Len()
        }
        return 0
}

// close closes the reader if it is an io.Closer.
func (m content) close() {
        if closer, ok := m.r.(io.Closer); ok {
                _ = closer.Close()
        }
}

// writeTo writes the message to w.
func (m content) writeTo(w io.Writer) error {
        if m.data != nil {
                _, err := w.Write(m.data)
                return err
        }

        buf := getCopyBuffer()
        defer copyBuffers.Put(buf)
        if m.r != nil {
                _, err := io.CopyBuffer(w, m.r, *buf)
                return err
        }
        // []byte(m.str), also by io.WriteString, would copy the whole message.
        for s := m.str; len(s) > 0; {
                n := copy(*buf, s)
                if _, err := w.Write((*buf)[:n]); err != nil {
                        return err
                }
                s = s[n:]
        }
        return nil
}

// copyBuffers avoids that io.Copy allocates 32 KB for every message.
var copyBuffers sync.Pool

func getCopyBuffer() *[]byte {
        if b, ok := copyBuffers.Get().(*[]byte); ok {
                return b
        }
        b := make([]byte, 32*1024)
        return &b
}

// Mailer implements a smtp client sending mail to a server.

        verifying bool
        // transactions counts the MAIL commands sent on the connection (RFC 9422 MAILMAX).
        transactions int

// New returns a new smtp client.
// You need to set at least the server address to get a working mailer.
//

        return NewFromConfig(NewConfig(opts...))
}

        return &Mailer{
                cfg:    cfg.extra,
                client: client.NewFromConfig(cfg.client),
        }
}

// Client returns the inner smtp client. Use with caution.
func (c *Mailer) Client() *client.Client {
        return c.client
}

// Connect connects to one of the server addresses, secured as configured (see
// WithSecurity) and authenticated if a SASL client is set and the server supports AUTH.
// On an error the connection is closed. It fails with ErrVerifying while a Verifier of
// c is active, a new connection would lose its transaction.
func (c *Mailer) Connect(ctx context.Context) error {
        if c.verifying {
                return ErrVerifying
        }
        return c.connect(ctx)
}

// errNoServerAddress is returned by connect without a server address.
var errNoServerAddress = errors.New("smtp: no server address")

func (c *Mailer) connect(ctx context.Context) error {
        err := errNoServerAddress
        for _, addrs := range c.cfg.serverAddresses {
                for p := range addrs {
                        i := (p + c.cfg.serverAddressIndex) % len(addrs)
                        if i < 0 { // a negative index counts from the end
                                i += len(addrs)
                        }
                        if err = c.connectAddress(ctx, addrs[i]); err == nil {
                                return nil
                        }
                }
        }
        return err
}

func (c *Mailer) connectAddress(ctx context.Context, addr string) (err error) {
        c.transactions = 0
        sec := c.cfg.security
        if sec == SecurityTLS {
                err = c.client.DialTLS(ctx, c.cfg.tlsConfig, addr)
        } else {
                err = c.client.Dial(ctx, addr)
        }
        if err != nil {
                return err // closed by Dial
        }

        // A connection without the configured security and auth mustn't be reused.
        defer func() {
                if err != nil {
                        _ = c.client.Close()
                }
        }()

        if sec == SecurityStartTLS || sec == SecurityPreferStartTLS {
                if ok, _ := c.client.Extension("STARTTLS"); ok {
                        serverName, _, _ := net.SplitHostPort(addr)
                        if err = c.client.StartTLS(c.cfg.tlsConfig, serverName); err != nil {
                                if sec != SecurityPreferStartTLS {
                                        return err
                                }
                                if err = c.client.Dial(ctx, addr); err != nil { // fall back to plain
                                        return err
                                }
                        }
                } else if sec == SecurityStartTLS {
                        _ = c.Disconnect()
                        return errors.New("smtp: server doesn't support STARTTLS")
                }
        }

        if ok, _ := c.client.Extension("AUTH"); ok && c.cfg.saslClient != nil {
                if err = c.client.Auth(c.cfg.saslClient); err != nil {
                        _ = c.Disconnect()
                        return err
                }
        }
        return nil
}

// reset sends RSET.
func (c *Mailer) reset() error {
        return c.cmd(c.client.Reset(), c.client.ResetResponse)
}

// cmd returns the error of sending a command, or with pipelining its response.
func (c *Mailer) cmd(err error, response func() error) error {
        if err != nil || !c.client.PipeliningActive() {
                return err
        }
        return response()
}

// Disconnect ends current connection gracefully, if any exists.
func (c *Mailer) Disconnect() error {
        return c.cmd(c.client.Quit(), c.client.QuitResponse)
}

// Terminate ends current connection forcefully.
func (c *Mailer) Terminate() error {
        return c.client.Close()
}

// Connected reports whether c has a connection.
func (c *Mailer) Connected() bool {
        return c.client.Connected()
}

// ServerAddress returns the current server address.
func (c *Mailer) ServerAddress() string {
        return c.client.ServerAddress()
}

// ServerName returns the current server name.
func (c *Mailer) ServerName() string {
        return c.client.ServerName()
}

// mail sends MAIL, counted against the transaction limit, see mailError for the errors.
// wait reads a pipelined response as well, otherwise it has to be passed to mailError.
func (c *Mailer) mail(from string, opts *client.MailOptions, wait bool) error {
        c.transactions++
        err := c.client.Mail(from, opts)
        if wait {
                err = c.cmd(err, c.client.MailResponse)
        }
        return c.mailError(err)
}

// errRenew asks to send a MAIL again on a new connection, see mailError.
var errRenew = errors.New("smtp: transaction limit of the connection reached")

// mailError returns err of MAIL, or errRenew on a 421 (the client closed the connection)
// after earlier transactions: a server may limit them without MAILMAX (e.g. Exim). The
// limit isn't learned, the 421 may have another reason.
func (c *Mailer) mailError(err error) error {
        // Not errors.As, it would allocate st for every MAIL.
        if st, ok := errors.AsType[*deltasmtp.Status](err); ok && st.Code == 421 && c.transactions > 1 {
                return errRenew
        }
        return err
}

// ready connects if needed, or again once the connection has used up its transactions.
func (c *Mailer) ready(ctx context.Context) error {
        if c.client.Connected() && !c.exhausted() {
                return nil
        }
        _ = c.Disconnect() // no-op without a connection, closes it also on failure
        return c.connect(ctx)
}

// exhausted reports whether the connection reached RFC 9422 MAILMAX or WithMaxTransactions.
func (c *Mailer) exhausted() bool {
        n, ok := c.client.MaxTransactions()
        if !ok {
                n = c.cfg.maxTransactions
        }
        return n > 0 && c.transactions >= n
}

// maxRecipients returns RFC 9422 RCPTMAX or WithMaxRecipients, 0 is unlimited.
func (c *Mailer) maxRecipients() int {
        if n, ok := c.client.MaxRecipients(); ok {
                return n
        }
        return c.cfg.maxRecipients
}

// supportsUTF8 reports whether the server supports SMTPUTF8.
func (c *Mailer) supportsUTF8() bool {
        ok, _ := c.client.Extension("SMTPUTF8")
        return ok
}

// ReaderWriteToLen extends io.Reader with Len and WriteTo.
type ReaderWriteToLen interface {
        io.Reader
        io.WriterTo
        Len() int
}

// Modified from https://github.com/golang/go/blob/master/src/io/multi_test.go
type multiReader struct {
        readers []ReaderWriteToLen
}

type eofReader struct{}

func (eofReader) Read([]byte) (int, error) {
        return 0, io.EOF
}

func (eofReader) Len() int {
        return 0
}

func (eofReader) WriteTo(_ io.Writer) (sum int64, err error) {
        return 0, nil
}

func (mr *multiReader) Read(p []byte) (n int, err error) {
        for len(mr.readers) > 0 {
                // Optimization to flatten nested multiReaders (Issue 13558).
                if len(mr.readers) == 1 {
                        if r, ok := mr.readers[0].(*multiReader); ok {
                                mr.readers = r.readers
                                continue
                        }
                }
                n, err = mr.readers[0].Read(p)
                if errors.Is(err, io.EOF) {
                        // Use eofReader instead of nil to avoid nil panic
                        // after performing flatten (Issue 18232).
                        mr.readers[0] = eofReader{} // permit earlier GC
                        mr.readers = mr.readers[1:]
                }
                if n > 0 || !errors.Is(err, io.EOF) {
                        if errors.Is(err, io.EOF) && len(mr.readers) > 0 {
                                // Don't return EOF yet. More readers remain.
                                err = nil
                        }
                        return n, err
                }
        }
        return 0, io.EOF
}

func (mr *multiReader) WriteTo(w io.Writer) (sum int64, err error) {
        n := int64(0)
        for _, r := range mr.readers {
                rn, err := r.WriteTo(w)
                n += rn
                if err != nil {
                        return n, err
                }
        }
        return n, nil
}

func (mr *multiReader) Len() int {
        length := 0
        for _, r := range mr.readers {
                length += r.Len()
        }
        return length
}

// MultiReader is a utility struct to ease MultiReader creation.
type MultiReader struct {
        readers []ReaderWriteToLen
}

// AddReader adds the next readers to the factory.
// The new [MultiReader] takes ownership of the Readers,
func (mrf *MultiReader) AddReader(readers ...ReaderWriteToLen) {
        mrf.readers = append(mrf.readers, readers...)
}

// AddBytes adds the multiple byte slices to the factory.
// The new [MultiReader] takes ownership of the bytes,
func (mrf *MultiReader) AddBytes(arrbytes ...[]byte) {
        for _, reader := range arrbytes {
                mrf.readers = append(mrf.readers, bytes.NewBuffer(reader))
        }
}

// Reader creates a MultiReader from all added readers.
func (mrf *MultiReader) Reader() ReaderWriteToLen {
        if len(mrf.readers) == 0 {
                return eofReader{}
        }

        var reader ReaderWriteToLen

        if len(mrf.readers) == 1 {
                reader = mrf.readers[0]
        } else {
                reader = NewMultiReader(mrf.readers...)
        }

        // reset state
        mrf.readers = nil

        return reader
}

// NewMultiReader returns a Reader that's the logical concatenation of
// the provided input readers. They're read sequentially. Once all
// inputs have returned EOF, Read will return EOF.  If any of the readers
// return a non-nil, non-EOF error, Read will return that error.
// The new [Reader] takes ownership of the Readers,
func NewMultiReader(readers ...ReaderWriteToLen) ReaderWriteToLen {
        return &multiReader{readers}
}

// Message is an RFC 822-style mail: headers, a blank line and the body, with CRLF

// It is read once per transaction (see Mailer.SendAdvanced):
//
//   - string or []byte: isn't modified or retained. The size is announced with
//     SIZE and BDAT sends it as a single chunk, if the server supports them.
//   - func() io.Reader or func() (io.ReadCloser, error): called once per
//     transaction, it has to return the whole message every time, e.g. by
//     opening a file. The reader is closed after the transaction if it is an
//     io.Closer. An error fails the remaining recipients. A reader with Len is
//     announced with SIZE.
//
// Only these exact types are accepted, convert named types at the call site.
type Message interface {
        string | []byte | func() io.Reader | func() (io.ReadCloser, error)
}

// Send sends msg from from to rcpts, see SendAdvanced.
func (c *Mailer) Send[M Message](ctx context.Context, from string, rcpts []string, msg M) (Report, error) {
        return c.SendAdvanced(ctx, from, client.MailOptions{}, rcpts, nil, msg)
}

// SendAdvanced sends msg from from to rcpts, rcptsOptions[i] applies to rcpts[i].
// It fails with ErrVerifying while a Verifier of c is active.
//
// It uses the connection of c or opens one, also once the connection reached RFC 9422
// MAILMAX (otherwise WithMaxTransactions) or the server rejects MAIL with 421 after
// earlier transactions. mailOptions.Size defaults to the size of msg, SMTPUTF8 is used
// if the server supports it.
//
// A transaction has at most RFC 9422 RCPTMAX (otherwise WithMaxRecipients) accepted
// recipients, further transactions take the remaining ones. A recipient rejected for
// too many recipients (452, RFC 5321 section 4.5.3.1.10) ends the transaction, it and
// the remaining ones go to the next, with at most as many as were accepted.
//
// The Report has a response per transaction and the rejected recipients. On an error
// the remaining recipients fail with it and no further transaction is started.
func (c *Mailer) SendAdvanced[M Message](
        ctx context.Context,
        from string,
        mailOptions client.MailOptions,
        rcpts []string,
        rcptsOptions []*deltasmtp.RcptOptions,
        msg M,
) (res Report, err error) {
        if len(rcpts) < 1 {
                return res, errors.New("no recipients")
        }
        if c.verifying {
                return res.fail(rcpts, nil, ErrVerifying)
        }

        for len(rcpts) > 0 {
                if err := c.ready(ctx); err != nil {
                        return res.fail(rcpts, nil, err)
                }

                body, err := open(msg)
                if err != nil {
                        return res.fail(rcpts, nil, err)
                }
                t := transaction{c: c, rcpts: rcpts}
                status, err := t.send(from, mailOptions, rcptsOptions, body)
                res.Failures = append(res.Failures, t.failures...)

                switch {
                case errors.Is(err, errRenew):
                        continue
                case err != nil:
                        return res.fail(rcpts, t.failures, err)
                case status == nil: // every recipient was rejected
                default:
                        res.Responses = append(res.Responses, Response{Status: status, Rcpts: t.accepted})
                }
                rcpts, rcptsOptions = t.remaining(rcpts, rcptsOptions)
        }

        return res, nil
}

// remaining returns the deferred recipients and the ones of all that weren't sent,
// with their options.
func (t *transaction) remaining(all []string, opts []*deltasmtp.RcptOptions) ([]string, []*deltasmtp.RcptOptions) {
        n := len(t.rcpts)
        if len(t.deferred) == 0 {
                return all[n:], opts[min(n, len(opts)):]
        }
        rcpts := make([]string, 0, len(t.deferred)+len(all)-n)
        rcptsOpts := make([]*deltasmtp.RcptOptions, 0, cap(rcpts))
        for _, i := range t.deferred {
                rcpts, rcptsOpts = append(rcpts, all[i]), append(rcptsOpts, rcptOption(opts, i))
        }
        // opts may be shorter than all, the missing options are nil either way.
        return append(rcpts, all[n:]...), append(rcptsOpts, opts[min(n, len(opts)):]...)
}

// fail adds rcpts, without the ones in failures, as failure with err.
func (r Report) fail(rcpts []string, failures []resolve.Failure, err error) (Report, error) {
        if rcpts = acceptedRcpts(rcpts, failures); len(rcpts) > 0 {
                r.Failures = append(r.Failures, resolve.Failure{Rcpts: rcpts, Error: err})
        }
        return r, err
}

// Send sends the mail msg from from to rcpts.
//
// The recipients are grouped by their MX servers, or sent to the servers of
// WithServerAddresses, and every server gets its own connection, see
// Mailer.SendAdvanced for the transactions on it and Message for msg.
//
// Recipients that couldn't be resolved, were rejected or whose server failed are
// in Report.Failures. err is only set if the MX lookup failed temporarily.
//
// The default SecurityStartTLS also applies to MX delivery, pass
// WithSecurity(SecurityPreferStartTLS) for opportunistic TLS as usual between
// mail servers.
func Send[M Message](ctx context.Context, from string, rcpts []string, msg M, opts ...Option) (res Report, err error) {
        unresolved, err := perServer(ctx, rcpts, opts, func(m *Mailer, rcpts []string) {
                r, _ := m.Send(ctx, from, rcpts, msg) // the error is in r.Failures
                res.Responses = append(res.Responses, r.Responses...)
                res.Failures = append(res.Failures, r.Failures...)
        })
        res.Failures = append(unresolved, res.Failures...)
        return res, err
}

// perServer calls f with a new Mailer for every server of rcpts, the configured server
// addresses or the MX servers, and disconnects it. It returns the unresolved recipients.
func perServer(ctx context.Context, rcpts []string, opts []Option, f func(m *Mailer, rcpts []string)) ([]resolve.Failure, error) {
        config := NewConfig(opts...)
        mx := resolve.Result{Servers: []resolve.Server{{Rcpts: rcpts, Addresses: config.extra.serverAddresses}}}
        if len(config.extra.serverAddresses) == 0 {
                var err error
                r := resolve.New(nil)
                if mx, err = r.Recipients(ctx, rcpts); err != nil {
                        return nil, err
                }
        }

        for _, server := range mx.Servers {
                config.extra.serverAddresses = server.Addresses
                m := NewFromConfig(config)
                f(m, server.Rcpts)
                _ = m.Disconnect() // ends an open transaction as well
        }
        return mx.Failures, nil
}

// acceptedRcpts returns rcpts without the ones in failures, rcpts isn't modified.
func acceptedRcpts(rcpts []string, failures []resolve.Failure) []string {
        if len(failures) == 0 {
                return rcpts
        }

        failed := make(map[string]struct{}, len(failures))
        for _, fail := range failures {
                for _, rcpt := range fail.Rcpts {
                        failed[rcpt] = struct{}{}
                }
        }

        accepted := make([]string, 0, max(len(rcpts)-len(failed), 0))
        for _, rcpt := range rcpts {
                if _, ok := failed[rcpt]; !ok {
                        accepted = append(accepted, rcpt)
                }
        }
        return accepted
}

import (
        "cmp"
        "errors"

        "github.com/uponusolutions/deltasmtp"
        "github.com/uponusolutions/deltasmtp/client"
        "github.com/uponusolutions/deltasmtp/resolve"
)

// transaction sends a mail to the first of rcpts: MAIL, RCPT and DATA, pipelined if possible.
type transaction struct {
        c        *Mailer
        rcpts    []string // cut to the recipients sent, once prepared
        accepted []string
        failures []resolve.Failure
        deferred []int // indexes of rcpts rejected for too many recipients, sent in the next transaction

        // Pending responses with pipelining: MAIL and the RCPTs from rcpts[next] on.
        mail bool
        next int
}

// send sends msg and closes it. A nil status without an error means every
// recipient was rejected, the connection may be closed then (e.g. Postfix's error limit),
// the next transaction connects again. On an error the transaction is ended or the
// connection closed, errRenew asks to send it on a new connection.
func (t *transaction) send(from string, opts client.MailOptions, rcptsOptions []*deltasmtp.RcptOptions,
        msg content,
) (*deltasmtp.Status, error) {
        c := t.c
        defer msg.close()
        size := msg.size()
        opts.Size = cmp.Or(opts.Size, int64(size))
        opts.SMTPUTF8 = opts.SMTPUTF8 || c.supportsUTF8()

        // From here on every failure has to end the transaction.
        w, err := t.prepare(from, &opts, rcptsOptions, size)
        switch {
        case err != nil:
                return nil, c.abort(err) // keeps errRenew, the connection is closed already
        case w == nil:
                _ = c.resetOrClose() // every recipient has its result
                return nil, nil
        }

        if err = msg.writeTo(w.Writer()); err != nil {
                // Partly sent, the connection can't be reused. A rejected BDAT chunk would leave it
                // in sync, but can't be told apart from a *deltasmtp.Status returned by a reader.
                return nil, errors.Join(err, c.client.Close())
        }

        status, err := w.CloseWithResponse()
        if err != nil {
                return nil, c.abort(err)
        }
        return status, nil
}

// prepare sends MAIL, RCPT and DATA for a message of size, with at most maxRecipients
// accepted, and returns the writer of the content, nil if every recipient was rejected.
// A recipient rejected for too many recipients (e.g. Postfix without RCPTMAX) stops
// further RCPTs, is deferred and the accepted ones are learned as limit.
// The transaction may be left open, the caller has to end it.
func (t *transaction) prepare(from string, mailOptions *client.MailOptions, rcptsOptions []*deltasmtp.RcptOptions,
        size int,
) (*client.ContentCloser, error) {
        c := t.c
        pipelining := c.client.PipeliningActive()
        limit := c.maxRecipients()
        // full reports whether the recipients before i, without the rejected ones, reach limit.
        full := func(i int) bool { return limit > 0 && i-len(t.failures)-len(t.deferred) >= limit }

        if err := c.mail(from, mailOptions, false); err != nil {
                return nil, err
        }
        t.mail = true

        for i, addr := range t.rcpts {
                // RCPTMAX counts accepted recipients, rejected pending ones make room for more.
                if full(i) && pipelining {
                        if err := t.flush(i); err != nil {
                                return nil, err
                        }
                }
                if len(t.deferred) > 0 || full(i) {
                        t.rcpts = t.rcpts[:i]
                        break
                }

                opts := rcptOption(rcptsOptions, i)
                err := c.client.Rcpt(addr, opts)
                if errors.Is(err, client.ErrPipeliningCongestion) {
                        if err = t.flush(i); err != nil {
                                return nil, err
                        }
                        err = c.client.Rcpt(addr, opts)
                }
                if err == nil && pipelining {
                        continue
                }
                // The pending responses are matched by position, addr wasn't sent.
                if errors.Is(err, client.ErrRefused) && pipelining {
                        if errFlush := t.flush(i); errFlush != nil {
                                return nil, errFlush
                        }
                        t.next = i + 1
                }
                if err = t.rcptResult(i, err); err != nil {
                        return nil, err
                }
        }

        if !pipelining && len(t.accepted) == 0 {
                return nil, nil
        }

        w, err := c.client.Content(size)
        if errors.Is(err, client.ErrPipeliningCongestion) {
                if err = t.flush(len(t.rcpts)); err != nil {
                        return nil, err // DATA wasn't sent
                }
                w, err = c.client.Content(size)
        }
        if err == nil && w == nil { // pipelined, an error clears the DATA response
                if err = t.flush(len(t.rcpts)); err == nil {
                        w, err = t.contentResponse(size)
                }
        }
        if err == nil && len(t.deferred) > 0 {
                c.cfg.maxRecipients = len(t.accepted)
        }
        return w, err
}

func rcptOption(rcptsOptions []*deltasmtp.RcptOptions, i int) *deltasmtp.RcptOptions {
        if i < len(rcptsOptions) {
                return rcptsOptions[i]
        }
        return nil
}

// flush reads the pending responses of MAIL and the RCPTs before rcpts[end], on an error
// all pending ones.
func (t *transaction) flush(end int) error {
        c := t.c
        if t.mail {
                t.mail = false
                if err := c.client.MailResponse(); err != nil {
                        return c.mailError(c.clearResponses(err))
                }
        }
        for ; t.next < end; t.next++ {
                if err := t.rcptResult(t.next, c.client.RcptResponse()); err != nil {
                        return c.clearResponses(err)
                }
        }
        return nil
}

// rcptResult records the response err to the RCPT of rcpts[i], errors other than a
// rejection are returned.
func (t *transaction) rcptResult(i int, err error) error {
        switch {
        case err == nil && len(t.accepted) == i: // none rejected yet, no need to copy
                t.accepted = t.rcpts[: i+1 : i+1]
        case err == nil:
                t.accepted = append(t.accepted, t.rcpts[i])
        case !isRejection(err):
                return err
        case len(t.accepted) > 0 && tooManyRcpts(err): // the next transaction makes progress
                t.deferred = append(t.deferred, i)
        default:
                t.failures = append(t.failures, resolve.Failure{Rcpts: []string{t.rcpts[i]}, Error: err})
        }
        return nil
}

// contentResponse reads the response to the pipelined DATA (or sends BDAT). Without an
// accepted recipient its failure is expected, a 421 closes the connection.
func (t *transaction) contentResponse(size int) (*client.ContentCloser, error) {
        c := t.c
        if len(t.accepted) > 0 {
                return c.client.ContentResponse(size)
        }
        if c.client.ChunkingActive() {
                return nil, nil // BDAT isn't sent yet
        }
        if w, err := c.client.ContentResponse(size); err == nil {
                _ = w.Close() // RFC 2920: DATA may be accepted even if every RCPT was rejected
        }
        return nil, nil
}

// tooManyRcpts reports whether err rejects a RCPT for too many recipients: 452 (RFC 5321
// section 4.5.3.1.10) with X.5.3 or no enhanced code, or 552 with X.5.3, treated as 452.
func tooManyRcpts(err error) bool {
        st, ok := errors.AsType[*deltasmtp.Status](err)
        if !ok {
                return false
        }
        x53 := st.EnhancedCode[1] == 5 && st.EnhancedCode[2] == 3
        return st.Code == 452 && (x53 || st.EnhancedCode == deltasmtp.NoEnhancedCode) || st.Code == 552 && x53
}

// clearResponses reads the pending responses after err, if still connected.
func (c *Mailer) clearResponses(err error) error {
        if c.client.Connected() {
                if errClear := c.client.ClearResponses(0); errClear != nil {
                        return errors.Join(err, errClear)
                }
        }
        return err
}

// abort ends a failed transaction and returns err. After a rejection the connection is
// in sync and kept if RSET succeeds, otherwise it is closed.
func (c *Mailer) abort(err error) error {
        if !isRejection(err) {
                return errors.Join(err, c.client.Close())
        }
        if errReset := c.resetOrClose(); errReset != nil {
                return errors.Join(err, errReset)
        }
        return err
}

// resetOrClose ends the transaction by RSET, or closes the connection if that fails.
func (c *Mailer) resetOrClose() error {
        if err := c.reset(); err != nil {
                return errors.Join(err, c.client.Close())
        }
        return nil
}

// isRejection reports whether err rejects a command and leaves the connection in sync:
// a server reply other than 421 or a refusal of the client.
func isRejection(err error) bool {
        if st, ok := errors.AsType[*deltasmtp.Status](err); ok {
                return st.Code != 421
        }
        return errors.Is(err, client.ErrRefused)
}

package mailer

import (
        "context"
        "errors"

        "github.com/uponusolutions/deltasmtp"
        "github.com/uponusolutions/deltasmtp/client"
        "github.com/uponusolutions/deltasmtp/resolve"
)

// VerifyMode selects the command a Verifier checks recipients with.
type VerifyMode int

const (
        // VerifyVrfy checks recipients with VRFY (RFC 5321 section 3.5.1).
        // 252 (can't verify, but accepts mail) counts as accepted, many servers answer every VRFY with it.
        VerifyVrfy VerifyMode = iota
        // VerifyRcpt checks recipients with RCPT TO in a mail transaction that is ended by RSET.
        VerifyRcpt
)

var (
        // ErrVerifying is returned by Mailer.Connect, Mailer.Send, Mailer.SendAdvanced and Mailer.Verifier
        // while a Verifier of the Mailer is active.
        ErrVerifying = errors.New("smtp: a verifier is active")
        // ErrVerifierEnded is returned by Verifier.Verify after the verifier ended.
        ErrVerifierEnded = errors.New("smtp: verifier ended")
)

// Verifier verifies recipients one by one on the connection of its Mailer, which can't
// send while the Verifier is active, from Mailer.Verifier to End.
//
// With VerifyRcpt the transaction stays open, every recipient costs a single RCPT. A full
// transaction (see SendAdvanced) is replaced by RSET and MAIL, on a new connection once
// it used up its transactions.
//
// A rejection keeps the Verifier active, any other error closes the connection and ends it.
type Verifier struct {
        mailer   *Mailer
        mode     VerifyMode
        from     string
        begun    bool
        ended    bool
        accepted int // recipients accepted in the transaction
}

// Verifier returns an active Verifier checking recipients of a mail from from with mode.
// It returns ErrVerifying if another Verifier of c is active.
func (c *Mailer) Verifier(mode VerifyMode, from string) (*Verifier, error) {
        if c.verifying {
                return nil, ErrVerifying
        }
        c.verifying = true
        return &Verifier{mailer: c, mode: mode, from: from}, nil
}

// Active reports whether the Verifier hasn't ended.
func (v *Verifier) Active() bool {
        return !v.ended
}

// Verify returns nil if the server accepts addr as recipient. A rejection is a
// *deltasmtp.Status, a refusal of the client matches client.ErrRefused. A failing
// connect or MAIL, or with VerifyVrfy a server not supporting VRFY (500, 502, 504), ends
// the Verifier. With VerifyRcpt these replies reject the address, RCPT is mandatory.
// SMTPUTF8 is used if the server supports it.
func (v *Verifier) Verify(ctx context.Context, addr string) error {
        if v.ended {
                return ErrVerifierEnded
        }
        err := v.verify(ctx, addr)
        // e.g. Postfix without RCPTMAX: learn the limit, check addr in a new transaction
        if v.accepted > 0 && tooManyRcpts(err) {
                v.mailer.cfg.maxRecipients = v.accepted
                err = v.verify(ctx, addr)
        }
        return err
}

// verify checks addr, in a new transaction if the current one is full.
func (v *Verifier) verify(ctx context.Context, addr string) error {
        c := v.mailer
        if limit := c.maxRecipients(); !v.begun || limit > 0 && v.accepted >= limit {
                if err := v.begin(ctx); err != nil {
                        v.dispose()
                        return err
                }
        }

        var err error
        if v.mode == VerifyRcpt {
                if err = c.cmd(c.client.Rcpt(addr, nil), c.client.RcptResponse); err == nil {
                        v.accepted++
                }
        } else {
                err = c.cmd(c.client.Verify(addr, &client.VrfyOptions{SMTPUTF8: c.supportsUTF8()}), c.client.VerifyResponse)
        }
        if err != nil && (!isRejection(err) || v.mode == VerifyVrfy && unsupported(err)) {
                v.dispose()
        }
        return c.closeUnlessRejected(err)
}

// unsupported reports whether err of VRFY rejects the command itself, not the address: 500,
// 502 or 504 (RFC 5321 section 4.2.3), e.g. Postfix's "502 5.5.1 VRFY command is disabled".
// Not used for RCPT: it is mandatory, there these replies reject the address (e.g. its
// syntax), and ending the Verifier would leave the transaction open.
func unsupported(err error) bool {
        st, ok := errors.AsType[*deltasmtp.Status](err)
        return ok && (st.Code == 500 || st.Code == 502 || st.Code == 504)
}

// begin connects if needed and with VerifyRcpt ends the previous transaction and sends
// MAIL. A rejected MAIL keeps the connection, any other error closes it.
func (v *Verifier) begin(ctx context.Context) error {
        c := v.mailer
        // A connection renewed by ready needs no RSET.
        if v.begun && v.mode == VerifyRcpt && c.client.Connected() && !c.exhausted() {
                if err := c.resetOrClose(); err != nil {
                        return err
                }
        }
        if err := c.ready(ctx); err != nil {
                return err
        }

        if v.mode == VerifyRcpt {
                v.accepted = 0
                switch err := c.mail(v.from, &client.MailOptions{SMTPUTF8: c.supportsUTF8()}, true); {
                case errors.Is(err, errRenew):
                        return v.begin(ctx)
                case err != nil:
                        return c.closeUnlessRejected(err)
                }
        }

        v.begun = true
        return nil
}

// End ends the Verifier, with VerifyRcpt by RSET, and releases the Mailer.
// If RSET fails, the connection is closed.
func (v *Verifier) End() error {
        if v.ended {
                return nil
        }
        v.dispose()
        if !v.begun || v.mode != VerifyRcpt {
                return nil
        }
        return v.mailer.resetOrClose()
}

// dispose ends the Verifier without RSET and releases the Mailer.
func (v *Verifier) dispose() {
        v.ended = true
        v.mailer.verifying = false
}

// closeUnlessRejected closes the connection unless err is nil or a rejection.
func (c *Mailer) closeUnlessRejected(err error) error {
        if err == nil || isRejection(err) {
                return err
        }
        return errors.Join(err, c.client.Close())
}

// VerifyReport contains the result of verifying recipients.
type VerifyReport struct {
        // Accepted are the recipients accepted by their server.
        Accepted []string
        // Failures are the recipients that couldn't be resolved, were rejected or couldn't be checked.
        Failures []resolve.Failure
}

// Verify checks with mode whether the servers of rcpts, see Send, accept them as recipients
// of a mail from from.
func Verify(ctx context.Context, mode VerifyMode, from string, rcpts []string, opts ...Option) (res VerifyReport, err error) {
        unresolved, err := perServer(ctx, rcpts, opts, func(m *Mailer, rcpts []string) {
                res.add(ctx, m, mode, from, rcpts)
        })
        res.Failures = append(unresolved, res.Failures...)
        return res, err
}

// add adds the results of checking rcpts with a Verifier of the new Mailer m.
func (r *VerifyReport) add(ctx context.Context, m *Mailer, mode VerifyMode, from string, rcpts []string) {
        v, _ := m.Verifier(mode, from) // a new Mailer has no active Verifier
        for i, rcpt := range rcpts {
                switch err := v.Verify(ctx, rcpt); {
                case err == nil:
                        r.Accepted = append(r.Accepted, rcpt)
                case v.ended: // the remaining recipients can't be checked
                        r.Failures = append(r.Failures, resolve.Failure{Rcpts: rcpts[i:], Error: err})
                        return
                default:
                        r.Failures = append(r.Failures, resolve.Failure{Rcpts: []string{rcpt}, Error: err})
                }
        }
}

// Package resolve implements a resolver to get prioritized server addresses for recipients.
// Typically by retrieving mx records by dns.
package resolve

import (
        "cmp"
        "context"
        "errors"
        "fmt"
        "net"
        "slices"
        "strings"
)

// LookupMX describes the functions needed for a struct to be used as a resolver.
type LookupMX interface {
        LookupMX(ctx context.Context, name string) ([]*net.MX, error)
}

// Resolver is
type Resolver struct {
        resolver LookupMX
}

// New creates a new resolver. If nil is given the net.DefaultResolver is used.
func New(resolver LookupMX) Resolver {
        return Resolver{
                resolver: cmp.Or[LookupMX](resolver, net.DefaultResolver),
        }
}

// Server contains prioritized server addresses for specific recipients.
type Server struct {
        Addresses [][]string
        Rcpts     []string
}

// Failure contains recipients where no server could be resolved and the corresponding error.
type Failure struct {
        Error error
        Rcpts []string
}

// Result contains Servers and Fails of an recipients server resolve.
type Result struct {
        Servers  []Server
        Failures []Failure
}

func (r *Result) Error() error {
        fails := make([]error, len(r.Failures))
        for i, fail := range r.Failures {
                fails[i] = fmt.Errorf("error for %s: %w", fail.Rcpts, fail.Error)
        }
        return errors.Join(fails...)
}

func (r *Result) addError(rcpt string, err error) int {
        r.Failures = append(r.Failures, Failure{
                Rcpts: []string{rcpt},
                Error: err,
        })
        return len(r.Failures) - 1
}

func (r *Result) addErrorRcpt(index int, rcpt string) int {
        r.Failures[index].Rcpts = append(r.Failures[index].Rcpts, rcpt)
        return len(r.Failures) - 1
}

func (r *Result) addServer(rcpt string, addresses [][]string) int {
        for i, server := range r.Servers {
                if sliceEqual(addresses, server.Addresses) {
                        r.Servers[i].Rcpts = append(r.Servers[i].Rcpts, rcpt)
                        return i
                }
        }

        r.Servers = append(r.Servers, Server{Rcpts: []string{rcpt}, Addresses: addresses})

        return len(r.Servers) - 1
}

func (r *Result) addServerRcpt(i int, rcpt string) {
        r.Servers[i].Rcpts = append(r.Servers[i].Rcpts, rcpt)
}

type cache struct {
        index int
        err   error
}

// Recipients resolves the smtp servers for specific recipients and groups them.
// If an error returns, then there was an error resolving the MX-Record for the domains of the recipients.
// For example if the network is offline or the dns server isn't responding.
// Failures inside the result are permanent errors like no mx record could be found or
// the domain could not be extracted.
func (r *Resolver) Recipients(ctx context.Context, rcpts []string) (Result, error) {
        res := Result{}
        rcptToDomain := map[string]string{}
        domainToServer := map[string]cache{}

        for _, rcpt := range rcpts {
                _, domain, found := strings.CutLast(rcpt, "@")
                if !found {
                        res.addError(rcpt, fmt.Errorf("couldn't extract domain part of %s", rcpt))
                        continue
                }

                rcptToDomain[rcpt] = domain
                c, ok := domainToServer[domain]
                if ok {
                        if c.err != nil {
                                res.addErrorRcpt(c.index, rcpt)
                        } else {
                                res.addServerRcpt(c.index, rcpt)
                        }
                        continue
                }

                addresses, err := r.Lookup(ctx, domain)
                if err != nil {
                        return res, err
                }

                if len(addresses) == 0 {
                        err = fmt.Errorf("domain resolve failed, no mx record found for %s", domain)
                        domainToServer[domain] = cache{
                                index: res.addError(rcpt, err),
                                err:   err,
                        }
                        continue
                }
                domainToServer[domain] = cache{
                        index: res.addServer(rcpt, addresses),
                }
        }

        return res, nil
}

// Lookup returns prioritized server addresses for a specific domains.
// It returns nil,nil if no mx record is found and an error if the dns request didn't worked.
func (r *Resolver) Lookup(ctx context.Context, domain string) ([][]string, error) {
        // LookupMX returns the DNS MX records for the given domain name sorted by preference.
        // => We can assume it is sorted and just need
        mxs, err := r.resolver.LookupMX(ctx, domain)
        if err != nil {
                netErr := &net.DNSError{}
                if errors.As(err, &netErr) && netErr.IsNotFound {
                        return nil, nil
                }
                return nil, err
        }

        res := [][]string{}

        var prio uint16
        for i := range mxs {
                host := net.JoinHostPort(strings.TrimSuffix(mxs[i].Host, "."), "25")
                if mxs[i].Pref > prio || i == 0 {
                        prio = mxs[i].Pref
                        res = append(res, []string{
                                host,
                        })
                        continue
                }
                res[len(res)-1] = append(
                        res[len(res)-1],
                        host,
                )
        }

        return res, nil
}

func sliceEqual(a [][]string, b [][]string) bool {
        if len(a) != len(b) {
                return false
        }

        for i := range a {
                if len(a[i]) != len(b[i]) {
                        return false
                }
        }

        for i := range a {
                if !slices.Equal(a[i], b[i]) {
                        return false
                }
        }

        return true
}

import (
        "regexp"

/*
   https://github.com/moisseev/rspamd/blob/master/rules/misc.lua

    Detect PRVS/BATV addresses to avoid FORGED_SENDER
    https://en.wikipedia.org/wiki/Bounce_Address_Tag_Validation

    Signature syntax:
        prvs=TAG=USER@example.com       BATV draft (https://tools.ietf.org/html/draft-levine-smtp-batv-01)
        prvs=USER=TAG@example.com
        btv1==TAG==USER@example.com     Barracuda appliance
        msprvs1=TAG=USER@example.com    Sparkpost email delivery service
*/

const (
        regexpBATV = "^(?:(?:prvs|msprvs1)=[^=]+=|btv1==[^=]+==)([^@]+@(?:[^@]+))$"
        regexpSRS  = "^([^+]+)\\+SRS=[^=]+=[^=]+=[^=]+=[^@]+@([^@]+)$"
)

var (
        compiledRegexpBATV = regexp.MustCompile(regexpBATV)
        compiledRegexpSRS  = regexp.MustCompile(regexpSRS)
)

// ParseBATV parses src to extract a BATV address.
// When BATV extration is not possible/needed src is returned.
func ParseBATV(src string) string {
        res := compiledRegexpBATV.FindStringSubmatch(src)
        if len(res) == 2 {
                return res[1]
        }
        return src
}

// ParseSRS parses src to extract the forwarding sender from SRS (Exchange Online).
// When SRS extration is not possible/needed src is returned.
func ParseSRS(src string) string {
        res := compiledRegexpSRS.FindStringSubmatch(src)
        if len(res) == 3 {
                return res[1] + "@" + res[2]
        }
        return src
}

// ParseSender combines ParseSRS and ParseBATV.
func ParseSender(src string) string {
        return ParseBATV(ParseSRS(src))
}

// UTF8Clean is an address validator for WithAddressValidator which rejects
// addresses containing invisible or direction changing characters.
//
// RFC 6531 allows every UTF-8 character in an address, but some of them make
// an address look different from what it is, or end a line in logs:
//
//   - bidirectional formatting characters (U+202A to U+202E, U+2066 to U+2069)
//   - line and paragraph separators (U+0085, U+2028, U+2029)
//   - zero width space, word joiner and byte order mark (U+200B, U+2060, U+FEFF)
//
// Zero width joiner and non-joiner (U+200D, U+200C) and the directional marks
// (U+200E, U+200F, U+061C) are accepted, since they are needed to spell names
// in some scripts. Other spaces (e.g. U+00A0) and the Unicode normalization
// form are not checked.
func UTF8Clean(addr string) error {
        if parse.IsASCII(addr) {
                return nil
        }

        for _, r := range addr {
                switch {
                case r >= 0x202A && r <= 0x202E, // bidirectional embedding and override
                        r >= 0x2066 && r <= 0x2069,            // bidirectional isolates
                        r == 0x0085, r == 0x2028, r == 0x2029, // line and paragraph separators
                        r == 0x200B, r == 0x2060, r == 0xFEFF: // zero width space, word joiner, BOM
                        return fmt.Errorf("address contains the disallowed character %U", r)
                }
        }

        return nil
}

// Backend is a SMTP server backend.
type Backend interface {
        NewSession(ctx context.Context, c *Conn) (context.Context, Session, error)
}

// BackendFunc is an adapter to allow the use of an ordinary function as a
// Backend.
type BackendFunc func(ctx context.Context, c *Conn) (context.Context, Session, error)

// NewSession calls f(c).
// The returning context is used in the session.
func (f BackendFunc) NewSession(ctx context.Context, c *Conn) (context.Context, Session, error) {
        return f(ctx, c)
}

// Session is used by servers to respond to an SMTP client.
//
// The methods are called when the remote client issues the matching command.
type Session interface {
        // Discard currently processed message.
        // The returning context replaces the context used in the current session.
        // Upgrade is true when the reset is called after a tls upgrade.
        Reset(ctx context.Context, upgrade bool) (context.Context, error)

        // Free all resources associated with session.
        // Error is set if an error occurred during session or connection.
        // Close is always called after the session is done.
        Close(ctx context.Context, err error)

        // Returns logger to use when an error occurs inside a session.
        // If no logger is returned the default *slog.Logger is used.
        Logger(ctx context.Context) *slog.Logger

        // Set return path for currently processed message.
        Mail(ctx context.Context, from string, opts *deltasmtp.MailOptions) error

        // Add recipient for currently processed message.
        Rcpt(ctx context.Context, to string, opts *deltasmtp.RcptOptions) error

        // Verify checks the validity of an email address on the server.
        // If error is nil then smtp code 252 is send
        // if error is smtp status then the smtp status is send
        // else internal server error is returned and connection is closed
        Verify(ctx context.Context, addr string, opts *deltasmtp.VrfyOptions) error

        // Set currently processed message contents and send it.
        // If r is called then the data must be consumed completely before returning.
        // The queuedid must not be unique.
        Data(ctx context.Context, r func() io.Reader) (queueid string, err error)

        // AuthMechanisms returns valid auth mechanism.
        // Nil or an empty list means no authentication mechanism is allowed.
        AuthMechanisms(ctx context.Context) []string

        // Auth returns a matching sasl server for the given mech.
        Auth(ctx context.Context, mech string) (sasl.Server, error)

        // STARTTLS returns a valid *tls.Config.
        // Is called with the default tls config and the returned tls config is used in the tls upgrade.
        // If the tls.Config is nil or an error is returned, the tls upgrade is aborted and the connection closed.
        // The *tls.Config received must not be changed.
        STARTTLS(ctx context.Context, tls *tls.Config) (*tls.Config, error)
}

// errMessageTooLarge is returned by the DATA and BDAT handlers if the message
// exceeded the maximum message size. The payload remaining on the wire was not
// consumed, so the connection is out of sync and must be closed.
var errMessageTooLarge = errors.New("smtp: maximum message size exceeded")

// errTooManyTransactions is returned by the MAIL handler if the client exceeded
// the transactions per session, it can only continue in a new session.
var errTooManyTransactions = errors.New("smtp: maximum transactions exceeded")

type state int32

const (
        stateInit                    state = 0
        stateUpgrade                 state = 1
        stateEnforceAuthentication   state = 2
        stateEnforceSecureConnection state = 3
        stateGreeted                 state = 4
        stateMail                    state = 5
)

// Conn is a connection inside a smtp server.
type Conn struct {
        ctx context.Context

        conn     net.Conn
        receiver *smtpreader.Receiver
        sender   *smtpwriter.Sender

        deadline deadline.Renewer // command deadline of conn

        state state

        server *Server

        session Session
        client  clientState // what the client negotiated
        tx      transaction // current mail transaction

        transactions int // MAIL commands in this session (RFC 9422 MAILMAX)

        data     messageReader // reads the message of DATA or BDAT
        bdatLine string        // the last BDAT command line, see commandLine
}

// messageReader holds the readers of the message content. They are reused for
// every message of a session, so receiving a message doesn't allocate them.
type messageReader struct {
        dot     smtpreader.Dot
        bdat    smtpreader.Bdat
        isBdat  bool // open returns bdat instead of dot
        started bool // DATA: the 354 reply was sent and dot is set up
        closed  bool // BDAT: the transfer failed, no further chunk is read

        // Bound to the Conn by bindMessageFuncs, the same funcs serve every
        // message of the session instead of new closures for each.
        open        func() io.Reader               // passed to Session.Data
        nextCommand func() (string, string, error) // reads the next BDAT command
}

// clientState holds what the server knows about the client. It is discarded
// after STARTTLS (RFC 3207 section 4.2).
type clientState struct {
        esmtp      bool     // EHLO was used instead of HELO
        helo       string   // domain given in HELO / EHLO
        mechanisms []string // auth mechanisms offered in EHLO
        didAuth    bool     // client is authenticated
}

// transaction holds the state of a mail transaction. It is reset by MAIL,
// RSET and after the message has been received.
type transaction struct {
        binarymime bool // BODY=BINARYMIME was given in MAIL
        smtputf8   bool // SMTPUTF8 was given in MAIL
        recipients int  // number of accepted recipients
}

// run loops until an error occurs (quit for example)
func (c *Conn) run() (err error) {
        var (
                cmd string
                arg string
        )
        c.greet()

        for {
                c.renewDeadline()
                cmd, arg, err = c.nextCommand()
                if err != nil {
                        return err
                }

                err = c.handle(cmd, arg)
                if err != nil {
                        // if error is a smtp status it isn't necessary to close the connection
                        if smtpErr, ok := errors.AsType[*deltasmtp.Status](err); ok {
                                // Service closing transmission channel, after quit
                                if smtpErr.Code == 221 {
                                        return smtpErr
                                }

                                // ToDo: close connection on repeated errors (e.g. authentication tries)

                                // fresh deadlines for the reply after a DATA or BDAT transfer
                                c.renewDeadline()
                                c.writeStatus(smtpErr)
                                continue
                        }

                        return err
                }
        }
}

// nextCommand reads the next line and parses it.
// The command is always returned upper case.
func (c *Conn) nextCommand() (cmd string, arg string, err error) {
        b, err := c.receiver.ReadFullLineBytes()
        if err != nil {
                return "", "", err
        }
        line := c.commandLine(b)
        c.logger().LogAttrs(c.ctx, slog.LevelDebug, "read", slog.String("line", line))
        return parse.Cmd(line)
}

// commandLine returns the command line b as a string. A string is only
// allocated for a line it hasn't returned just before: the commands without
// arguments are constants, and the BDAT line of every full chunk of a message
// is the same (comparing string(b) doesn't allocate). Other lines aren't kept,
// e.g. an AUTH line with credentials.
func (c *Conn) commandLine(b []byte) string {
        switch string(b) {
        case "DATA":
                return "DATA"
        case "RSET":
                return "RSET"
        case "NOOP":
                return "NOOP"
        case "QUIT":
                return "QUIT"
        case c.bdatLine:
                return c.bdatLine
        }

        line := string(b)
        if len(line) > 5 && strings.EqualFold(line[:5], "BDAT ") {
                c.bdatLine = line
        }
        return line
}

// Commands are dispatched to the appropriate handler functions.
func (c *Conn) handle(cmd string, arg string) error {
        if cmd == "" {
                return deltasmtp.ErrBadSyntax
        }
        switch c.state {
        case stateInit, stateUpgrade:
                return c.handleStateInit(cmd, arg)
        case stateEnforceSecureConnection:
                return c.handleStateEnforceSecureConnection(cmd, arg)
        case stateEnforceAuthentication:
                return c.handleStateEnforceAuthentication(cmd, arg)
        case stateGreeted:
                return c.handleStateGreeted(cmd, arg)
        case stateMail:
                return c.handleStateMail(cmd, arg)
        default:
                return fmt.Errorf("unsupported state %d, how?", c.state)
        }
}

func (c *Conn) handleStateInit(cmd string, arg string) error {
        switch cmd {
        case "HELO", "EHLO":
                return c.handleGreet(cmd == "EHLO", arg)
        case "NOOP":
                return deltasmtp.Noop
        case "VRFY":
                return c.handleVrfy(arg)
        case "RSET": // Reset session
                return c.handleRSET()
        case "QUIT":
                return deltasmtp.Quit
        default:
                return c.commandUnknown(cmd)
        }
}

func (c *Conn) handleStateEnforceAuthentication(cmd string, arg string) error {
        switch cmd {
        case "HELO", "EHLO":
                return c.handleGreet(cmd == "EHLO", arg)
        case "NOOP":
                return deltasmtp.Noop
        case "VRFY":
                return c.handleVrfy(arg)
        case "RSET": // Reset session
                return c.handleRSET()
        case "QUIT":
                return deltasmtp.Quit
        case "AUTH":
                // there is always a mechanism, as it is an enforce authentication precondition
                return c.handleAuth(arg)
        case "STARTTLS":
                return c.handleStartTLS()
        // RFC 5321, always consume bdat data if capabilities were sent
        case "BDAT":
                return c.handleBdatDiscard(arg)
        default:
                return deltasmtp.ErrAuthRequired
        }
}

func (c *Conn) handleStateGreeted(cmd string, arg string) error {
        switch cmd {
        case "HELO", "EHLO":
                return c.handleGreet(cmd == "EHLO", arg)
        case "MAIL":
                return c.handleMail(arg)
        case "NOOP":
                return deltasmtp.Noop
        case "VRFY":
                return c.handleVrfy(arg)
        case "RSET": // Reset session
                return c.handleRSET()
        case "QUIT":
                return deltasmtp.Quit
        case "AUTH":
                if len(c.client.mechanisms) > 0 {
                        return c.handleAuth(arg)
                }
                return deltasmtp.ErrAuthUnsupported
        case "STARTTLS":
                return c.handleStartTLS()
        // RFC 5321, always consume bdat data if capabilities were sent
        case "BDAT":
                return c.handleBdatDiscard(arg)
        default:
                return c.commandUnknown(cmd)
        }
}

func (c *Conn) handleStateMail(cmd string, arg string) error {
        switch cmd {
        case "HELO", "EHLO":
                return c.handleGreet(cmd == "EHLO", arg)
        case "RCPT":
                return c.handleRcpt(arg)
        case "NOOP":
                return deltasmtp.Noop
        case "VRFY":
                return c.handleVrfy(arg)
        case "RSET": // Reset session
                return c.handleRSET()
        case "BDAT":
                if !c.server.enableCHUNKING {
                        return c.handleBdatDiscard(arg)
                }
                return c.handleBdat(arg)
        case "DATA":
                return c.handleData(arg)
        case "QUIT":
                return deltasmtp.Quit
        case "STARTTLS":
                return c.handleStartTLS()
        default:
                return c.commandUnknown(cmd)
        }
}

func (c *Conn) handleStateEnforceSecureConnection(cmd string, arg string) error {
        switch cmd {
        case "HELO", "EHLO":
                return c.handleGreet(cmd == "EHLO", arg)
        case "NOOP":
                return deltasmtp.Noop
        case "VRFY":
                return c.handleVrfy(arg)
        case "STARTTLS":
                return c.handleStartTLS()
        case "QUIT":
                return deltasmtp.Quit
        // RFC 5321, always consume bdat data if capabilities were sent
        case "BDAT":
                return c.handleBdatDiscard(arg)
        default:
                return deltasmtp.ErrSecureConnectionRequired
        }
}

func (c *Conn) commandUnknown(cmd string) *deltasmtp.Status {
        return deltasmtp.NewStatusS(502, deltasmtp.EnhancedCode{5, 5, 1},
                fmt.Sprintf("%s command unknown, state %d", cmd, c.state))
}

// Close closes the connection.
func (c *Conn) Close(err error) {
        c.logger().DebugContext(c.ctx, "connection is closing")

        // flush any pending data from writer before closing connection, the
        // writer is gone if the connection was closed before
        if c.sender != nil {
                _ = c.sender.Flush()
        }

        closeErr := c.conn.Close()
        if closeErr != nil {
                if err == nil {
                        err = closeErr
                } else {
                        err = errors.Join(err, closeErr)
                }
        }

        if err != nil {
                c.logger().ErrorContext(c.ctx, "close error", slog.Any("err", err))
        }

        if c.session != nil {
                c.session.Close(c.ctx, err)
                c.session = nil
        }
}

func (c *Conn) handleRSET() error {
        err := c.reset()

// GREET state -> waiting for HELO
func (c *Conn) handleGreet(esmtp bool, arg string) error {
        domain, err := parse.HelloArgument(arg)
        if err != nil {
                return deltasmtp.ErrHeloArgRequired
        }

        // c.client.esmtp is populated before NewSession so
        // NewSession can access it via Conn.Esmtp.
        c.client.esmtp = esmtp

        // c.client.helo is populated before NewSession so
        // NewSession can access it via Conn.Hostname.
        c.client.helo = domain

        // RFC 5321: "An EHLO command MAY be issued by a client later in the session"
        // RFC 5321: "... the SMTP server MUST clear all buffers
        // and reset the state exactly as if a RSET command has been issued."
        if c.state != stateInit && c.state != stateEnforceSecureConnection && c.state != stateEnforceAuthentication {
                err := c.reset()
                if err != nil {
                        return err
                }
        }

        if c.server.enforceSecureConnection && !c.IsTLS() {
                c.state = stateEnforceSecureConnection
        } else if c.server.enforceAuthentication && !c.client.didAuth {
                c.state = stateEnforceAuthentication
        } else {
                c.state = stateGreeted
        }

        if !esmtp {
                return deltasmtp.NewStatusS(250, deltasmtp.NoEnhancedCode, c.server.hostname+" greets "+c.client.helo)
        }

        c.client.mechanisms = c.session.AuthMechanisms(c.ctx)

        if len(c.client.mechanisms) == 0 && c.server.enforceAuthentication {
                // without any auth mechanism, no authentication can happen => deadlock
                return c.newStatusError(deltasmtp.ErrNoAuthMechanism, err)
        }

        return c.handleGreetResponse()
}

func (c *Conn) handleGreetResponse() *deltasmtp.Status {
        ext := c.server.extensions
        lines := make([]string, 0, 3+len(ext.beforeTLS)+len(ext.afterAuth))

        lines = append(lines, c.server.hostname+" greets "+c.client.helo)
        lines = append(lines, ext.beforeTLS...)
        if !c.IsTLS() && c.server.tlsConfig != nil {
                lines = append(lines, "STARTTLS")
        }
        if len(c.client.mechanisms) > 0 {
                lines = append(lines, "AUTH "+strings.Join(c.client.mechanisms, " "))
        }
        lines = append(lines, ext.afterAuth...)

        return deltasmtp.NewStatusM(250, deltasmtp.NoEnhancedCode, lines)
}

// extensions are the EHLO reply lines which only depend on the server options,
// they are built once by New. STARTTLS and AUTH depend on the connection and
// are put in between.
type extensions struct {
        beforeTLS []string
        afterAuth []string
}

func newExtensions(s *Server) extensions {
        before := []string{"PIPELINING", "8BITMIME", "ENHANCEDSTATUSCODES"}
        if s.enableCHUNKING {
                before = append(before, "CHUNKING")
        }

        var after []string
        if s.enableSMTPUTF8 {
                after = append(after, "SMTPUTF8")
        }
        // We explicit allow require tls to be set if the connection is unencrypted
        // because if the server runs in a fully trusted zone
        // then the encryption is only relevant when it leaves this zone.
        if s.enableREQUIRETLS {
                after = append(after, "REQUIRETLS")
        }
        if s.enableBINARYMIME {
                after = append(after, "BINARYMIME")
        }
        if s.enableDSN {
                after = append(after, "DSN")
        }
        if s.enableXOORG {
                after = append(after, "XOORG")
        }
        if s.maxMessageBytes > 0 {
                after = append(after, "SIZE "+strconv.FormatInt(s.maxMessageBytes, 10))
        } else {
                after = append(after, "SIZE")
        }
        if limits := limits(s); limits != "" {
                after = append(after, "LIMITS"+limits)
        }
        if s.enableRRVS {
                after = append(after, "RRVS")
        }
        if s.enableDELIVERBY {
                // maximum of nine digits
                if s.minimumDeliverInSeconds > 0 && s.minimumDeliverInSeconds < 999999999 {
                        after = append(after, "DELIVERBY "+strconv.Itoa(s.minimumDeliverInSeconds))
                } else {
                        after = append(after, "DELIVERBY")
                }
        }
        if s.enableMTPRIORITY {
                if s.mtPriorityProfile != deltasmtp.PriorityUnspecified {
                        after = append(after, "MT-PRIORITY "+string(s.mtPriorityProfile))
                } else {
                        after = append(after, "MT-PRIORITY")
                }
        }

        return extensions{beforeTLS: before, afterAuth: after}
}

// limits returns the RFC 9422 LIMITS parameters with a leading space, or ""
// if there are none. Values are only valid from 1 to 999999, a larger limit
// isn't advertised (but still enforced).
func limits(s *Server) string {
        var sb strings.Builder
        for _, l := range []struct {
                prefix string
                value  int
        }{
                {" MAILMAX=", s.maxTransactions},
                {" RCPTMAX=", s.maxRecipients},
        } {
                if l.value > 0 && l.value <= 999999 {
                        sb.WriteString(l.prefix)
                        sb.WriteString(strconv.Itoa(l.value))
                }
        }
        return sb.String()
}

// handleError handles error and closes the connection afterwards.
func (c *Conn) handleError(err error) {
        // e.g. after a timeout the deadline is over, the reply needs a new one
        c.renewDeadline()

        if errors.Is(err, io.EOF) || errors.Is(err, net.ErrClosed) {
                c.Close(fmt.Errorf("connection closed unexpectedly: %w", err))
                return
        }

        if neterr, ok := errors.AsType[net.Error](err); ok && neterr.Timeout() {
                c.writeStatus(deltasmtp.ErrIdleTimeout)
                c.Close(fmt.Errorf("idle timeout: %w", err))
                return
        }

        if smtpErr, ok := errors.AsType[*deltasmtp.Status](err); ok {
                c.writeStatus(smtpErr)

                if smtpErr.Code != 221 {
                        c.Close(fmt.Errorf("smtp error: %w", err))
                } else {
                        c.Close(nil)
                }

                return
        }

        if errors.Is(err, errTooManyTransactions) {
                c.writeStatus(deltasmtp.ErrTooManyTransactions)
                c.Close(err)
                return
        }

        if errors.Is(err, errMessageTooLarge) {
                c.writeStatus(deltasmtp.ErrDataTooLarge)
                c.Close(err)
                return
        }

        if errors.Is(err, deltasmtp.ErrTooLongLine) {
                c.writeStatus(deltasmtp.ErrCommandLineTooLong)
                c.Close(errors.New("line too long"))
                return
        }

        c.writeStatus(deltasmtp.ErrConnection)
        c.Close(fmt.Errorf("unknown error: %w", err))
}

func (c *Conn) logger() *slog.Logger {
        // Fallback if the connection couldn't be created or is already closed.
        if c.session == nil {
                return slog.Default()
        }
        logger := c.session.Logger(c.ctx)
        if logger == nil {
                return slog.Default()
        }
        return logger
}

// READY state -> waiting for MAIL
//
//nolint:revive // function-length: one block per MAIL parameter, splitting hides the order.
func (c *Conn) handleMail(arg string) error {
        // RFC 9422: every MAIL command counts, regardless of whether it succeeds.
        c.transactions++
        if c.server.maxTransactions > 0 && c.transactions > c.server.maxTransactions {
                return errTooManyTransactions
        }

        arg, ok := parse.CutPrefixFold(arg, "FROM:")
        if !ok {
                return deltasmtp.ErrMailSyntax
        }

        p := parse.Parser{S: strings.TrimSpace(arg)}
        from, err := p.ReversePath()
        if err != nil {
                return c.newStatusError(deltasmtp.ErrMailSyntax, err)
        }
        var argsBuf [8]parse.Arg
        args, err := parse.Args(argsBuf[:0], p.S)
        if err != nil {
                return c.newStatusError(deltasmtp.ErrMailParams, err)
        }

        opts := &deltasmtp.MailOptions{}

        c.tx = transaction{}
        // This is where the Conn may put BODY=8BITMIME, but we already
        // read the DATA as bytes, so it does not effect our processing.
        for _, arg := range args {
                switch arg.Key {
                case "SIZE":
                        size, err := strconv.ParseUint(arg.Value, 10, 32)
                        if err != nil {
                                return deltasmtp.ErrMalformedSize
                        }

                        if c.server.maxMessageBytes > 0 && int64(size) > c.server.maxMessageBytes {
                                return deltasmtp.ErrDataTooLarge
                        }

                        opts.Size = int64(size)
                case "XOORG":
                        value, err := decodeXtext(arg.Value)
                        if err != nil || value == "" {
                                return deltasmtp.ErrMalformedXOORG
                        }
                        if !c.server.enableXOORG {
                                return deltasmtp.ErrXOORGNotImplemented
                        }
                        opts.XOORG = &value
                case "SMTPUTF8":
                        if !c.server.enableSMTPUTF8 {
                                return deltasmtp.ErrSMTPUTF8NotImplemented
                        }
                        opts.UTF8 = true
                case "REQUIRETLS":
                        if !c.server.enableREQUIRETLS {
                                return deltasmtp.ErrREQUIRETLSNotImplemented
                        }
                        opts.RequireTLS = true
                case "BODY":
                        value := strings.ToUpper(arg.Value)
                        switch deltasmtp.BodyType(value) {
                        case deltasmtp.BodyBinaryMIME:
                                if !c.server.enableBINARYMIME {
                                        return deltasmtp.ErrBINARYMIMENotImplemented
                                }
                                c.tx.binarymime = true
                        case deltasmtp.Body7Bit, deltasmtp.Body8BitMIME:
                                // This space is intentionally left blank
                        default:
                                return deltasmtp.ErrUnknownBody
                        }
                        opts.Body = deltasmtp.BodyType(value)
                case "RET":
                        if !c.server.enableDSN {
                                return deltasmtp.ErrRETNotImplemented
                        }
                        value := strings.ToUpper(arg.Value)
                        switch deltasmtp.DSNReturn(value) {
                        case deltasmtp.DSNReturnFull, deltasmtp.DSNReturnHeaders:
                                // This space is intentionally left blank
                        default:
                                return deltasmtp.ErrUnknownRET
                        }
                        opts.Return = deltasmtp.DSNReturn(value)
                case "ENVID":
                        if !c.server.enableDSN {
                                return deltasmtp.ErrENVIDNotImplemented
                        }
                        value, err := decodeXtext(arg.Value)
                        if err != nil || value == "" || !parse.IsPrintableASCII(value) {
                                return deltasmtp.ErrMalformedENVID
                        }
                        opts.EnvelopeID = value
                case "AUTH":
                        value, err := decodeXtext(arg.Value)
                        if err != nil || value == "" {
                                return deltasmtp.ErrMalformedAUTH
                        }
                        if value == "<>" {
                                value = ""
                        } else {
                                p := parse.Parser{S: value}
                                value, err = p.Mailbox()
                                if err != nil || p.S != "" {
                                        return deltasmtp.ErrMalformedAUTHMailbox
                                }
                        }
                        opts.Auth = &value
                case "BY":
                        if err := handleMailBY(c.server, opts, arg.Value); err != nil {
                                return err
                        }
                case "MT-PRIORITY":
                        if err := handleMailMTPRIORITY(c.server, opts, arg.Value); err != nil {
                                return err
                        }
                default:
                        return deltasmtp.ErrUnknownMailArg
                }
        }

        if !opts.UTF8 && !parse.IsASCII(from) {
                return deltasmtp.ErrNonASCIISender
        }

        // The null reverse-path is not an address.
        if from != "" {
                if err := c.validateAddress(from, deltasmtp.ErrSenderNotAccepted); err != nil {
                        return err
                }
        }

        c.tx.smtputf8 = opts.UTF8

        if err := c.session.Mail(c.ctx, from, opts); err != nil {
                if smtpErr, ok := errors.AsType[*deltasmtp.Status](err); ok {
                        // a positive response also counts as a success
                        if smtpErr.Positive() {
                                c.state = stateMail
                        }
                        return smtpErr
                }
                return c.newStatusError(deltasmtp.ErrMailNotAccepted, err)
        }

        c.state = stateMail
        c.writeOK("Roger, accepting mail from <", from, ">")
        return nil
}

func handleMailBY(server *Server, opts *deltasmtp.MailOptions, value string) error {
        if !server.enableDELIVERBY {
                return deltasmtp.ErrDELIVERBYNotImplemented
        }
        deliverBy := parseDeliverByArgument(value)
        if deliverBy == nil {
                return deltasmtp.ErrMalformedBY
        }
        if server.minimumDeliverInSeconds != 0 &&
                deliverBy.Mode == deltasmtp.DeliverByReturn &&
                deliverBy.Seconds < server.minimumDeliverInSeconds {
                return deltasmtp.ErrBYBelowMinimum
        }
        opts.DeliverBy = deliverBy
        return nil
}

// validateAddress calls the address validator of the server, if set.
// If it returns an error that is not a negative *deltasmtp.Status, the reply is
// rejected, a 553 (mailbox name not allowed, RFC 5321 section 4.2.2).
func (c *Conn) validateAddress(addr string, rejected *deltasmtp.Status) error {
        if c.server.addressValidator == nil {
                return nil
        }

        err := c.server.addressValidator(addr)
        if err == nil {
                return nil
        }

        if status, ok := errors.AsType[*deltasmtp.Status](err); ok && status.Positive() {
                // A positive reply would end the command without a result.
                err = errors.New("address validator returned a positive reply")
        }

        return c.newStatusError(rejected, err)
}

func handleMailMTPRIORITY(server *Server, opts *deltasmtp.MailOptions, value string) error {
        if !server.enableMTPRIORITY {
                return deltasmtp.ErrMTPRIORITYNotImplemented
        }
        mtPriority, err := strconv.Atoi(value)
        if err != nil {
                return deltasmtp.ErrMalformedMTPRIORITY
        }
        if mtPriority < -9 || mtPriority > 9 {
                return deltasmtp.ErrMTPRIORITYOutOfRange
        }
        opts.MTPriority = &mtPriority
        return nil
}

// MAIL state -> waiting for RCPTs followed by DATA
func (c *Conn) handleRcpt(arg string) error {
        arg, ok := parse.CutPrefixFold(arg, "TO:")
        if !ok {
                return deltasmtp.ErrRcptSyntax
        }

        p := parse.Parser{S: strings.TrimSpace(arg)}
        recipient, err := p.Path()
        if err != nil {
                return deltasmtp.ErrRcptSyntax
        }
        if !c.tx.smtputf8 && !parse.IsASCII(recipient) {
                return deltasmtp.ErrNonASCIIRecipient
        }

        if err := c.validateAddress(recipient, deltasmtp.ErrRcptAddressNotAccepted); err != nil {
                return err
        }

        if c.server.maxRecipients > 0 && c.tx.recipients >= c.server.maxRecipients {
                return deltasmtp.NewStatusS(452, deltasmtp.EnhancedCode{4, 5, 3},
                        fmt.Sprintf("Maximum limit of %v recipients reached", c.server.maxRecipients),
                )
        }

        var argsBuf [8]parse.Arg
        args, err := parse.Args(argsBuf[:0], p.S)
        if err != nil {
                return c.newStatusError(deltasmtp.ErrRcptParams, err)
        }

        opts := &deltasmtp.RcptOptions{}

        for _, arg := range args {
                switch arg.Key {
                case "NOTIFY":
                        if err := handleRcptNotify(c.server, opts, arg.Value); err != nil {
                                return err
                        }
                case "ORCPT":
                        if err := handleRcptORCPT(c.server, opts, arg.Value); err != nil {
                                return err
                        }
                case "RRVS":
                        if err := handleRcptRRVS(c.server, opts, arg.Value); err != nil {
                                return err
                        }
                default:
                        return deltasmtp.ErrUnknownRcptArg
                }
        }

        if err := c.session.Rcpt(c.ctx, recipient, opts); err != nil {
                if smtpErr, ok := errors.AsType[*deltasmtp.Status](err); ok {
                        // a positive response also counts as a success
                        if smtpErr.Positive() {
                                c.tx.recipients++
                        }
                        return smtpErr
                }
                return c.newStatusError(deltasmtp.ErrRcptNotAccepted, err)
        }

        c.tx.recipients++
        c.writeOK("I'll make sure <", recipient, "> gets this")
        return nil
}

func handleRcptNotify(server *Server, opts *deltasmtp.RcptOptions, value string) error {
        if !server.enableDSN {
                return deltasmtp.ErrNOTIFYNotImplemented
        }
        notify := []deltasmtp.DSNNotify{}
        for val := range strings.SplitSeq(value, ",") {
                notify = append(notify, deltasmtp.DSNNotify(strings.ToUpper(val)))
        }
        if err := parse.CheckNotifySet(notify); err != nil {
                return deltasmtp.ErrMalformedNOTIFY
        }
        opts.Notify = notify
        return nil
}

func handleRcptORCPT(server *Server, opts *deltasmtp.RcptOptions, value string) error {
        if !server.enableDSN {
                return deltasmtp.ErrORCPTNotImplemented
        }
        aType, aAddr, err := decodeTypedAddress(value)
        if err != nil || aAddr == "" {
                return deltasmtp.ErrMalformedORCPT
        }
        opts.OriginalRecipientType = aType
        opts.OriginalRecipient = aAddr
        return nil
}

func handleRcptRRVS(server *Server, opts *deltasmtp.RcptOptions, value string) error {
        if !server.enableRRVS {
                return deltasmtp.ErrRRVSNotImplemented
        }
        value, _, _ = strings.Cut(value, ";") // discard the no-support action
        rrvsTime, err := time.Parse(time.RFC3339, value)
        if err != nil {
                return deltasmtp.ErrMalformedRRVS
        }
        opts.RequireRecipientValidSince = rrvsTime
        return nil
}

func (c *Conn) handleVrfy(arg string) error {
        p := parse.Parser{S: strings.TrimSpace(arg)}
        vrfy, err := p.Path()
        if err != nil {
                return deltasmtp.ErrVrfySyntax
        }
        var argsBuf [8]parse.Arg
        args, err := parse.Args(argsBuf[:0], p.S)
        if err != nil {
                return c.newStatusError(deltasmtp.ErrVrfyParams, err)
        }

        opts := &deltasmtp.VrfyOptions{}

        for _, arg := range args {
                if arg.Key == "SMTPUTF8" {
                        if !c.server.enableSMTPUTF8 {
                                return deltasmtp.ErrSMTPUTF8NotImplemented
                        }
                        opts.UTF8 = true
                }
        }
        if !opts.UTF8 && !parse.IsASCII(vrfy) {
                return deltasmtp.ErrUTF8ReplyRequired
        }

        if err := c.validateAddress(vrfy, deltasmtp.ErrVrfyAddressNotAccepted); err != nil {
                return err
        }

        res := c.session.Verify(c.ctx, vrfy, opts)

        if res == nil {
                return deltasmtp.VRFY
        }

        return res
}

func (c *Conn) handleAuth(arg string) error {
        if c.client.didAuth {
                return deltasmtp.ErrAlreadyAuthenticated
        }
        parts := strings.Fields(arg)
        if len(parts) == 0 {
                return deltasmtp.ErrAuthMissingParam
        }

        mechanism := strings.ToUpper(parts[0])

        // Is mechanism allowed?
        if !slices.Contains(c.client.mechanisms, mechanism) {
                return deltasmtp.ErrAuthUnknownMechanism
        }

        // Parse client initial response if there is one
        var ir []byte
        if len(parts) > 1 {
                var err error
                ir, err = decodeSASLResponse(parts[1])
                if err != nil {
                        return deltasmtp.ErrInvalidBase64
                }
        }

        sasl, err := c.session.Auth(c.ctx, mechanism)
        if err != nil {
                return c.newStatusError(deltasmtp.ErrAuthTemporaryFailure, err)
        }

        if sasl == nil {
                return c.newStatusError(deltasmtp.ErrNoAuthHandler, err)
        }

        response := ir
        for {
                challenge, done, err := sasl.Next(response)
                if err != nil {
                        return c.newStatusError(deltasmtp.ErrAuthTemporaryFailure, err)
                }

                if done {
                        break
                }

                encoded := ""
                if len(challenge) > 0 {
                        encoded = base64.StdEncoding.EncodeToString(challenge)
                }
                c.writeStatus(deltasmtp.NewStatusS(334, deltasmtp.NoEnhancedCode, encoded))

                c.renewDeadline()
                encoded, err = c.readLine()
                if err != nil {
                        return err
                }

                if encoded == "*" {
                        // https://tools.ietf.org/html/rfc4954#page-4
                        return deltasmtp.ErrAuthCancelled
                }

                response, err = decodeSASLResponse(encoded)
                if err != nil {
                        return deltasmtp.ErrInvalidBase64
                }
        }

        c.client.didAuth = true
        if c.state == stateEnforceAuthentication {
                c.state = stateGreeted
        }

        return deltasmtp.AuthSucceeded
}

func (c *Conn) handleStartTLS() error {
        if _, isTLS := c.TLSConnectionState(); isTLS {
                return deltasmtp.ErrAlreadyTLS
        }

        if c.server.tlsConfig == nil {
                return deltasmtp.ErrTLSNotSupported
        }

        // allow the session to change tlsConfig
        tlsConfig, err := c.session.STARTTLS(c.ctx, c.server.tlsConfig)
        if err != nil {
                return c.newStatusError(deltasmtp.ErrTLSConfig, err)
        }
        if tlsConfig == nil {
                return deltasmtp.ErrTLSConfigNil
        }

        c.writeStatus(deltasmtp.StartTLS)

        // Upgrade to TLS
        tlsConn := tls.Server(c.conn, tlsConfig)

        if err := c.handshake(tlsConn); err != nil {
                // err is not a smtp error and will terminate the connection
                return err
        }

        c.conn = tlsConn
        c.receiver.Reset(tlsConn)
        c.sender.Reset(tlsConn)
        // RFC 3207 section 4.2: the server MUST discard any knowledge obtained
        // from the client, such as the argument to the EHLO command, which was
        // not obtained from the TLS negotiation itself. This also revokes the
        // authentication.
        c.client = clientState{}
        c.state = stateUpgrade // same as StateInit but calls logout/reset on ehlo/helo

        return nil
}

// DATA
func (c *Conn) handleData(arg string) error {
        // at least a single recipient needs to be set
        if c.tx.recipients == 0 {
                return deltasmtp.ErrNoRecipients
        }

        if arg != "" {
                return deltasmtp.ErrDataArgs
        }
        if c.tx.binarymime {
                return deltasmtp.ErrDataBINARYMIME
        }

        c.bindMessageFuncs()
        c.data.isBdat = false
        c.data.started = false

        uuid, err := c.session.Data(c.ctx, c.data.open)

        // an error which isn't a smtp status error will always terminate the connection
        if _, ok := errors.AsType[*deltasmtp.Status](err); err != nil && !ok {
                return err
        }

        // Make sure all the data has been consumed, regardless of what the backend returned.
        // The reader keeps returning ErrDataTooLarge once the limit is hit, so this also
        // detects a breach the backend swallowed. In that case the rest of the message is
        // still on the wire and the connection can't be resynchronised.
        if c.data.started {
                if _, discardErr := io.Copy(io.Discard, &c.data.dot); errors.Is(discardErr, deltasmtp.ErrDataTooLarge) {
                        return errMessageTooLarge
                }
        }

        // Without a 354 the client sent no message, a rejected DATA command leaves the transaction open.
        if !c.data.started && err != nil {
                return err
        }

        // RFC 5321 4.1.1.4: the end of mail data ends the transaction, whether the message
        // was accepted or rejected. Keeping it would leave sender and recipients behind,
        // a following MAIL is refused and a pipelined RCPT/DATA reuses the old envelope.
        if resetErr := c.reset(); resetErr != nil {
                return resetErr
        }
        if err != nil {
                return err
        }
        return c.accepted(uuid)
}

// discardBdat discards a BDAT chunk that isn't processed. Like a processed
// chunk, it is limited to the maximum message size. A larger chunk isn't read,
// it stays on the wire and the connection must be closed.
func (c *Conn) discardBdat(size int64) error {
        if c.server.maxMessageBytes > 0 && size > c.server.maxMessageBytes {
                return errMessageTooLarge
        }
        // A chunk size can exceed int on 32-bit platforms, discard it in steps.
        for size > 0 {
                step := min(size, math.MaxInt32)
                if _, err := c.receiver.Discard(int(step)); err != nil {
                        return err
                }
                size -= step
        }
        return nil
}

func (c *Conn) handleBdatDiscard(arg string) error {
        size, _, err := smtpreader.BdatArg(arg)
        if err != nil {
                return err
        }
        c.setDataDeadline()
        if err = c.discardBdat(size); err != nil {
                return err
        }
        if !c.server.enableCHUNKING {
                return deltasmtp.ErrCHUNKINGNotImplemented
        }
        return deltasmtp.ErrBadSequence
}

func (c *Conn) handleBdat(arg string) error {
        size, last, err := smtpreader.BdatArg(arg)
        if err != nil {
                return err
        }

        // The whole transfer, all chunks until LAST, has to finish within the data timeout.
        // The BDAT lines between chunks are read without renewing the deadline.
        c.setDataDeadline()

        // at least a single recipient needs to be set
        if c.tx.recipients == 0 {
                if err = c.discardBdat(size); err != nil {
                        return err
                }
                return deltasmtp.ErrNoRecipients
        }

        c.bindMessageFuncs()
        c.data.isBdat = true
        c.data.closed = false
        c.data.bdat.Reset(size, last, c.server.maxMessageBytes, c.receiver, c.data.nextCommand)

        queueid, err := c.session.Data(c.ctx, c.data.open)

        smtpErr, isStatus := errors.AsType[*deltasmtp.Status](err)

        // an error which isn't a SMTPStatus error will always terminate the connection
        if err != nil && !isStatus {
                return err
        }

        if err != nil {
                // read anything left to continue after this failure
                // https://www.rfc-editor.org/rfc/rfc3030.html
                // If a 5XX or 4XX code is received by the sender-SMTP in response to a BDAT
                // chunk, the transaction should be considered failed and the sender-
                // SMTP MUST NOT send any additional BDAT segments.  If the receiver-
                // SMTP has declared support for command pipelining [PIPE], the receiver
                // SMTP MUST be prepared to accept and discard additional BDAT chunks
                // already in the pipeline after the failed BDAT.
                c.data.closed = true
        }

        // Make sure all the data has been consumed, regardless of what the backend returned.
        // The reader keeps returning ErrDataTooLarge once the limit is hit, so this also
        // detects a breach the backend swallowed. In that case the chunk is still on the
        // wire and the connection can't be resynchronised.
        if _, discardErr := io.Copy(io.Discard, &c.data.bdat); errors.Is(discardErr, deltasmtp.ErrDataTooLarge) {
                return errMessageTooLarge
        }

        if err != nil {
                // write down error after data is discarded to prevent deadlock because of pipelining
                c.writeStatus(smtpErr)
                return c.reset()
        }

        if err = c.reset(); err != nil {
                return err
        }
        return c.accepted(queueid)
}

// bindMessageFuncs binds the funcs of c.data on the first message, they serve
// every message of the session.
func (c *Conn) bindMessageFuncs() {
        if c.data.open == nil {
                c.data.open = c.messageContent
                c.data.nextCommand = c.nextBdatCommand
        }
}

// messageContent returns the reader of the message. For DATA the 354 reply is
// sent on the first call, a backend that never calls it rejects the message
// before the client sends it.
func (c *Conn) messageContent() io.Reader {
        if c.data.isBdat {
                return &c.data.bdat
        }
        if !c.data.started {
                c.data.started = true
                // We have recipients, go to accept data
                c.writeStatus(deltasmtp.StartMailInput)
                c.setDataDeadline()
                c.data.dot.Reset(c.receiver.Reader, c.server.maxMessageBytes)
        }
        return &c.data.dot
}

// nextBdatCommand reads the BDAT command of the next chunk after replying to
// the previous one.
func (c *Conn) nextBdatCommand() (cmd string, arg string, err error) {
        // if bdat is closed (error occurred)
        if c.data.closed {
                return "", "", io.EOF
        }
        // the reply between chunks isn't limited by the data deadline
        deadline.SetWrite(c.conn, c.server.commandTimeout)
        c.writeStatus(deltasmtp.BDATContinue)
        return c.nextCommand()
}

func (*Conn) accepted(queueid string) *deltasmtp.Status {
        if queueid != "" {
                // limit length if queueid is too long (< 1000)
                if len(queueid) > 977 {
                        queueid = queueid[:974] + "..."
                }
                return deltasmtp.NewStatusS(250, deltasmtp.EnhancedCode{2, 0, 0}, "OK: queued as "+queueid)
        }
        return deltasmtp.Queued
}

func (c *Conn) greet() {
        c.writeStatus(c.server.greeting)
}

func (c *Conn) writeStatus(status *deltasmtp.Status) {
        c.logger().LogAttrs(c.ctx, slog.LevelDebug, "write status", slog.Any("status", status))

        _, err := status.WriteTo(c.sender)
        c.flushReply(err)
}

// writeOK writes the single line reply "250 2.0.0 " followed by the parts of
// text. Unlike writeStatus it needs neither a Status nor the joined text, the
// replies to MAIL and RCPT, which name the address, don't allocate.
func (c *Conn) writeOK(text ...string) {
        // Like a status returned to run, the reply gets a fresh deadline after
        // the session handled the command.
        c.renewDeadline()

        if logger := c.logger(); logger.Enabled(c.ctx, slog.LevelDebug) {
                logger.LogAttrs(c.ctx, slog.LevelDebug, "write status",
                        slog.Any("status", deltasmtp.NewStatusS(250, deltasmtp.EnhancedCode{2, 0, 0}, strings.Join(text, ""))))
        }

        b := append(c.sender.AvailableBuffer(), "250 2.0.0 "...)
        for _, t := range text {
                b = append(b, t...)
        }
        b = append(b, '\r', '\n')

        _, err := c.sender.Write(b)
        c.flushReply(err)
}

// flushReply finishes writing a reply, err is the error of writing it.
func (c *Conn) flushReply(err error) {
        // PIPELINE support
        // If there is something buffered in c.text.R then we can assume another command is following.
        // This means the client is doing pipelining and we don't need to respond just now.
        if err == nil && c.receiver.Buffered() == 0 {
                err = c.sender.Flush()
        }

        // A failed write (e.g. write timeout) loses the reply, the session can't go on.
        // Closing the connection makes the next read fail and ends the session.
        if err != nil {
                _ = c.conn.Close()
        }
}

// newStatusError returns err if it is a status, otherwise it logs err and returns status.
func (c *Conn) newStatusError(status *deltasmtp.Status, err error) *deltasmtp.Status {
        if smtpErr, ok := errors.AsType[*deltasmtp.Status](err); ok {
                return smtpErr
        }
        c.logger().ErrorContext(c.ctx, status.Text(), slog.Any("err", err))
        return status
}

// renewDeadline renews the deadline to at least now plus the command timeout,
// a zero timeout means no deadline. It is called before a command line is read
// and covers the reply to the command as well.
func (c *Conn) renewDeadline() {
        c.deadline.Set(c.conn, c.server.commandTimeout)
}

// setDataDeadline sets the read and write deadline for a whole DATA or BDAT
// transfer to now plus the data timeout, a zero data timeout means no deadline.
func (c *Conn) setDataDeadline() {
        deadline.Set(c.conn, c.server.dataTimeout)
        c.deadline.Reset()
}

// handshake runs the TLS handshake within the TLS handshake timeout,
// the next renewal sets the command deadline again.
func (c *Conn) handshake(tlsConn *tls.Conn) error {
        deadline.Set(tlsConn, c.server.tlsHandshakeTimeout)
        c.deadline.Reset()
        return tlsConn.HandshakeContext(c.ctx)
}

// readLine reads a line of input.
// It doesn't touch the read deadline, the caller has to set it.
func (c *Conn) readLine() (string, error) {
        line, err := c.receiver.ReadFullLine()
        if err == nil {
                c.logger().LogAttrs(c.ctx, slog.LevelDebug, "read", slog.String("line", line))
        }
        return line, err
}

func (c *Conn) reset() error {
        // Reset state to Greeted
        if c.state == stateMail {
                c.state = stateGreeted
        }

        c.tx = transaction{}

        ctx, err := c.session.Reset(c.ctx, c.state == stateUpgrade)
        c.ctx = ctx
        return err
}

// Parses the BY argument defined in RFC2852 section 4.
// Returns pointer to options or nil if invalid.
func parseDeliverByArgument(arg string) *deltasmtp.DeliverByOptions {
        secondsStr, modeStr, ok := strings.Cut(arg, ";")
        if !ok {
                return nil
        }
        modeStr, traceValue := strings.CutSuffix(modeStr, "T")
        if modeStr != string(deltasmtp.DeliverByNotify) && modeStr != string(deltasmtp.DeliverByReturn) {
                return nil
        }
        modeValue := deltasmtp.DeliverByMode(modeStr)
        secondsValue, err := strconv.Atoi(secondsStr)
        if err != nil ||
                (modeValue == deltasmtp.DeliverByReturn && (secondsValue < 1 || secondsValue > 999999999)) ||
                (modeValue == deltasmtp.DeliverByNotify && (secondsValue < -999999999 || secondsValue > 999999999)) {
                return nil
        }
        return &deltasmtp.DeliverByOptions{
                Seconds: secondsValue,
                Mode:    modeValue,
                Trace:   traceValue,
        }
}

func decodeSASLResponse(s string) ([]byte, error) {
        if s == "=" {
                return []byte{}, nil
        }
        return base64.StdEncoding.DecodeString(s)
}

// This regexp matches 'hexchar' token defined in
// https://tools.ietf.org/html/rfc4954#section-8 however it is intentionally
// relaxed by requiring only '+' to be present.  It allows us to detect
// malformed values such as +A or +HH and report them appropriately.
var hexcharRe = regexp.MustCompile(`\+[0-9A-F]?[0-9A-F]?`)

func decodeXtext(val string) (string, error) {
        if !strings.Contains(val, "+") {
                return val, nil
        }

        var replaceErr error
        decoded := hexcharRe.ReplaceAllStringFunc(val, func(match string) string {
                if len(match) != 3 {
                        replaceErr = errors.New("incomplete hexchar")
                        return ""
                }
                char, err := strconv.ParseInt(match, 16, 8)
                if err != nil {
                        replaceErr = err
                        return ""
                }

                return string(rune(char))
        })
        if replaceErr != nil {
                return "", replaceErr
        }

        return decoded, nil
}

// This regexp matches 'EmbeddedUnicodeChar' token defined in
// https://datatracker.ietf.org/doc/html/rfc6533.html#section-3
// however it is intentionally relaxed by requiring only '\x{HEX}' to be
// present.  It also matches disallowed characters in QCHAR and QUCHAR defined
// in above.
// So it allows us to detect malformed values and report them appropriately.
var eUOrDCharRe = regexp.MustCompile(`\\x[{][0-9A-F]+[}]|[[:cntrl:] \\+=]`)

// Decodes the utf-8-addr-xtext or the utf-8-addr-unitext form.
func decodeUTF8AddrXtext(val string) (string, error) {
        var replaceErr error
        decoded := eUOrDCharRe.ReplaceAllStringFunc(val, func(match string) string {
                if len(match) == 1 {
                        replaceErr = errors.New("disallowed character:" + match)
                        return ""
                }

                hexpoint := match[3 : len(match)-1]
                char, err := strconv.ParseUint(hexpoint, 16, 21)
                if err != nil {
                        replaceErr = err
                        return ""
                }
                switch len(hexpoint) {
                case 2:
                        switch {
                        // all xtext-specials
                        case 0x01 <= char && char <= 0x09 ||
                                0x11 <= char && char <= 0x19 ||
                                char == 0x10 || char == 0x20 ||
                                char == 0x2B || char == 0x3D || char == 0x7F:
                        // 2-digit forms
                        case char == 0x5C || 0x80 <= char && char <= 0xFF:
                                // This space is intentionally left blank
                        default:
                                replaceErr = errors.New("illegal hexpoint:" + hexpoint)
                                return ""
                        }
                // 3-digit forms
                case 3:
                        switch {
                        case 0x100 <= char && char <= 0xFFF:
                                // This space is intentionally left blank
                        default:
                                replaceErr = errors.New("illegal hexpoint:" + hexpoint)
                                return ""
                        }
                // 4-digit forms excluding surrogate
                case 4:
                        switch {
                        case 0x1000 <= char && char <= 0xD7FF:
                        case 0xE000 <= char && char <= 0xFFFF:
                                // This space is intentionally left blank
                        default:
                                replaceErr = errors.New("illegal hexpoint:" + hexpoint)
                                return ""
                        }
                // 5-digit forms
                case 5:
                        switch {
                        case 0x1_0000 <= char && char <= 0xF_FFFF:
                                // This space is intentionally left blank
                        default:
                                replaceErr = errors.New("illegal hexpoint:" + hexpoint)
                                return ""
                        }
                // 6-digit forms
                case 6:
                        switch {
                        case 0x10_0000 <= char && char <= 0x10_FFFF:
                                // This space is intentionally left blank
                        default:
                                replaceErr = errors.New("illegal hexpoint:" + hexpoint)
                                return ""
                        }
                // the other invalid forms
                default:
                        replaceErr = errors.New("illegal hexpoint:" + hexpoint)
                        return ""
                }

                return string(rune(char))
        })
        if replaceErr != nil {
                return "", replaceErr
        }

        return decoded, nil
}

func decodeTypedAddress(val string) (deltasmtp.DSNAddressType, string, error) {
        aType, aAddr, ok := strings.Cut(val, ";")
        if !ok || aType == "" || aAddr == "" {
                return "", "", errors.New("bad address")
        }
        aType = strings.ToUpper(aType)

        var err error
        switch deltasmtp.DSNAddressType(aType) {
        case deltasmtp.DSNAddressTypeRFC822:
                aAddr, err = decodeXtext(aAddr)
                if err == nil && !parse.IsPrintableASCII(aAddr) {
                        err = errors.New("illegal address:" + aAddr)
                }
        case deltasmtp.DSNAddressTypeUTF8:
                aAddr, err = decodeUTF8AddrXtext(aAddr)
        default:
                err = errors.New("unknown address type:" + aType)
        }
        if err != nil {
                return "", "", err
        }

        return deltasmtp.DSNAddressType(aType), aAddr, nil
}

// Additional extensions may be handled by other packages.

// Serve accepts incoming connections on the Listener l.
func (s *Server) Serve(ctx context.Context, l net.Listener) error {
        s.locker.Lock()
        s.listeners = append(s.listeners, l)
        s.locker.Unlock()

        var tempDelay time.Duration // how long to sleep on accept failure

        for {
                c, err := l.Accept()
                if err != nil {
                        select {
                        case <-s.done:
                                // we called Close()
                                return nil
                        default:
                        }
                        if ne, ok := errors.AsType[net.Error](err); ok && ne.Timeout() {
                                // Doubles from 5ms up to 1s.
                                tempDelay = min(max(2*tempDelay, 5*time.Millisecond), time.Second)
                                s.logger.ErrorContext(
                                        ctx,
                                        "accept error, retrying",
                                        slog.Any("err", err),
                                        slog.Any("temp_delay", tempDelay),
                                )
                                time.Sleep(tempDelay)
                                continue
                        }
                        return err
                }

                s.wg.Add(1)
                go s.handleConn(ctx, c)
        }
}

func (s *Server) handleConn(ctx context.Context, conn net.Conn) {
        ctx, cancel := context.WithCancel(ctx)

        c := &Conn{
                ctx:      ctx,
                server:   s,
                conn:     conn,
                sender:   s.newSender(conn),
                receiver: s.newReceiver(conn),
        }
        c.renewDeadline() // for the greeting

        var ip string
        if s.maxConnectionsPerIP > 0 {
                ip = remoteIP(conn)
        }
        limited := ip != ""

        s.locker.Lock()
        s.conns[c] = struct{}{}
        if limited {
                s.connsByIP[ip]++
        }
        exceeded := s.maxConnections > 0 && len(s.conns) > s.maxConnections
        exceededPerIP := limited && s.connsByIP[ip] > s.maxConnectionsPerIP
        s.locker.Unlock()

        var err error

        defer func() {
                if err := recover(); err != nil {
                        c.writeStatus(deltasmtp.ErrInternal)
                        stack := debug.Stack()
                        c.logger().ErrorContext(
                                c.ctx,
                                "panic serving",
                                slog.Any("err", err),
                                slog.Any("stack", string(stack)),
                        )
                        c.Close(errors.New("recovered from panic inside handleConn"))
                }

                s.locker.Lock()
                delete(s.conns, c)
                if limited {
                        if s.connsByIP[ip]--; s.connsByIP[ip] <= 0 {
                                delete(s.connsByIP, ip)
                        }
                }
                s.locker.Unlock()

                s.releaseBuffers(c)

                s.wg.Done()

                cancel()
        }()

        if exceeded || exceededPerIP {
                // Overall limit first, the client isn't at fault for a busy server.
                status := deltasmtp.ErrTooManyConnections
                if !exceeded {
                        // Same reply as Postfix for smtpd_client_connection_count_limit.
                        status = deltasmtp.NewStatusS(421, deltasmtp.EnhancedCode{4, 7, 0},
                                fmt.Sprintf("%s Error: too many connections from %s", s.hostname, ip))
                }
                c.writeStatus(status)
                c.Close(nil)
                return
        }

        sctx, session, err := s.backend.NewSession(ctx, c)
        if err != nil {
                c.Close(fmt.Errorf("couldn't create connection wrapper: %w", err))
                return
        }

        // update ctx and set session
        c.ctx = sctx
        c.session = session

        c.logger().DebugContext(c.ctx, "connection is opened")

        // explicit tls handshake call
        if tlsConn, ok := c.conn.(*tls.Conn); ok {
                if err := c.handshake(tlsConn); err != nil {
                        c.handleError(err)
                        return
                }
        }

        // run always returns an error when finished
        c.handleError(c.run())
}

// newSender returns the writer of a new connection, with the buffer of a
// closed one if possible.
func (s *Server) newSender(conn net.Conn) *smtpwriter.Sender {
        if sender, ok := s.senders.Get().(*smtpwriter.Sender); ok {
                sender.Reset(conn)
                return sender
        }
        return smtpwriter.NewSender(conn, s.writerSize)
}

// newReceiver returns the reader of a new connection, with the buffer of a
// closed one if possible.
func (s *Server) newReceiver(conn net.Conn) *smtpreader.Receiver {
        if receiver, ok := s.receivers.Get().(*smtpreader.Receiver); ok {
                receiver.Reset(conn)
                return receiver
        }
        return smtpreader.NewReceiver(conn, s.readerSize, s.maxLineLength, 0)
}

// releaseBuffers returns the buffers of the closed connection c for reuse.
// The connection can't read or write anymore, Close does nothing more than
// closing it again.
func (s *Server) releaseBuffers(c *Conn) {
        sender, receiver := c.sender, c.receiver
        c.sender, c.receiver = nil, nil

        // Drop the connection, a pooled buffer must not keep it alive.
        sender.Reset(nil)
        receiver.Reset(nil)
        s.senders.Put(sender)
        s.receivers.Put(receiver)
}

// remoteIP returns the remote IP of a TCP connection, or "" for other networks.
func remoteIP(conn net.Conn) string {
        if addr, ok := conn.RemoteAddr().(*net.TCPAddr); ok {
                return addr.IP.String()
        }
        return ""
}

// Listen listens on the network address s.Addr
// to handle requests on incoming connections.
//
// If s.Addr is blank and LMTP is disabled, ":smtp" is used.
func (s *Server) Listen() (net.Listener, error) {
        network := cmp.Or(s.network, "tcp")
        addr := cmp.Or(s.addr, ":smtp")

        var l net.Listener
        var err error

        if s.implicitTLS {
                l, err = tls.Listen(network, addr, s.tlsConfig)
        } else {
                l, err = net.Listen(network, addr)
        }

        if err != nil {
                return nil, err
        }

        return l, nil
}

// ListenAndServe listens on the network address s.Addr and then calls Serve
// to handle requests on incoming connections.
//
// If s.Addr is blank and LMTP is disabled, ":smtp" is used.
func (s *Server) ListenAndServe(ctx context.Context) error {
        network := cmp.Or(s.network, "tcp")
        addr := cmp.Or(s.addr, ":smtp")

        var l net.Listener
        var err error

        if s.implicitTLS {
                l, err = tls.Listen(network, addr, s.tlsConfig)
        } else {
                l, err = net.Listen(network, addr)
        }

        if err != nil {
                return err
        }

        return s.Serve(ctx, l)
}

// Close immediately closes all active listeners and connections.
//
// Close returns any error returned from closing the server's underlying
// listener(s).
func (s *Server) Close() error {
        select {
        case <-s.done:
                return ErrServerClosed
        default:
                close(s.done)
        }

        var err error
        s.locker.Lock()
        for _, l := range s.listeners {
                if lerr := l.Close(); lerr != nil && err == nil {
                        err = lerr
                }
        }

        for conn := range s.conns {
                // directly close underlying connection
                _ = conn.conn.Close()
        }
        s.locker.Unlock()

        return err
}

// Shutdown gracefully shuts down the server without interrupting any
// active connections. Shutdown works by first closing all open
// listeners and then waiting indefinitely for connections to return to
// idle and then shut down.
// If the provided context expires before the shutdown is complete,
// Shutdown returns the context's error, otherwise it returns any
// error returned from closing the Server's underlying Listener(s).
func (s *Server) Shutdown(ctx context.Context) error {
        select {
        case <-s.done:
                return ErrServerClosed
        default:
                close(s.done)
        }

        var err error
        s.locker.Lock()
        for _, l := range s.listeners {
                if lerr := l.Close(); lerr != nil && err == nil {
                        err = lerr
                }
        }
        s.locker.Unlock()

        connDone := make(chan struct{})
        go func() {
                defer close(connDone)
                s.wg.Wait()
        }()

        select {
        case <-ctx.Done():
                return ctx.Err()
        case <-connDone:
                return err
        }
}

// Server implements a SMTP server.
type Server struct {
        // The type of network, "tcp" or "unix".
        network string
        // TCP or Unix address to listen on.
        addr string
        // The server TLS configuration.
        tlsConfig *tls.Config

        hostname string

        greeting   *deltasmtp.Status // see New
        extensions extensions        // see New

        maxRecipients int
        // Max transactions (MAIL commands) per session.
        maxTransactions int
        // Max concurrent connections.
        maxConnections int
        // Max concurrent connections per remote IP.
        maxConnectionsPerIP int
        // Max line length for every command except data and bdat.
        maxLineLength int
        // Maximum size when receiving data and bdat.
        maxMessageBytes int64
        // Reader buffer size.
        readerSize int
        // Writer buffer size.
        writerSize int

        commandTimeout      time.Duration
        dataTimeout         time.Duration
        tlsHandshakeTimeout time.Duration

        implicitTLS bool

        // Enforces usage of implicit tls or starttls before accepting commands except NOOP, EHLO, STARTTLS, or QUIT.
        enforceSecureConnection bool

        // Enforces usage of authentication.
        enforceAuthentication bool

        // Advertise SMTPUTF8 (RFC 6531) capability.
        // Should be used only if backend supports it.
        enableSMTPUTF8 bool

        // Advertise REQUIRETLS (RFC 8689) capability.
        // Should be used only if backend supports it.
        enableREQUIRETLS bool

        // Advertise CHUNKING (RFC 3030) capability, defaults to true.
        enableCHUNKING bool

        // Advertise BINARYMIME (RFC 3030) capability.
        // Should be used only if backend supports it.
        enableBINARYMIME bool

        // Advertise DSN (RFC 3461) capability.
        // Should be used only if backend supports it.
        enableDSN bool

        // Advertise XOORG capability.
        // Should be used only if backend supports it.
        enableXOORG bool

        // Advertise RRVS (RFC 7293) capability.
        // Should be used only if backend supports it.
        enableRRVS bool

        // Advertise DELIVERBY (RFC 2852) capability.
        // Should be used only if backend supports it.
        enableDELIVERBY bool
        // The minimum time, with seconds precision, that a client
        // may specify in the BY argument with return mode.
        // A zero value indicates no set minimum.
        // Only use if DELIVERBY is enabled.
        // 99999999999 is the maximum and 1 the minimum value allowed.
        minimumDeliverInSeconds int

        // Advertise MT-PRIORITY (RFC 6710) capability.
        // Should only be used if backend supports it.
        enableMTPRIORITY bool
        // The priority profile mapping as defined
        // in RFC 6710 section 10.2.
        //
        // Default value of NONE to advertise no specific profile.
        mtPriorityProfile deltasmtp.PriorityProfile

        // Additional check for every parsed mailbox, nil means no check.
        addressValidator func(addr string) error

        // The server backend.
        backend Backend

        logger *slog.Logger

        wg   sync.WaitGroup
        done chan struct{}

        locker    sync.Mutex
        listeners []net.Listener
        conns     map[*Conn]struct{}
        connsByIP map[string]int

        // Buffers of closed connections, reused by new ones. All connections of a
        // server have the same buffer sizes.
        senders   sync.Pool // *smtpwriter.Sender
        receivers sync.Pool // *smtpreader.Receiver
}

// Backend returns the servers Backend.
func (s *Server) Backend() Backend {
        return s.backend
}

// Option is an option for the server.
type Option func(*Server)

// New creates a new SMTP server.
//
// The server is bounded by default (line length, message size, recipients,
// concurrent connections (total and per IP) and read/write timeouts). Each limit can be
// disabled explicitly by setting it to zero with the corresponding option.
func New(opts ...Option) *Server {
        s := &Server{
                done:      make(chan struct{}, 1),
                conns:     make(map[*Conn]struct{}),
                connsByIP: make(map[string]int),
                hostname:  "localhost",

                // RFC 4954 extends the maximum line length of the AUTH command (and
                // the SASL responses) to 12288 octets, which also covers commands with
                // all usual ESMTP parameters (RFC 5321 allows 512 octets plus extensions).
                maxLineLength: 12288,

                // 32 MiB
                maxMessageBytes: 32 << 20,

                // RFC 5321 requires at least 100 recipients to be accepted.
                maxRecipients: 1000,

                // Exim's default for smtp_accept_max_per_connection.
                maxTransactions: 1000,

                // Protects against file descriptor and memory exhaustion while being
                // high enough for a busy MTA.
                maxConnections: 1000,

                // Common MTA default (e.g. Postfix smtpd_client_connection_count_limit).
                maxConnectionsPerIP: 10,

                // RFC 6531 allows them, but invisible and direction changing characters
                // make an address look different from what it is.
                addressValidator: UTF8Clean,

                // BDAT (RFC 3030) is passed to Session.Data like DATA, so the backend
                // needs no support for it.
                enableCHUNKING: true,

                // RFC 5321 (Section 4.5.3.2.7) recommends at least 5 minutes.
                commandTimeout: 5 * time.Minute,

                // Whole DATA or BDAT transfer, RFC 5321 (Section 4.5.3.2.6) lets a
                // client wait 10 minutes for the reply after the data, so the transfer
                // itself gets the same. With 32 MiB this requires about 55 KB/s.
                dataTimeout: 10 * time.Minute,

                // Implicit TLS and STARTTLS, very generous like the client.
                tlsHandshakeTimeout: 30 * time.Second,

                // Commands and the message are read through this buffer. DATA is read
                // line by line, with 32 KiB a 13 MB message needs about 400 instead
                // of 3300 reads, about 3 times faster. Larger buffers gain little.
                readerSize: 32 << 10,

                // The server only writes replies, they are small.
                writerSize: 4096,
        }

        for _, o := range opts {
                o(s)
        }

        s.logger = cmp.Or(s.logger, slog.Default())

        // Replies which only depend on the options, built once instead of for
        // every connection.
        s.greeting = deltasmtp.NewStatusS(220, deltasmtp.NoEnhancedCode, s.hostname+" ESMTP Service Ready")
        s.extensions = newExtensions(s)

        return s
}

        Crnl = []byte{'\r', '\n'}
        // Dotcrnl .\r\n
        Dotcrnl = []byte{'.', '\r', '\n'}
        // Crlfdot \r\n.
        Crlfdot = []byte{'\r', '\n', '.'}

// EnhancedCode is the SMTP enhanced code
type EnhancedCode [3]int

// Status specifies the error code, enhanced error code (if any) and
// message returned by the server.
type Status struct {
        Code         int
        EnhancedCode EnhancedCode
        Lines        []string
}

// NoEnhancedCode is used to indicate that enhanced error code should not be
// included in response.
//
// Note that RFC 2034 requires an enhanced code to be included in all 2xx, 4xx
// and 5xx responses.
var NoEnhancedCode = EnhancedCode{0, 0, 0}

// ToPart returns the part of the string after the code
// which is defined by the enhanced code with the trailing whitespace.
// E.g. "5.1.1 " or "5.7.26 " (RFC 3463: subject and detail have up to 3 digits).
func (enhCode EnhancedCode) ToPart() []byte {
        if enhCode == NoEnhancedCode {
                return nil
        }
        // 10 is the longest valid code with the trailing space, e.g. "5.123.123 ".
        return enhCode.appendPart(make([]byte, 0, 10))
}

// appendPart appends ToPart to b, nothing for NoEnhancedCode.
func (enhCode EnhancedCode) appendPart(b []byte) []byte {
        if enhCode == NoEnhancedCode {
                return b
        }
        b = strconv.AppendInt(b, int64(enhCode[0]), 10)
        b = append(b, '.')
        b = strconv.AppendInt(b, int64(enhCode[1]), 10)
        b = append(b, '.')
        b = strconv.AppendInt(b, int64(enhCode[2]), 10)
        return append(b, ' ')
}

// NewStatusM creates a new status with multiple message lines.
func NewStatusM(code int, enhCode EnhancedCode, msg []string) *Status {
        return &Status{
                Code:         code,
                EnhancedCode: enhCode,
                Lines:        msg,
        }
}

// NewStatusS creates a new status with a single message line.
func NewStatusS(code int, enhCode EnhancedCode, msg string) *Status {
        // The status and its line in one allocation instead of two.
        s := &struct {
                Status
                line [1]string
        }{line: [1]string{msg}}
        s.Status = Status{Code: code, EnhancedCode: enhCode, Lines: s.line[:]}
        return &s.Status
}

// Error returns a error string.
func (s *Status) Error() string {
        return s.String()
}

// Positive returns true if the status code is 2xx.
func (s *Status) Positive() bool {
        return s.Code/100 == 2
}

// Temporary returns true if the status code is 4xx.
func (s *Status) Temporary() bool {
        return s.Code/100 == 4
}

// Permanent returns true if the status code is 5xx.
func (s *Status) Permanent() bool {
        return s.Code/100 == 5
}

// Text returns all lines joined by \n in a single string.
func (s *Status) Text() string {
        return strings.Join(s.Lines, "\n")
}

// String returns the status as a formatted message.
// Important: Should not be used for sending a response.
func (s *Status) String() string {
        base := fmt.Sprintf("%03d", s.Code)
        if s.EnhancedCode != NoEnhancedCode {
                base += fmt.Sprintf(" %d.%d.%d", s.EnhancedCode[0], s.EnhancedCode[1], s.EnhancedCode[2])
        }
        if len(s.Lines) > 0 {
                return base + " " + s.Text()
        }
        return base
}

// StatusWriter is an interface which needs to be implemented by a writer to be used by Status.WriteTo.
type StatusWriter interface {
        io.ByteWriter
        io.StringWriter
        io.Writer
}

// WriteTo writes the smtp status reply. Every line is appended to the free
// buffer of w if it has one (e.g. bufio.Writer) and written at once, so
// writing a reply doesn't allocate.
func (s *Status) WriteTo(w StatusWriter) (int64, error) {
        lines := s.Lines
        if len(lines) == 0 {
                lines = noLines
        }

        var n int64
        for i, m := range lines {
                var b []byte
                if ab, ok := w.(interface{ AvailableBuffer() []byte }); ok {
                        b = ab.AvailableBuffer()
                }
                b = strconv.AppendInt(b, int64(s.Code), 10)
                // RFC 5321 permits omitting the space when there is no text, but
                // RFC 4954 requires it for the 334 challenge and net/textproto
                // rejects any reply shorter than four bytes. Always emit it.
                if i == len(lines)-1 {
                        b = append(b, ' ')
                } else {
                        b = append(b, '-')
                }
                b = s.EnhancedCode.appendPart(b)
                b = append(b, m...)
                b = append(b, '\r', '\n')

                p, err := w.Write(b)
                n += int64(p)
                if err != nil {
                        return n, err
                }
        }

        return n, nil
}

// noLines is written for a status without a line, a reply has at least one.
var noLines = []string{""}

// LogValue implements slog.LogValuer so that formatting a Status is deferred
// to the handler and skipped entirely when the record is dropped.
func (s *Status) LogValue() slog.Value {
        return slog.GroupValue(
                slog.Int("code", s.Code),
                slog.String("enhCode", fmt.Sprintf("%d.%d.%d", s.EnhancedCode[0], s.EnhancedCode[1], s.EnhancedCode[2])),
                slog.String("text", s.Text()),
        )
}

// A Buffer is a variable-sized buffer of bytes with [Buffer.Read] and [Buffer.Write] methods.
// The zero value for Buffer is an empty buffer ready to use.
type Buffer struct {
        buf []byte // contents are the bytes buf[off : len(buf)]
        off int    // read at &buf[off], write at &buf[len(buf)]
}

// empty reports whether the unread portion of the buffer is empty.
func (b *Buffer) empty() bool { return len(b.buf) <= b.off }

// Reset resets the buffer to be empty,
// but it retains the underlying storage for use by future writes.
// Reset is the same as [Buffer.Truncate](0).
func (b *Buffer) Reset() {
        b.buf = b.buf[:0]
        b.off = 0
}

// Read reads the next len(p) bytes from the buffer or until the buffer
// is drained. The return value n is the number of bytes read. If the
// buffer has no data to return, err is [io.EOF] (unless len(p) is zero);
// otherwise it is nil.
func (b *Buffer) Read(p []byte) (n int, err error) {
        if b.empty() {
                // Buffer is empty, reset to recover space.
                b.Reset()
                if len(p) == 0 {
                        return 0, nil
                }
                return 0, io.EOF
        }
        n = copy(p, b.buf[b.off:])
        b.off += n
        return n, nil
}

// NewBuffer creates and initializes a new [Buffer] using buf as its
// initial contents. The new [Buffer] takes ownership of buf, and the
// caller should not use buf after this call. NewBuffer is intended to
// prepare a [Buffer] to read existing data.
//
// In most cases, new([Buffer]) (or just declaring a [Buffer] variable) is
// sufficient to initialize a [Buffer].
//
// Buffer just implements a reader without any extra functionality.
func NewBuffer(buf []byte) io.Reader { return &Buffer{buf: buf} }

// GenX509KeyPair generates a self signed smtp server certificate with the given domain.
func GenX509KeyPair(domain string) (tls.Certificate, error) {
        now := time.Now()
        template := &x509.Certificate{
                SerialNumber: big.NewInt(now.Unix()),
                Subject: pkix.Name{
                        CommonName:         domain,
                        Country:            []string{"DE"},
                        Organization:       []string{"Testcompany"},
                        OrganizationalUnit: []string{"Development"},
                },
                NotBefore:             now.AddDate(0, 0, -1),
                NotAfter:              now.AddDate(999, 0, 0),
                SubjectKeyId:          []byte{113, 117, 105, 99, 107, 115, 101, 114, 118, 101},
                BasicConstraintsValid: true,
                ExtKeyUsage:           []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth, x509.ExtKeyUsageClientAuth},
                KeyUsage:              x509.KeyUsageKeyEncipherment | x509.KeyUsageDigitalSignature,
                DNSNames:              []string{domain},
        }

        priv, err := rsa.GenerateKey(rand.Reader, 2048)
        if err != nil {
                return tls.Certificate{}, err
        }

        cert, err := x509.CreateCertificate(rand.Reader, template, template,
                priv.Public(), priv)
        if err != nil {
                return tls.Certificate{}, err
        }

        var outCert tls.Certificate
        outCert.Certificate = append(outCert.Certificate, cert)
        outCert.PrivateKey = priv

        return outCert, nil
}

// FakeHandler handles a single command line. Returning false closes the connection.
type FakeHandler func(s *FakeSession, line string) bool

// FakeServer is a scripted SMTP server for client tests. It records every
// received command line (message content excluded) and answers with a default
// reply unless a handler for the verb is set: EHLO/HELO advertise Extensions,
// DATA consumes the message, QUIT closes the connection and everything else
// gets 250.
type FakeServer struct {
        Extensions []string               // advertised in the EHLO reply
        Handlers   map[string]FakeHandler // keyed by upper case verb, e.g. "STARTTLS"

        mu       sync.Mutex
        commands []string
}

// FakeSession is a single connection of a FakeServer.
type FakeSession struct {
        conn net.Conn
        r    *bufio.Reader
}

// Reply writes each line terminated by CRLF.
func (s *FakeSession) Reply(lines ...string) {
        for _, l := range lines {
                _, _ = s.conn.Write([]byte(l + "\r\n"))
        }
}

// ReadMessage consumes the message after a 354 reply up to the terminating dot line.
func (s *FakeSession) ReadMessage() error {
        for {
                l, err := s.r.ReadString('\n')
                if err != nil {
                        return err
                }
                if l == ".\r\n" {
                        return nil
                }
        }
}

// StartTLS upgrades the connection to TLS, the handshake runs on the next read.
func (s *FakeSession) StartTLS(cfg *tls.Config) {
        s.conn = tls.Server(s.conn, cfg)
        s.r = bufio.NewReader(s.conn)
}

// Commands returns all received command lines in order, without CRLF.
func (f *FakeServer) Commands() []string {
        f.mu.Lock()
        defer f.mu.Unlock()
        return slices.Clone(f.commands)
}

// Received reports whether a command with the given verb was received.
func (f *FakeServer) Received(verb string) bool {
        return slices.ContainsFunc(f.Commands(), func(c string) bool { return fakeVerb(c) == verb })
}

func fakeVerb(line string) string {
        verb, _, _ := strings.Cut(strings.ToUpper(line), " ")
        return verb
}

// Listen starts the server on a random local port and returns its address.
// The server is stopped when the test ends.
func (f *FakeServer) Listen(t testing.TB) string {
        l, err := net.Listen("tcp", "127.0.0.1:0")
        if err != nil {
                t.Fatal(err)
        }
        t.Cleanup(func() { _ = l.Close() })

        go func() {
                for {
                        conn, err := l.Accept()
                        if err != nil {
                                return
                        }
                        go f.serve(&FakeSession{conn: conn, r: bufio.NewReader(conn)})
                }
        }()
        return l.Addr().String()
}

func (f *FakeServer) serve(s *FakeSession) {
        defer func() { _ = s.conn.Close() }()

        s.Reply("220 fake ESMTP")
        for {
                line, err := s.r.ReadString('\n')
                if err != nil {
                        return
                }
                line = strings.TrimRight(line, "\r\n")
                verb := fakeVerb(line)

                f.mu.Lock()
                f.commands = append(f.commands, line)
                f.mu.Unlock()

                if h, ok := f.Handlers[verb]; ok {
                        if !h(s, line) {
                                return
                        }
                        continue
                }

                switch verb {
                case "EHLO", "HELO":
                        reply := []string{"fake"}
                        reply = append(reply, f.Extensions...)
                        for i, r := range reply {
                                sep := "-"
                                if i == len(reply)-1 {
                                        sep = " "
                                }
                                s.Reply("250" + sep + r)
                        }
                case "DATA":
                        s.Reply("354 go ahead")
                        if s.ReadMessage() != nil {
                                return
                        }
                        s.Reply("250 queued")
                case "QUIT":
                        s.Reply("221 bye")
                        return
                default:
                        s.Reply("250 ok")
                }
        }
}

        if !ok {
                return nil, ok
        }

        m, ok := i.(*tester.Mail)
        return m, ok
}
